CREATE POLICY group_members_read
ON public.group_members FOR SELECT TO authenticated
USING (
  parapost_groups_private.allowed(group_id, 'directory')
  AND (
    user_id = auth.uid()
    OR parapost_groups_private.allowed(group_id, 'admin')
  )
);

CREATE POLICY group_posts_read
ON public.group_posts FOR SELECT TO authenticated
USING (
  NOT is_deleted
  AND parapost_groups_private.allowed(group_id, 'read')
  AND parapost_groups_private.author_visible_to_viewer(author_id)
  AND (
    published
    OR (
      author_id = auth.uid()
      AND parapost_groups_private.allowed(group_id, 'member')
    )
  )
);

CREATE POLICY group_post_media_read
ON public.group_post_media FOR SELECT TO authenticated
USING (
  EXISTS (
    SELECT 1 FROM public.group_posts p
    WHERE p.id = group_post_media.post_id
      AND p.group_id = group_post_media.group_id
  )
);

CREATE POLICY groups_read
ON public.groups FOR SELECT TO authenticated
USING (
  parapost_groups_private.allowed(id, 'directory')
);

CREATE POLICY group_post_comments_read
ON public.group_post_comments FOR SELECT TO authenticated
USING (
  parapost_groups_private.author_visible_to_viewer(author_id)
  AND EXISTS (
    SELECT 1 FROM public.group_posts p
    WHERE p.id = group_post_comments.post_id
      AND p.published
      AND NOT p.is_deleted
      AND parapost_groups_private.allowed(p.group_id, 'read')
  )
);

CREATE POLICY group_post_likes_read
ON public.group_post_likes FOR SELECT TO authenticated
USING (
  parapost_groups_private.author_visible_to_viewer(user_id)
  AND EXISTS (
    SELECT 1 FROM public.group_posts p
    WHERE p.id = group_post_likes.post_id
      AND p.published
      AND NOT p.is_deleted
      AND parapost_groups_private.allowed(p.group_id, 'read')
  )
);

-- CREATORVERSE: PAGES AND POSTS
-- ================================================

CREATE POLICY creatorverse_page_read
ON public.creatorverse_pages FOR SELECT
TO anon, authenticated
USING (creatorverse_private.read_page(id));

CREATE POLICY creatorverse_page_owner_read
ON public.creatorverse_pages FOR SELECT
TO authenticated
USING (creatorverse_private.owns(id));

CREATE POLICY creatorverse_page_create
ON public.creatorverse_pages FOR INSERT
TO authenticated
WITH CHECK (
  creatorverse_private.write_enabled()
  AND owner_id = auth.uid()
  AND status = 'draft'
);

CREATE POLICY creatorverse_page_edit
ON public.creatorverse_pages FOR UPDATE
TO authenticated
USING (
  creatorverse_private.write_enabled()
  AND creatorverse_private.owns(id)
)
WITH CHECK (
  creatorverse_private.write_enabled()
  AND owner_id = auth.uid()
  AND status IN ('draft', 'active', 'deleted')
);

CREATE POLICY creatorverse_post_read
ON public.creatorverse_posts FOR SELECT
TO anon, authenticated
USING (creatorverse_private.read_post(id));

CREATE POLICY creatorverse_post_owner_read
ON public.creatorverse_posts FOR SELECT
TO authenticated
USING (creatorverse_private.owns(page_id));

CREATE POLICY creatorverse_post_create
ON public.creatorverse_posts FOR INSERT
TO authenticated
WITH CHECK (
  creatorverse_private.write_enabled()
  AND creatorverse_private.owns(page_id)
  AND author_id = auth.uid()
  AND status = 'draft'
);

CREATE POLICY creatorverse_post_edit
ON public.creatorverse_posts FOR UPDATE
TO authenticated
USING (
  creatorverse_private.write_enabled()
  AND creatorverse_private.owns(page_id)
  AND author_id = auth.uid()
  AND status = 'draft'
)
WITH CHECK (
  creatorverse_private.write_enabled()
  AND creatorverse_private.owns(page_id)
  AND author_id = auth.uid()
  AND status = 'draft'
);

CREATE POLICY creatorverse_media_owner_read
ON public.creatorverse_media FOR SELECT
TO authenticated
USING (creatorverse_private.owns(page_id));

CREATE POLICY cv_media_lock_delete
ON storage.objects AS RESTRICTIVE FOR DELETE
TO public
USING (bucket_id <> 'creatorverse-media');

CREATE POLICY cv_media_lock_insert
ON storage.objects AS RESTRICTIVE FOR INSERT
TO public
WITH CHECK (
  bucket_id <> 'creatorverse-media'
  OR (
    auth.uid() IS NOT NULL
    AND storage.allow_only_operation('object.upload')
    AND creatorverse_private.storage_allowed(name, 'upload')
  )
);

CREATE POLICY cv_media_lock_select
ON storage.objects AS RESTRICTIVE FOR SELECT
TO public
USING (bucket_id <> 'creatorverse-media');

CREATE POLICY cv_media_lock_update
ON storage.objects AS RESTRICTIVE FOR UPDATE
TO public
USING (bucket_id <> 'creatorverse-media')
WITH CHECK (bucket_id <> 'creatorverse-media');

CREATE POLICY cv_media_owner_upload
ON storage.objects FOR INSERT
TO authenticated
WITH CHECK (
  bucket_id = 'creatorverse-media'
  AND storage.allow_only_operation('object.upload')
  AND creatorverse_private.storage_allowed(name, 'upload')
);
