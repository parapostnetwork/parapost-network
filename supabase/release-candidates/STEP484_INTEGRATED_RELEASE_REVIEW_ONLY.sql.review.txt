CREATE TABLE public.creatorverse_pages (
  id uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  owner_id uuid NOT NULL REFERENCES auth.users(id),
  name text NOT NULL CHECK (char_length(btrim(name)) BETWEEN 3 AND 100),
  category text NOT NULL CHECK (
    category IN (
      'business','creator','investigation_team','podcast',
      'entertainment','organization','community_project','other'
    )
  ),
  short_bio text NOT NULL DEFAULT ''
    CHECK (char_length(short_bio) <= 240),
  description text NOT NULL DEFAULT ''
    CHECK (char_length(description) <= 5000),
  website text CHECK (
    website IS NULL OR (
      char_length(website) <= 2048 AND
      website ~ '^https?://[^[:space:]@/]+([/?#][^[:space:]]*)?$'
    )
  ),
  social_links jsonb NOT NULL DEFAULT '[]'::jsonb
    CHECK (
      jsonb_typeof(social_links) = 'array'
      AND jsonb_array_length(social_links) <= 8
    ),
  public_email text CHECK (
    public_email IS NULL OR (
      char_length(public_email) <= 254 AND
      public_email ~ '^[^[:space:]@]+@[^[:space:]@]+\.[^[:space:]@]+$'
    )
  ),
  location text NOT NULL DEFAULT ''
    CHECK (char_length(location) <= 160),
  logo_media_id uuid,
  cover_media_id uuid,
  status text NOT NULL DEFAULT 'draft'
    CHECK (status IN ('draft','active','suspended','deleted')),
  created_at timestamptz NOT NULL DEFAULT now(),
  updated_at timestamptz NOT NULL DEFAULT now(),
  search_document tsvector GENERATED ALWAYS AS (
    to_tsvector(
      'simple'::regconfig,
      name || ' ' || short_bio || ' ' || description
    )
  ) STORED,
  UNIQUE (id, owner_id)
);

CREATE TABLE public.creatorverse_posts (
  id uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  page_id uuid NOT NULL REFERENCES public.creatorverse_pages(id),
  author_id uuid NOT NULL REFERENCES auth.users(id),
  content text NOT NULL DEFAULT ''
    CHECK (char_length(content) <= 10000),
  status text NOT NULL DEFAULT 'draft'
    CHECK (status IN ('draft','published','deleted')),
  comments_enabled boolean NOT NULL DEFAULT true,
  created_at timestamptz NOT NULL DEFAULT now(),
  updated_at timestamptz NOT NULL DEFAULT now(),
  UNIQUE (page_id, id),
  FOREIGN KEY (page_id, author_id)
    REFERENCES public.creatorverse_pages(id, owner_id)
);

CREATE TABLE public.creatorverse_media (
  id uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  page_id uuid NOT NULL REFERENCES public.creatorverse_pages(id),
  post_id uuid,
  purpose text NOT NULL
    CHECK (purpose IN ('logo','cover','post')),
  media_type text NOT NULL
    CHECK (media_type IN ('image','video')),
  storage_path text NOT NULL UNIQUE,
  mime_type text NOT NULL CHECK (
    mime_type IN (
      'image/jpeg','image/png','image/webp',
      'video/mp4','video/webm'
    )
  ),
  verification text NOT NULL DEFAULT 'pending'
    CHECK (verification IN ('pending','verified','rejected')),
  byte_size bigint CHECK (byte_size > 0),
  duration_seconds numeric CHECK (
    duration_seconds > 0 AND duration_seconds <= 120
  ),
  display_order smallint NOT NULL DEFAULT 0
    CHECK (display_order BETWEEN 0 AND 3),
  created_at timestamptz NOT NULL DEFAULT now(),
  UNIQUE (page_id, id),
  FOREIGN KEY (page_id, post_id)
    REFERENCES public.creatorverse_posts(page_id, id),
  CHECK (storage_path LIKE page_id::text || '/%'),
  CHECK (
    char_length(storage_path) <= 240
    AND storage_path NOT LIKE '%..%'
    AND storage_path NOT LIKE '%//%'
  ),
  CHECK (
    (purpose = 'post' AND post_id IS NOT NULL)
    OR (
      purpose IN ('logo','cover')
      AND post_id IS NULL
      AND media_type = 'image'
    )
  ),
  CHECK (
    (media_type = 'image'
      AND mime_type IN ('image/jpeg','image/png','image/webp')
      AND duration_seconds IS NULL
      AND (byte_size IS NULL OR byte_size <= 8388608))
    OR
    (media_type = 'video'
      AND mime_type IN ('video/mp4','video/webm')
      AND (byte_size IS NULL OR byte_size <= 33554432))
  ),
  CHECK (
    verification <> 'verified'
    OR (
      byte_size IS NOT NULL
      AND (media_type = 'image' OR duration_seconds IS NOT NULL)
    )
  )
);

ALTER TABLE public.creatorverse_pages

CREATE OR REPLACE FUNCTION public.get_parachat_inbox_summaries()
RETURNS TABLE (
  conversation_id uuid,
  last_message_id uuid,
  last_sender_id uuid,
  last_body text,
  last_created_at timestamptz,
  last_is_read boolean,
  last_message_type text,
  last_image_path text,
  last_image_mime_type text,
  last_image_size_bytes integer,
  last_image_width integer,
  last_image_height integer,
  unread_count bigint
)
LANGUAGE sql
STABLE
SECURITY INVOKER
SET search_path = public
AS $$
  WITH my_conversations AS (
    SELECT dc.id
    FROM public.direct_conversations dc
    WHERE auth.uid() IS NOT NULL
      AND (
        dc.user_one_id = auth.uid()
        OR dc.user_two_id = auth.uid()
      )
  )
  SELECT
    mc.id AS conversation_id,
    latest.id AS last_message_id,
    latest.sender_id AS last_sender_id,
    latest.body AS last_body,
    latest.created_at AS last_created_at,
    latest.is_read AS last_is_read,
    latest.message_type AS last_message_type,
    latest.image_path AS last_image_path,
    latest.image_mime_type AS last_image_mime_type,
    latest.image_size_bytes AS last_image_size_bytes,
    latest.image_width AS last_image_width,
    latest.image_height AS last_image_height,
    COALESCE(unread.unread_count, 0)::bigint AS unread_count
  FROM my_conversations mc

  LEFT JOIN LATERAL (
    SELECT
      dm.id,
      dm.sender_id,
      dm.body,
      dm.created_at,
      dm.is_read,
      dm.message_type,
      dm.image_path,
      dm.image_mime_type,
      dm.image_size_bytes,
      dm.image_width,
      dm.image_height
    FROM public.direct_messages dm
    WHERE dm.conversation_id = mc.id
    ORDER BY dm.created_at DESC, dm.id DESC
    LIMIT 1
  ) latest ON true

  LEFT JOIN LATERAL (
    SELECT COUNT(*)::bigint AS unread_count
    FROM public.direct_messages dm
    WHERE dm.conversation_id = mc.id
      AND dm.sender_id <> auth.uid()
      AND dm.is_read = false
  ) unread ON true;
$$;

CREATE OR REPLACE FUNCTION public.create_comment_like_notification()
RETURNS trigger
LANGUAGE plpgsql
SET search_path TO 'public', 'pg_temp'
AS $function$
declare
  comment_owner_id uuid;
  comment_post_id uuid;
begin
  select
    comments.user_id,
    comments.post_id
  into
    comment_owner_id,
    comment_post_id
  from public.comments
  where comments.id = new.comment_id;

  if comment_owner_id is not null
     and comment_owner_id <> new.user_id then

    insert into public.notifications (
      user_id,
      actor_id,
      type,
      post_id,
      comment_id,
      message
    )
    values (
      comment_owner_id,
      new.user_id,
      'comment_like',
      comment_post_id,
      new.comment_id,
      'liked your comment'
    )
    on conflict do nothing;

  end if;

  return new;
end;
$function$;

CREATE OR REPLACE FUNCTION public.create_comment_reply_notification()
RETURNS trigger
LANGUAGE plpgsql
SET search_path TO 'public', 'pg_temp'
AS $function$
declare
  reply_target_id uuid;
  post_owner_id uuid;
begin
  if new.parent_comment_id is null then
    return new;
  end if;

  reply_target_id := new.reply_to_user_id;

  if reply_target_id is null then
    select comments.user_id
    into reply_target_id
    from public.comments
    where comments.id = new.parent_comment_id;
  end if;

  select posts.user_id
  into post_owner_id
  from public.posts
  where posts.id = new.post_id;

  if reply_target_id is not null
     and reply_target_id <> new.user_id then

    insert into public.notifications (
      user_id,
      actor_id,
      type,
      post_id,
      comment_id,
      message
    )
    values (
      reply_target_id,
      new.user_id,
      'comment_reply',
      new.post_id,
      new.id,
      'replied to your comment'
    )
    on conflict do nothing;

  end if;

  if post_owner_id is not null
     and post_owner_id <> new.user_id
     and post_owner_id is distinct from reply_target_id then

    insert into public.notifications (
      user_id,
      actor_id,
      type,
      post_id,
      comment_id,
      message
    )
    values (
      post_owner_id,
      new.user_id,
      'comment_reply',
      new.post_id,
      new.id,
      'replied to a comment on your post'
    )
    on conflict do nothing;

  end if;

  return new;
end;
$function$;

CREATE OR REPLACE FUNCTION public.sync_post_share_notification()
RETURNS trigger
LANGUAGE plpgsql
SECURITY DEFINER
SET search_path TO 'public', 'pg_temp'
AS $function$
DECLARE
  old_active boolean := false;
  new_active boolean := false;
  should_create boolean := false;
  post_owner_id uuid;
BEGIN

  -- ----------------------------------------------------------
  -- Determine previous active state when OLD exists.
  -- ----------------------------------------------------------

  IF TG_OP <> 'INSERT' THEN
    old_active :=
      OLD.deleted_at IS NULL
      AND COALESCE(OLD.share_destination, 'feed') = 'feed'
      AND OLD.user_id IS NOT NULL
      AND OLD.post_id IS NOT NULL;
  END IF;


  -- ----------------------------------------------------------
  -- Determine new active state when NEW exists.
  -- ----------------------------------------------------------

  IF TG_OP <> 'DELETE' THEN
    new_active :=
      NEW.deleted_at IS NULL
      AND COALESCE(NEW.share_destination, 'feed') = 'feed'
      AND NEW.user_id IS NOT NULL
      AND NEW.post_id IS NOT NULL;
  END IF;


  -- ----------------------------------------------------------
  -- HARD DELETE
  --
  -- If the removed row was the final active share for this
  -- actor + post, remove its notification.
  -- ----------------------------------------------------------

  IF TG_OP = 'DELETE' THEN

    IF old_active
       AND NOT EXISTS (
         SELECT 1
         FROM public.shares AS shares
         WHERE shares.user_id = OLD.user_id
           AND shares.post_id = OLD.post_id
           AND shares.deleted_at IS NULL
           AND COALESCE(shares.share_destination, 'feed') = 'feed'
       )
    THEN

      DELETE FROM public.notifications
      WHERE type IN ('post_share', 'share')
        AND actor_id = OLD.user_id
        AND post_id = OLD.post_id;

    END IF;

    RETURN OLD;
  END IF;


  -- ----------------------------------------------------------
  -- UPDATE
  --
  -- Clean the old notification when an active share becomes
  -- inactive or changes actor/post.
  -- ----------------------------------------------------------

  IF TG_OP = 'UPDATE' THEN

    IF old_active
       AND (
         NOT new_active
         OR OLD.user_id IS DISTINCT FROM NEW.user_id
         OR OLD.post_id IS DISTINCT FROM NEW.post_id
       )
       AND NOT EXISTS (
         SELECT 1
         FROM public.shares AS shares
         WHERE shares.user_id = OLD.user_id
           AND shares.post_id = OLD.post_id
           AND shares.deleted_at IS NULL
           AND COALESCE(shares.share_destination, 'feed') = 'feed'
       )
    THEN

      DELETE FROM public.notifications
      WHERE type IN ('post_share', 'share')
        AND actor_id = OLD.user_id
        AND post_id = OLD.post_id;

    END IF;


    IF new_active
       AND (
         NOT old_active
         OR OLD.user_id IS DISTINCT FROM NEW.user_id
         OR OLD.post_id IS DISTINCT FROM NEW.post_id
       )
    THEN
      should_create := true;
    END IF;

  END IF;


  -- ----------------------------------------------------------
  -- INSERT
  -- ----------------------------------------------------------

  IF TG_OP = 'INSERT' AND new_active THEN
    should_create := true;
  END IF;


  -- ----------------------------------------------------------
  -- CREATE NOTIFICATION
  -- ----------------------------------------------------------

CREATE OR REPLACE FUNCTION public.enforce_comment_share_context()
RETURNS trigger
LANGUAGE plpgsql
SET search_path TO 'public', 'pg_temp'
AS $function$
DECLARE
  parent_post_id uuid;
  parent_share_id uuid;
  reply_post_id uuid;
  reply_share_id uuid;
BEGIN

  -- ----------------------------------------------------------
  -- Existing comments may not be moved to another post/thread.
  -- ----------------------------------------------------------

  IF TG_OP = 'UPDATE' THEN
    IF NEW.post_id IS DISTINCT FROM OLD.post_id
       OR NEW.share_id IS DISTINCT FROM OLD.share_id
       OR NEW.parent_comment_id IS DISTINCT FROM OLD.parent_comment_id
    THEN
      RAISE EXCEPTION 'Comment discussion context cannot be changed.';
    END IF;
  END IF;


  -- ----------------------------------------------------------
  -- Replies inherit the exact parent discussion context.
  -- ----------------------------------------------------------

  IF NEW.parent_comment_id IS NOT NULL THEN

    SELECT
      comments.post_id,
      comments.share_id
    INTO
      parent_post_id,
      parent_share_id
    FROM public.comments AS comments
    WHERE comments.id = NEW.parent_comment_id;

    IF NOT FOUND THEN
      RAISE EXCEPTION 'Parent comment does not exist.';
    END IF;

    IF NEW.post_id IS DISTINCT FROM parent_post_id THEN
      RAISE EXCEPTION 'Reply post does not match parent comment.';
    END IF;

    NEW.share_id := parent_share_id;

  END IF;


  -- ----------------------------------------------------------
  -- Exact reply targets must belong to the same discussion.
  -- ----------------------------------------------------------

  IF NEW.reply_to_comment_id IS NOT NULL THEN

    SELECT
      comments.post_id,
      comments.share_id
    INTO
      reply_post_id,
      reply_share_id
    FROM public.comments AS comments
    WHERE comments.id = NEW.reply_to_comment_id;

    IF NOT FOUND THEN
      RAISE EXCEPTION 'Reply target comment does not exist.';
    END IF;

    IF NEW.post_id IS DISTINCT FROM reply_post_id
       OR NEW.share_id IS DISTINCT FROM reply_share_id
    THEN
      RAISE EXCEPTION 'Reply target belongs to another discussion.';
    END IF;

  END IF;


  -- ----------------------------------------------------------
  -- A share-specific comment must reference an active feed share
  -- for the exact same original post.
  -- ----------------------------------------------------------

  IF NEW.share_id IS NOT NULL
     AND NOT EXISTS (
       SELECT 1
       FROM public.shares AS shares
       WHERE shares.id = NEW.share_id
         AND shares.post_id = NEW.post_id
         AND shares.deleted_at IS NULL
         AND COALESCE(shares.share_destination, 'feed') = 'feed'
     )
  THEN
    RAISE EXCEPTION 'Shared post discussion is not active.';
  END IF;


  RETURN NEW;
END;
$function$;

CREATE OR REPLACE FUNCTION public.sync_post_share_notification()
RETURNS trigger
LANGUAGE plpgsql
SECURITY DEFINER
SET search_path TO 'public', 'pg_temp'
AS $function$
DECLARE
  old_active boolean := false;
  new_active boolean := false;
  should_create boolean := false;
  post_owner_id uuid;
BEGIN

  IF TG_OP <> 'INSERT' THEN

    old_active :=
      OLD.deleted_at IS NULL
      AND COALESCE(OLD.share_destination, 'feed') = 'feed'
      AND OLD.user_id IS NOT NULL
      AND OLD.post_id IS NOT NULL;

  END IF;


  IF TG_OP <> 'DELETE' THEN

    new_active :=
      NEW.deleted_at IS NULL
      AND COALESCE(NEW.share_destination, 'feed') = 'feed'
      AND NEW.user_id IS NOT NULL
      AND NEW.post_id IS NOT NULL;

  END IF;


  -- ----------------------------------------------------------
  -- HARD DELETE
  -- ----------------------------------------------------------

  IF TG_OP = 'DELETE' THEN

    IF old_active THEN

      DELETE FROM public.notifications
      WHERE type IN ('post_share', 'share')
        AND (
          share_id = OLD.id

          OR (
            share_id IS NULL
            AND actor_id = OLD.user_id
            AND post_id = OLD.post_id
          )
        );

    END IF;


    RETURN OLD;
  END IF;


  -- ----------------------------------------------------------
  -- UPDATE
  -- ----------------------------------------------------------

  IF TG_OP = 'UPDATE' THEN

    IF old_active
       AND (
         NOT new_active
         OR OLD.user_id IS DISTINCT FROM NEW.user_id
         OR OLD.post_id IS DISTINCT FROM NEW.post_id
       )
    THEN

      DELETE FROM public.notifications
      WHERE type IN ('post_share', 'share')
        AND (
          share_id = OLD.id

          OR (
            share_id IS NULL
            AND actor_id = OLD.user_id
            AND post_id = OLD.post_id
          )
        );

    END IF;


    IF new_active
       AND (
         NOT old_active
         OR OLD.user_id IS DISTINCT FROM NEW.user_id
         OR OLD.post_id IS DISTINCT FROM NEW.post_id
       )
    THEN
      should_create := true;
    END IF;

  END IF;


  -- ----------------------------------------------------------
  -- INSERT
  -- ----------------------------------------------------------

  IF TG_OP = 'INSERT'
     AND new_active
  THEN
    should_create := true;
  END IF;


  -- ----------------------------------------------------------
  -- CREATE EXACT SHARE NOTIFICATION
  -- ----------------------------------------------------------

  IF should_create THEN

    SELECT posts.user_id
    INTO post_owner_id
    FROM public.posts AS posts
    WHERE posts.id = NEW.post_id;


    IF post_owner_id IS NOT NULL
       AND post_owner_id <> NEW.user_id
    THEN

      INSERT INTO public.notifications (
        user_id,
        actor_id,
        type,
        post_id,
        share_id,
        message,
        is_read
      )
      VALUES (
        post_owner_id,
        NEW.user_id,
        'post_share',
        NEW.post_id,
        NEW.id,
        'shared your post.',
        false
      )
      ON CONFLICT DO NOTHING;

    END IF;

  END IF;


  RETURN NEW;
END;

CREATE OR REPLACE FUNCTION public.sync_reel_share_count()
RETURNS trigger
LANGUAGE plpgsql
SECURITY DEFINER
SET search_path TO ''
AS $$
BEGIN
  IF TG_OP = 'INSERT' THEN
    UPDATE public.reels
    SET shares = coalesce(shares, 0) + 1
    WHERE id = NEW.reel_id;

    RETURN NEW;
  END IF;

  IF TG_OP = 'DELETE' THEN
    UPDATE public.reels
    SET shares = greatest(coalesce(shares, 0) - 1, 0)
    WHERE id = OLD.reel_id;

    RETURN OLD;
  END IF;

  -- Only a Reel move changes the number of shares belonging
  -- to either Reel. Other UPDATE operations leave counts alone.
  IF OLD.reel_id IS DISTINCT FROM NEW.reel_id THEN
    UPDATE public.reels
    SET shares = greatest(coalesce(shares, 0) - 1, 0)
    WHERE id = OLD.reel_id;

    UPDATE public.reels
    SET shares = coalesce(shares, 0) + 1
    WHERE id = NEW.reel_id;
  END IF;

  RETURN NEW;
END;
$$;

CREATE OR REPLACE FUNCTION public.parapost_guard_message_updates()
RETURNS trigger
LANGUAGE plpgsql
SET search_path = ''
AS $$
DECLARE
  v_user_id uuid;
BEGIN
  -- Allow trusted service-role operations.
  IF auth.role() = 'service_role' THEN
    RETURN NEW;
  END IF;

  -- Reject anonymous or unauthenticated updates.
  IF auth.role() IS DISTINCT FROM 'authenticated' THEN
    RAISE EXCEPTION 'Authentication required';
  END IF;

  v_user_id := auth.uid();

  IF v_user_id IS NULL THEN
    RAISE EXCEPTION 'Authentication required';
  END IF;

  -- The original sender may edit the message body only.
  IF OLD.sender_id = v_user_id THEN
    IF (to_jsonb(NEW) - 'body') IS DISTINCT FROM
       (to_jsonb(OLD) - 'body') THEN
      RAISE EXCEPTION 'Message sender may update body only';
    END IF;

    RETURN NEW;
  END IF;

  -- A recipient may mark an incoming message as read only.
  IF public.is_direct_conversation_participant(
       OLD.conversation_id,
       v_user_id
     )
     AND OLD.sender_id <> v_user_id
     AND OLD.is_read IS DISTINCT FROM TRUE
     AND NEW.is_read IS TRUE
     AND (to_jsonb(NEW) - 'is_read') IS NOT DISTINCT FROM
         (to_jsonb(OLD) - 'is_read') THEN
    RETURN NEW;
  END IF;

  RAISE EXCEPTION 'Message update not permitted';
END;

CREATE FUNCTION parapost_groups_private.request_membership(
  p_group_id uuid
)
RETURNS text
LANGUAGE plpgsql
SECURITY DEFINER
SET search_path = ''
AS $function$
DECLARE
  v_user_id uuid := auth.uid();
  v_visibility text;
  v_owner_id uuid;
  v_status text;
BEGIN
  IF v_user_id IS NULL THEN
    RAISE EXCEPTION 'Authentication required'
      USING ERRCODE = '28000';
  END IF;

  SELECT g.visibility, g.owner_id
    INTO v_visibility, v_owner_id
  FROM public.groups g
  WHERE g.id = p_group_id
  FOR SHARE;

  IF NOT FOUND THEN
    RAISE EXCEPTION 'Group unavailable';
  END IF;

  IF EXISTS (
    SELECT 1
    FROM public.blocked_users b
    WHERE
      (b.blocker_id = v_user_id
       AND b.blocked_id = v_owner_id)
      OR
      (b.blocked_id = v_user_id
       AND b.blocker_id = v_owner_id)
  ) THEN
    RAISE EXCEPTION 'Membership unavailable';
  END IF;

  INSERT INTO public.group_members (
    group_id,
    user_id,
    role,
    status,
    approved_at
  )
  VALUES (
    p_group_id,
    v_user_id,
    'member',
    CASE
      WHEN v_visibility = 'public'
      THEN 'approved'
      ELSE 'pending'
    END,
    CASE
      WHEN v_visibility = 'public'
      THEN now()
      ELSE NULL
    END
  )
  ON CONFLICT (group_id, user_id)
  DO NOTHING;

  SELECT gm.status
    INTO v_status
  FROM public.group_members gm
  WHERE gm.group_id = p_group_id
    AND gm.user_id = v_user_id;

  IF v_status NOT IN ('approved', 'pending')
     OR v_status IS NULL THEN
    RAISE EXCEPTION 'Membership unavailable';
  END IF;

  RETURN v_status;
END;
$function$;

CREATE FUNCTION parapost_groups_private.review_membership(
  p_group_id uuid,
  p_target_user_id uuid,
  p_decision text
)
RETURNS text
LANGUAGE plpgsql
SECURITY DEFINER
SET search_path = ''
AS $function$
DECLARE
  v_reviewer uuid := auth.uid();
  v_owner_id uuid;
  v_reviewer_role text;
  v_final_status text;
BEGIN
  IF v_reviewer IS NULL THEN
    RAISE EXCEPTION 'Authentication required';
  END IF;

  IF p_group_id IS NULL
     OR p_target_user_id IS NULL
     OR p_decision IS NULL
     OR p_decision NOT IN ('approved', 'rejected') THEN
    RAISE EXCEPTION 'Invalid membership review';
  END IF;

  SELECT g.owner_id
    INTO v_owner_id
  FROM public.groups g
  WHERE g.id = p_group_id
  FOR SHARE;

  IF NOT FOUND THEN
    RAISE EXCEPTION 'Group unavailable';
  END IF;

  SELECT gm.role
    INTO v_reviewer_role
  FROM public.group_members gm
  WHERE gm.group_id = p_group_id
    AND gm.user_id = v_reviewer
    AND gm.status = 'approved'
    AND gm.role IN ('owner', 'admin')
  FOR SHARE;

  IF NOT FOUND THEN
    RAISE EXCEPTION 'Not authorized to review membership';
  END IF;

  IF p_decision = 'approved' AND EXISTS (
    SELECT 1
    FROM public.blocked_users b
    WHERE
      (
        b.blocker_id = p_target_user_id
        AND b.blocked_id IN (v_owner_id, v_reviewer)
      )
      OR
      (
        b.blocked_id = p_target_user_id
        AND b.blocker_id IN (v_owner_id, v_reviewer)
      )
  ) THEN
    RAISE EXCEPTION 'Membership unavailable';
  END IF;

  UPDATE public.group_members gm
  SET
    status = p_decision,
    approved_at = CASE
      WHEN p_decision = 'approved'
      THEN now()
      ELSE NULL
    END
  WHERE gm.group_id = p_group_id
    AND gm.user_id = p_target_user_id
    AND gm.role = 'member'
    AND gm.status = 'pending'
  RETURNING gm.status INTO v_final_status;

  IF NOT FOUND THEN
    RAISE EXCEPTION 'Pending membership unavailable';
  END IF;

  RETURN v_final_status;
END;
$function$;

CREATE FUNCTION parapost_groups_private.command(action text, gid uuid, payload jsonb)
RETURNS jsonb LANGUAGE plpgsql SECURITY DEFINER SET search_path = '' AS $$
DECLARE
 uid uuid := auth.uid(); g public.groups; m public.group_members; p public.group_posts;
 target uuid; pid uuid; mid uuid; path text; mime text; role_name text; result jsonb; item record;
BEGIN
 IF NOT parapost_groups_private.enabled() THEN RAISE EXCEPTION 'Groups unavailable' USING ERRCODE='42501'; END IF;
 IF action='create' THEN
   IF (SELECT count(*) FROM public.groups WHERE owner_id=uid)>=100 THEN RAISE EXCEPTION 'Group creation limit reached'; END IF;
   INSERT INTO public.groups(owner_id,name,description,visibility)
   VALUES(uid,btrim(payload->>'name'),coalesce(payload->>'description',''),coalesce(payload->>'visibility','public')) RETURNING * INTO g;
   RETURN jsonb_build_object('id',g.id);
 END IF;
 -- Group lock serializes ALL writes, including membership removal vs publish/role changes.
 SELECT * INTO g FROM public.groups WHERE id=gid FOR UPDATE;
 IF NOT FOUND OR NOT parapost_groups_private.allowed(gid,'directory') THEN RAISE EXCEPTION 'Group unavailable' USING ERRCODE='42501'; END IF;
 SELECT * INTO m FROM public.group_members WHERE group_id=gid AND user_id=uid;
 IF action='join' THEN
   IF m.status IS NOT NULL THEN
     IF m.status NOT IN ('pending','approved') THEN RAISE EXCEPTION 'Membership unavailable'; END IF;
   ELSE
     INSERT INTO public.group_members(group_id,user_id,role,status,approved_at)
     VALUES(gid,uid,'member',CASE WHEN g.visibility='public' THEN 'approved' ELSE 'pending' END,CASE WHEN g.visibility='public' THEN now() END);
   END IF;
 ELSIF action='leave' THEN
   IF m.role='owner' THEN RAISE EXCEPTION 'Owner must delete the Group; transfer is not supported'; END IF;
   -- Only voluntary departures remove the row. Removed/banned/rejected rows persist.
   DELETE FROM public.group_members WHERE group_id=gid AND user_id=uid AND status IN ('pending','approved');
 ELSIF action IN ('approve','reject','remove','ban','role','restore') THEN
   target := (payload->>'user_id')::uuid;
   IF NOT parapost_groups_private.allowed(gid,'admin') OR target=uid OR target=g.owner_id THEN RAISE EXCEPTION 'Not authorized' USING ERRCODE='42501'; END IF;
   SELECT * INTO m FROM public.group_members WHERE group_id=gid AND user_id=target;
   IF NOT FOUND THEN RAISE EXCEPTION 'Membership unavailable'; END IF;
   IF NOT parapost_groups_private.allowed(gid,'owner') AND (m.role <> 'member' OR action IN ('role','restore')) THEN RAISE EXCEPTION 'Owner permission required'; END IF;
   IF action IN ('approve','restore') AND parapost_groups_private.blocked(uid,target) THEN RAISE EXCEPTION 'Membership unavailable'; END IF;
   IF action IN ('approve','restore') AND parapost_groups_private.blocked(g.owner_id,target) THEN RAISE EXCEPTION 'Membership unavailable'; END IF;
   IF action IN ('approve','reject') THEN
     IF m.status <> 'pending' THEN RAISE EXCEPTION 'Request is no longer pending'; END IF;
     UPDATE public.group_members SET status=CASE WHEN action='approve' THEN 'approved' ELSE 'rejected' END,
       approved_at=CASE WHEN action='approve' THEN now() END WHERE group_id=gid AND user_id=target;
   ELSIF action='role' THEN
     role_name := payload->>'role';
     IF m.status <> 'approved' OR role_name IS NULL OR role_name NOT IN ('admin','moderator','member') THEN RAISE EXCEPTION 'Invalid role change'; END IF;
     UPDATE public.group_members SET role=role_name WHERE group_id=gid AND user_id=target;
   ELSIF action='restore' THEN
     IF m.status NOT IN ('removed','rejected','banned') THEN RAISE EXCEPTION 'Invalid restoration'; END IF;
     -- Restoration requires owner action AND a fresh approval, never immediate access.
     UPDATE public.group_members SET role='member',status='pending',approved_at=NULL WHERE group_id=gid AND user_id=target;
   ELSE
     UPDATE public.group_members SET role='member',status=CASE WHEN action='ban' THEN 'banned' ELSE 'removed' END,approved_at=NULL WHERE group_id=gid AND user_id=target;
   END IF;
 ELSIF action='edit' THEN
   IF NOT parapost_groups_private.allowed(gid,'admin') THEN RAISE EXCEPTION 'Not authorized'; END IF;
   -- Privacy changes are owner-only. Existing approvals survive; pending requests stay pending.
   IF payload->>'visibility' IS DISTINCT FROM g.visibility AND NOT parapost_groups_private.allowed(gid,'owner') THEN RAISE EXCEPTION 'Owner permission required'; END IF;
   UPDATE public.groups SET name=btrim(payload->>'name'),description=coalesce(payload->>'description',''),visibility=payload->>'visibility',updated_at=now() WHERE id=gid;
 ELSIF action='delete' THEN
   IF NOT parapost_groups_private.allowed(gid,'owner') OR payload->>'confirmation' IS DISTINCT FROM g.name THEN RAISE EXCEPTION 'Owner and Group name confirmation required'; END IF;
   DELETE FROM public.groups WHERE id=gid;
 ELSIF action='draft' THEN
   IF NOT parapost_groups_private.allowed(gid,'member') THEN RAISE EXCEPTION 'Approved membership required'; END IF;
   IF (SELECT count(*) FROM public.group_posts WHERE group_id=gid AND author_id=uid AND NOT published)>=10 THEN RAISE EXCEPTION 'Too many unfinished uploads'; END IF;
   INSERT INTO public.group_posts(group_id,author_id,content) VALUES(gid,uid,coalesce(payload->>'content','')) RETURNING id INTO pid;
   RETURN jsonb_build_object('id',pid);
 ELSIF action IN ('reserve_media','publish','edit_post','delete_post') THEN
   pid := (payload->>'post_id')::uuid;
   SELECT * INTO p FROM public.group_posts WHERE id=pid AND group_id=gid;
   IF NOT FOUND OR p.is_deleted THEN RAISE EXCEPTION 'Post unavailable'; END IF;
   IF NOT parapost_groups_private.allowed(gid,'member') OR (p.author_id<>uid AND NOT (action='delete_post' AND parapost_groups_private.allowed(gid,'moderate'))) THEN RAISE EXCEPTION 'Not authorized'; END IF;
   IF action='delete_post' THEN
     DELETE FROM public.group_posts WHERE id=pid;
   ELSIF action='edit_post' THEN
     IF char_length(btrim(coalesce(payload->>'content','')))=0 AND NOT EXISTS(SELECT 1 FROM public.group_post_media WHERE post_id=pid) THEN RAISE EXCEPTION 'Post cannot be empty'; END IF;
     UPDATE public.group_posts SET content=coalesce(payload->>'content',''),updated_at=now() WHERE id=pid;
   ELSIF action='reserve_media' THEN
     IF p.published OR (SELECT count(*) FROM public.group_post_media WHERE post_id=pid)>=4 THEN RAISE EXCEPTION 'Media limit or published post'; END IF;
     mime := payload->>'mime_type';
     IF mime IS NULL OR mime NOT IN ('image/jpeg','image/png','image/webp','video/mp4','video/webm') THEN RAISE EXCEPTION 'Unsupported media'; END IF;
     mid := gen_random_uuid();
     path := gid::text || '/posts/' || pid::text || '/' || uid::text || '/' || mid::text;
     INSERT INTO public.group_post_media(id,group_id,post_id,uploader_id,media_type,storage_path,mime_type,display_order)
     VALUES(mid,gid,pid,uid,CASE WHEN mime LIKE 'image/%' THEN 'image' ELSE 'video' END,path,mime,(SELECT count(*) FROM public.group_post_media WHERE post_id=pid));
     RETURN jsonb_build_object('path',path);
   ELSE
     IF p.published THEN RETURN jsonb_build_object('id',pid); END IF;
     IF char_length(btrim(p.content))=0 AND NOT EXISTS(SELECT 1 FROM public.group_post_media WHERE post_id=pid) THEN RAISE EXCEPTION 'Post cannot be empty'; END IF;
     FOR item IN SELECT * FROM public.group_post_media WHERE post_id=pid LOOP
       IF NOT EXISTS(SELECT 1 FROM storage.objects o WHERE o.bucket_id='group-media' AND o.name=item.storage_path
         AND o.metadata->>'mimetype'=item.mime_type AND (o.metadata->>'size')::bigint BETWEEN 1 AND CASE WHEN item.media_type='image' THEN 8388608 ELSE 33554432 END) THEN RAISE EXCEPTION 'Media upload incomplete or invalid'; END IF;
     END LOOP;
     UPDATE public.group_posts SET published=true,updated_at=now() WHERE id=pid;
   END IF;
 ELSIF action='remove_cover' THEN
   IF NOT parapost_groups_private.allowed(gid,'admin') THEN RAISE EXCEPTION 'Not authorized'; END IF;
   UPDATE public.groups SET cover_storage_path=NULL,updated_at=now() WHERE id=gid;
 ELSIF action='reserve_cover' THEN
   IF NOT parapost_groups_private.allowed(gid,'admin') THEN RAISE EXCEPTION 'Not authorized'; END IF;
   IF (SELECT count(*) FROM parapost_groups_private.cover_uploads WHERE group_id=gid AND uploader_id=uid)>=10 THEN RAISE EXCEPTION 'Too many unfinished covers'; END IF;
   -- Covers are intentionally directory metadata, including those for private Groups.
   path := gid::text || '/covers/' || gen_random_uuid()::text;
   INSERT INTO parapost_groups_private.cover_uploads(storage_path,group_id,uploader_id) VALUES(path,gid,uid);
   RETURN jsonb_build_object('path',path);
 ELSIF action='set_cover' THEN
   path := payload->>'path';
   IF NOT parapost_groups_private.allowed(gid,'admin') OR NOT EXISTS(SELECT 1 FROM parapost_groups_private.cover_uploads c JOIN storage.objects o ON o.name=c.storage_path AND o.bucket_id='group-media'
     WHERE c.storage_path=path AND c.group_id=gid AND c.uploader_id=uid AND o.metadata->>'mimetype' IN ('image/jpeg','image/png','image/webp') AND (o.metadata->>'size')::bigint BETWEEN 1 AND 8388608) THEN RAISE EXCEPTION 'Invalid cover'; END IF;
   UPDATE public.groups SET cover_storage_path=path,updated_at=now() WHERE id=gid;
   DELETE FROM parapost_groups_private.cover_uploads WHERE storage_path=path;
 ELSE RAISE EXCEPTION 'Unknown Group action';
 END IF;
 RETURN jsonb_build_object('id',gid);
END; $$;

  p_group_id uuid, p_post_id uuid, p_action text,
  p_content text DEFAULT NULL, p_parent_id uuid DEFAULT NULL
) RETURNS jsonb
LANGUAGE plpgsql SECURITY DEFINER SET search_path='' AS $$
DECLARE
  v_uid uuid := auth.uid();
  v_group public.groups;
  v_post public.group_posts;
  v_parent public.group_post_comments;
  v_id uuid;
BEGIN
  IF v_uid IS NULL THEN
    RAISE EXCEPTION 'Sign in required' USING ERRCODE='42501';
  END IF;

  -- Serialize interactions with Group membership/privacy changes.
  SELECT * INTO v_group FROM public.groups
    WHERE id=p_group_id FOR SHARE;

  IF NOT FOUND OR NOT parapost_groups_private.allowed(p_group_id,'member') THEN
    RAISE EXCEPTION 'Approved membership required' USING ERRCODE='42501';
  END IF;

  SELECT * INTO v_post FROM public.group_posts
    WHERE id=p_post_id AND group_id=p_group_id
      AND published AND NOT is_deleted;

  IF NOT FOUND THEN
    RAISE EXCEPTION 'Post unavailable' USING ERRCODE='42501';
  END IF;

  -- Block interactions with posts from blocked accounts.
  IF p_action IN ('like','comment','reply','like_comment')
     AND parapost_groups_private.blocked(v_uid,v_post.author_id) THEN
    RAISE EXCEPTION 'Interaction unavailable'
      USING ERRCODE='42501';
  END IF;

  IF p_action='like' THEN
    INSERT INTO public.group_post_likes(post_id,user_id)
      VALUES(p_post_id,v_uid)
      ON CONFLICT (post_id,user_id) DO NOTHING;
    RETURN jsonb_build_object('liked',true);
  ELSIF p_action='unlike' THEN
    DELETE FROM public.group_post_likes
      WHERE post_id=p_post_id AND user_id=v_uid;
    RETURN jsonb_build_object('liked',false);
  ELSIF p_action IN ('like_comment','unlike_comment') THEN
    IF p_parent_id IS NULL THEN
      RAISE EXCEPTION 'Comment required';
    END IF;

    SELECT * INTO v_parent
      FROM public.group_post_comments
      WHERE id=p_parent_id AND post_id=p_post_id;

    IF NOT FOUND THEN
      RAISE EXCEPTION 'Comment unavailable'
        USING ERRCODE='42501';
    END IF;

    IF p_action='like_comment' THEN
      IF parapost_groups_private.blocked(
        v_uid,v_parent.author_id
      ) THEN
        RAISE EXCEPTION 'Comment Like unavailable'
          USING ERRCODE='42501';
      END IF;

      INSERT INTO public.group_comment_likes
        (comment_id,post_id,user_id)
      VALUES (p_parent_id,p_post_id,v_uid)
      ON CONFLICT (comment_id,user_id) DO NOTHING;

      RETURN jsonb_build_object('liked',true);
    ELSE
      DELETE FROM public.group_comment_likes
      WHERE comment_id=p_parent_id AND user_id=v_uid;

      RETURN jsonb_build_object('liked',false);
    END IF;
  ELSIF p_action IN ('comment','reply') THEN
    IF p_content IS NULL OR
       char_length(btrim(p_content)) NOT BETWEEN 1 AND 1200 THEN
      RAISE EXCEPTION 'Invalid comment';
    END IF;

    IF p_action='comment' AND p_parent_id IS NOT NULL THEN
      RAISE EXCEPTION 'Invalid comment parent';
    END IF;

    IF p_action='reply' THEN
      IF p_parent_id IS NULL THEN
        RAISE EXCEPTION 'Reply parent required';
      END IF;
      SELECT * INTO v_parent FROM public.group_post_comments
        WHERE id=p_parent_id AND post_id=p_post_id;
      IF NOT FOUND OR v_parent.parent_id IS NOT NULL THEN
        RAISE EXCEPTION 'Invalid reply parent';
      END IF;
      -- Do not allow replies to blocked comment authors.
      IF parapost_groups_private.blocked(v_uid,v_parent.author_id) THEN
        RAISE EXCEPTION 'Reply unavailable'
          USING ERRCODE='42501';
      END IF;
    END IF;

    INSERT INTO public.group_post_comments
      (post_id,author_id,parent_id,content)
    VALUES (p_post_id,v_uid,p_parent_id,btrim(p_content))
    RETURNING id INTO v_id;

    RETURN jsonb_build_object('id',v_id);
  ELSE
    RAISE EXCEPTION 'Unknown interaction action';
  END IF;
END;
$$;
CREATE FUNCTION public.groups_comment_like_stats(
  p_group_id uuid,
  p_post_id uuid
)

  like_count bigint,
  liked boolean
)
LANGUAGE sql

SECURITY DEFINER

AS $$

    SELECT c.id, c.created_at
    FROM public.group_post_comments c
    JOIN public.group_posts p ON p.id = c.post_id
    WHERE c.post_id = p_post_id

CREATE FUNCTION public.groups_share_post(
  p_group_id uuid,
  p_post_id uuid,
  p_caption text DEFAULT NULL
)
RETURNS jsonb
LANGUAGE plpgsql
SECURITY DEFINER
SET search_path = ''
AS $$
DECLARE
  v_uid uuid := auth.uid();
  v_group public.groups%ROWTYPE;
  v_post public.group_posts%ROWTYPE;
  v_share_id uuid;
BEGIN
  IF v_uid IS NULL THEN
    RAISE EXCEPTION 'Sign in required'
      USING ERRCODE = '42501';
  END IF;

  SELECT * INTO v_group
  FROM public.groups
  WHERE id = p_group_id
  FOR SHARE;

  IF NOT FOUND
     OR v_group.visibility <> 'public'
     OR NOT parapost_groups_private.allowed(
       p_group_id, 'member'
     )
  THEN
    RAISE EXCEPTION 'Public Group membership required'
      USING ERRCODE = '42501';
  END IF;

  SELECT * INTO v_post
  FROM public.group_posts
  WHERE id = p_post_id
    AND group_id = p_group_id
    AND published
    AND NOT is_deleted;

  IF NOT FOUND THEN
    RAISE EXCEPTION 'Group post unavailable'
      USING ERRCODE = '42501';
  END IF;

  IF parapost_groups_private.blocked(
    v_uid, v_post.author_id
  ) THEN
    RAISE EXCEPTION 'Sharing unavailable'
      USING ERRCODE = '42501';
  END IF;

  IF p_caption IS NOT NULL
     AND char_length(p_caption) > 1000 THEN
    RAISE EXCEPTION 'Share caption too long';
  END IF;

  INSERT INTO public.group_post_shares(
    group_id, post_id, user_id, caption
  )
  VALUES (
    p_group_id,
    p_post_id,
    v_uid,
    NULLIF(btrim(p_caption), '')
  )
  ON CONFLICT (post_id, user_id) DO NOTHING
  RETURNING id INTO v_share_id;

  IF v_share_id IS NULL THEN
    SELECT id INTO v_share_id
    FROM public.group_post_shares
    WHERE post_id = p_post_id
      AND user_id = v_uid;
  END IF;

  RETURN jsonb_build_object('id', v_share_id);
END;
$$;
CREATE FUNCTION public.groups_shared_feed(
  p_sharer_id uuid DEFAULT NULL,
  p_limit integer DEFAULT 60,
  p_offset integer DEFAULT 0
)
RETURNS TABLE (
  share_id uuid,
  group_id uuid,
  post_id uuid,
  sharer_id uuid,
  caption text,
  shared_at timestamptz,
  group_name text,
  original_author_id uuid,
  original_content text,
  original_created_at timestamptz,
  media jsonb
)
LANGUAGE sql
STABLE
SECURITY DEFINER
SET search_path = ''
AS $$
  SELECT
    s.id,
    s.group_id,
    s.post_id,
    s.user_id,
    s.caption,
    s.created_at,
    g.name,
    p.author_id,
    p.content,
    p.created_at,
    COALESCE(media_items.items, '[]'::jsonb)
  FROM public.group_post_shares s
  JOIN public.groups g
    ON g.id = s.group_id
  JOIN public.group_posts p
    ON p.id = s.post_id
   AND p.group_id = s.group_id
  LEFT JOIN LATERAL (
    SELECT jsonb_agg(
      jsonb_build_object(
        'media_type', m.media_type,
        'storage_path', m.storage_path,
        'display_order', m.display_order
      )
      ORDER BY m.display_order, m.id
    ) AS items
    FROM public.group_post_media m
    WHERE m.post_id = p.id
      AND m.group_id = g.id
  ) media_items ON true
  WHERE auth.uid() IS NOT NULL
    AND g.visibility = 'public'
    AND p.published
    AND NOT p.is_deleted
    AND parapost_groups_private.allowed(
      g.id, 'read'
    )
    AND NOT parapost_groups_private.blocked(
      auth.uid(), s.user_id
    )
    AND NOT parapost_groups_private.blocked(
      auth.uid(), p.author_id
    )
    AND (
      p_sharer_id IS NULL
      OR s.user_id = p_sharer_id
    )
  ORDER BY s.created_at DESC, s.id DESC
  LIMIT LEAST(GREATEST(COALESCE(p_limit,60),1),100)
  OFFSET LEAST(GREATEST(COALESCE(p_offset,0),0),10000);
$$;

CREATE OR REPLACE FUNCTION public.groups_shared_post_stats(
  p_share_id uuid
)
RETURNS jsonb
LANGUAGE plpgsql
STABLE
SECURITY DEFINER
SET search_path = ''
AS $function$
DECLARE
  v_uid uuid := auth.uid();
  v_group_id uuid;
  v_sharer_id uuid;
  v_author_id uuid;
  v_comments jsonb;
BEGIN
  IF v_uid IS NULL THEN
    RAISE EXCEPTION 'Sign in required'
      USING ERRCODE = '42501';
  END IF;

  SELECT s.group_id, s.user_id, p.author_id
  INTO v_group_id, v_sharer_id, v_author_id
  FROM public.group_post_shares s
  JOIN public.groups g ON g.id = s.group_id
  JOIN public.group_posts p
    ON p.id = s.post_id AND p.group_id = s.group_id
  WHERE s.id = p_share_id
    AND g.visibility = 'public'
    AND NOT p.is_deleted;

  IF NOT FOUND THEN
    RAISE EXCEPTION 'Share unavailable'
      USING ERRCODE = '42501';
  END IF;

  IF NOT parapost_groups_private.allowed(v_group_id, 'read')
     OR parapost_groups_private.blocked(v_uid, v_sharer_id)
     OR parapost_groups_private.blocked(v_uid, v_author_id)
  THEN
    RAISE EXCEPTION 'Share unavailable'
      USING ERRCODE = '42501';
  END IF;

  SELECT COALESCE(
    jsonb_agg(
      jsonb_build_object(
        'id', c.id,
        'author_id', c.author_id,
        'parent_id', c.parent_id,
        'content', c.content,
        'created_at', c.created_at,
        'updated_at', c.updated_at,
        'like_count', c.like_count,
        'liked', c.liked
      )
      ORDER BY c.created_at, c.id
    ),
    '[]'::jsonb
  )
  INTO v_comments
  FROM (
    SELECT
      cm.*,
      (
        SELECT count(*)::integer
        FROM public.group_post_share_comment_likes cl
        WHERE cl.comment_id = cm.id
          AND NOT parapost_groups_private.blocked(v_uid, cl.user_id)
      ) AS like_count,
      EXISTS (
        SELECT 1
        FROM public.group_post_share_comment_likes cl
        WHERE cl.comment_id = cm.id
          AND cl.user_id = v_uid
      ) AS liked
    FROM public.group_post_share_comments cm
    WHERE cm.share_id = p_share_id
      AND NOT parapost_groups_private.blocked(v_uid, cm.author_id)
      AND (
        cm.parent_id IS NULL
        OR EXISTS (
          SELECT 1
          FROM public.group_post_share_comments parent
          WHERE parent.id = cm.parent_id
            AND parent.share_id = cm.share_id
            AND parent.parent_id IS NULL
            AND NOT parapost_groups_private.blocked(
              v_uid, parent.author_id
            )
        )
      )
    ORDER BY cm.created_at, cm.id
    LIMIT 150
  ) c;

  RETURN jsonb_build_object(
    'liked', EXISTS (
      SELECT 1
      FROM public.group_post_share_likes l
      WHERE l.share_id = p_share_id
        AND l.user_id = v_uid
    ),
    'like_count', (
      SELECT count(*)::integer
      FROM public.group_post_share_likes l
      WHERE l.share_id = p_share_id
        AND NOT parapost_groups_private.blocked(v_uid, l.user_id)
    ),
    'comment_count', (
      SELECT count(*)::integer
      FROM public.group_post_share_comments c
      WHERE c.share_id = p_share_id
        AND NOT parapost_groups_private.blocked(v_uid, c.author_id)
        AND (
          c.parent_id IS NULL
          OR EXISTS (
            SELECT 1
            FROM public.group_post_share_comments parent
            WHERE parent.id = c.parent_id
              AND parent.share_id = c.share_id
              AND parent.parent_id IS NULL
              AND NOT parapost_groups_private.blocked(
                v_uid, parent.author_id
              )
          )
        )
    ),
    'comments_limited', (
      SELECT count(*) > 150
      FROM public.group_post_share_comments c
      WHERE c.share_id = p_share_id
        AND NOT parapost_groups_private.blocked(v_uid, c.author_id)
        AND (
          c.parent_id IS NULL
          OR EXISTS (
            SELECT 1
            FROM public.group_post_share_comments parent
            WHERE parent.id = c.parent_id
              AND parent.share_id = c.share_id
              AND parent.parent_id IS NULL
              AND NOT parapost_groups_private.blocked(
                v_uid, parent.author_id
              )
          )
        )
    ),
    'comments', v_comments
  );
END;
$function$;
CREATE OR REPLACE FUNCTION public.groups_shared_post_interact(
  p_share_id uuid,
  p_action text,
  p_content text DEFAULT NULL,
  p_parent_id uuid DEFAULT NULL
)
RETURNS jsonb
LANGUAGE plpgsql
SECURITY DEFINER
SET search_path = ''
AS $function$
DECLARE
  v_uid uuid := auth.uid();
  v_share public.group_post_shares%ROWTYPE;
  v_group public.groups%ROWTYPE;
  v_post public.group_posts%ROWTYPE;
  v_parent public.group_post_share_comments%ROWTYPE;
  v_id uuid;
  v_limit_kind text;
  v_limit_max integer;
  v_limit_now timestamptz;
  v_limit_count integer;
BEGIN
  IF v_uid IS NULL THEN
    RAISE EXCEPTION 'Sign in required'
      USING ERRCODE = '42501';
  END IF;

  SELECT * INTO v_share
  FROM public.group_post_shares
  WHERE id = p_share_id;

  IF NOT FOUND THEN
    RAISE EXCEPTION 'Share unavailable'
      USING ERRCODE = '42501';
  END IF;

  SELECT * INTO v_group
  FROM public.groups
  WHERE id = v_share.group_id
  FOR SHARE;

  IF NOT FOUND
     OR v_group.visibility <> 'public'
     OR NOT parapost_groups_private.allowed(
       v_share.group_id, 'read'
     ) THEN
    RAISE EXCEPTION 'Group unavailable'
      USING ERRCODE = '42501';
  END IF;

  SELECT * INTO v_post
  FROM public.group_posts
  WHERE id = v_share.post_id
    AND group_id = v_share.group_id
    AND NOT is_deleted;

  IF NOT FOUND THEN
    RAISE EXCEPTION 'Post unavailable'
      USING ERRCODE = '42501';
  END IF;

  IF parapost_groups_private.blocked(v_uid, v_share.user_id)
     OR parapost_groups_private.blocked(
       v_uid, v_post.author_id
     ) THEN
    RAISE EXCEPTION 'Interaction unavailable'
      USING ERRCODE = '42501';
  END IF;

  -- Atomic per-user limits across all shared Group posts.
  IF p_action IN ('comment', 'reply') THEN
    v_limit_kind := 'comment';
    v_limit_max := 10;
  ELSIF p_action IN (
    'like', 'unlike', 'like_comment', 'unlike_comment'
  ) THEN
    v_limit_kind := 'reaction';
    v_limit_max := 60;
  END IF;

  IF v_limit_kind IS NOT NULL THEN
    v_limit_now := clock_timestamp();

    INSERT INTO parapost_groups_private.shared_post_interaction_rate_limits AS rl
      (user_id, action_kind, window_started_at, action_count)
    VALUES (v_uid, v_limit_kind, v_limit_now, 1)
    ON CONFLICT (user_id, action_kind)
    DO UPDATE SET
      window_started_at = CASE
        WHEN rl.window_started_at <=
             v_limit_now - interval '60 seconds'
        THEN v_limit_now
        ELSE rl.window_started_at
      END,
      action_count = CASE
        WHEN rl.window_started_at <=
             v_limit_now - interval '60 seconds'
        THEN 1
        ELSE rl.action_count + 1
      END
    WHERE rl.window_started_at <=
          v_limit_now - interval '60 seconds'
       OR rl.action_count < v_limit_max
    RETURNING action_count INTO v_limit_count;

    IF NOT FOUND THEN
      RAISE EXCEPTION 'Rate limit exceeded. Try again shortly.'
        USING ERRCODE = 'P0001';
    END IF;
  END IF;

  IF p_action = 'like' THEN
    INSERT INTO public.group_post_share_likes
      (share_id, user_id)
    VALUES (p_share_id, v_uid)
    ON CONFLICT (share_id, user_id) DO NOTHING;

    RETURN jsonb_build_object('liked', true);

  ELSIF p_action = 'unlike' THEN
    DELETE FROM public.group_post_share_likes
    WHERE share_id = p_share_id
      AND user_id = v_uid;

    RETURN jsonb_build_object('liked', false);

  ELSIF p_action = 'comment' THEN
    IF p_parent_id IS NOT NULL
       OR p_content IS NULL
       OR char_length(btrim(p_content)) NOT BETWEEN 1 AND 1200 THEN
      RAISE EXCEPTION 'Invalid comment';
    END IF;

    INSERT INTO public.group_post_share_comments
      (share_id, author_id, content)
    VALUES (p_share_id, v_uid, btrim(p_content))
    RETURNING id INTO v_id;

    RETURN jsonb_build_object('id', v_id);

  ELSIF p_action IN (
    'reply', 'like_comment', 'unlike_comment',
    'edit_comment', 'delete_comment'
  ) THEN
    IF p_parent_id IS NULL THEN
      RAISE EXCEPTION 'Comment required';
    END IF;

    SELECT * INTO v_parent
    FROM public.group_post_share_comments
    WHERE id = p_parent_id
      AND share_id = p_share_id;

    IF NOT FOUND THEN
      RAISE EXCEPTION 'Comment unavailable'
        USING ERRCODE = '42501';
    END IF;

    IF p_action = 'reply' THEN
      IF v_parent.parent_id IS NOT NULL
         OR parapost_groups_private.blocked(
           v_uid, v_parent.author_id
         )
         OR p_content IS NULL
         OR char_length(btrim(p_content)) NOT BETWEEN 1 AND 1200 THEN
        RAISE EXCEPTION 'Reply unavailable'
          USING ERRCODE = '42501';
      END IF;

      INSERT INTO public.group_post_share_comments
        (share_id, author_id, parent_id, content)
      VALUES (
        p_share_id, v_uid, p_parent_id, btrim(p_content)
      )
      RETURNING id INTO v_id;

      RETURN jsonb_build_object('id', v_id);

    ELSIF p_action = 'like_comment' THEN
      IF parapost_groups_private.blocked(
        v_uid, v_parent.author_id
      )
      OR (
        v_parent.parent_id IS NOT NULL
        AND NOT EXISTS (
          SELECT 1
          FROM public.group_post_share_comments parent
          WHERE parent.id = v_parent.parent_id
            AND parent.share_id = p_share_id
            AND parent.parent_id IS NULL
            AND NOT parapost_groups_private.blocked(
              v_uid, parent.author_id
            )
        )
      ) THEN
        RAISE EXCEPTION 'Comment Like unavailable'
          USING ERRCODE = '42501';
      END IF;

      INSERT INTO public.group_post_share_comment_likes
        (comment_id, share_id, user_id)
      VALUES (p_parent_id, p_share_id, v_uid)
      ON CONFLICT (comment_id, user_id) DO NOTHING;

      RETURN jsonb_build_object('liked', true);

    ELSIF p_action = 'unlike_comment' THEN
      DELETE FROM public.group_post_share_comment_likes
      WHERE comment_id = p_parent_id
        AND share_id = p_share_id
        AND user_id = v_uid;

      RETURN jsonb_build_object('liked', false);

    ELSIF p_action = 'edit_comment' THEN
      IF v_parent.author_id <> v_uid
         OR p_content IS NULL
         OR char_length(btrim(p_content)) NOT BETWEEN 1 AND 1200 THEN
        RAISE EXCEPTION 'Edit unavailable'
          USING ERRCODE = '42501';
      END IF;

      UPDATE public.group_post_share_comments
      SET content = btrim(p_content),
          updated_at = now()
      WHERE id = p_parent_id
        AND share_id = p_share_id
        AND author_id = v_uid;

      RETURN jsonb_build_object('updated', true);

    ELSIF p_action = 'delete_comment' THEN
      IF v_parent.author_id <> v_uid THEN
        RAISE EXCEPTION 'Delete unavailable'
          USING ERRCODE = '42501';
      END IF;

      DELETE FROM public.group_post_share_comments
      WHERE id = p_parent_id
        AND share_id = p_share_id
        AND author_id = v_uid;

      RETURN jsonb_build_object('deleted', true);
    END IF;
  END IF;

  RAISE EXCEPTION 'Unknown interaction action';
END;
$function$;

CREATE FUNCTION creatorverse_private.validate_live_page_owner()
RETURNS trigger
LANGUAGE plpgsql
SECURITY DEFINER
SET search_path = ''
AS $function$
BEGIN
  -- Existing personal broadcasts require no new permission.
  IF NEW.creatorverse_page_id IS NULL THEN
    RETURN NEW;
  END IF;

  -- Only the authenticated broadcaster can associate a page.
  IF auth.uid() IS DISTINCT FROM NEW.user_id THEN
    RAISE EXCEPTION
      'Unauthorized CreatorVerse livestream association'
      USING ERRCODE = '42501';
  END IF;

  -- The page must be active and belong to the broadcaster.
  IF NOT EXISTS (
    SELECT 1
    FROM public.creatorverse_pages AS p
    WHERE p.id = NEW.creatorverse_page_id
      AND p.owner_id = NEW.user_id
      AND p.status = 'active'
  ) THEN
    RAISE EXCEPTION
      'CreatorVerse page is not owned or active'
      USING ERRCODE = '42501';
  END IF;

  RETURN NEW;
END;
$function$;

  IF v_post IS NULL THEN
    RAISE EXCEPTION 'Post unavailable'
      USING ERRCODE = '42501';
  END IF;

  UPDATE public.creatorverse_posts
  SET status = 'deleted',
      updated_at = now()
  WHERE id = v_post
  RETURNING id INTO v_post;

  RETURN v_post;
END;
$body$;
CREATE OR REPLACE FUNCTION public.creatorverse_directory_member_counts(
  p_page_ids uuid[]
)
RETURNS TABLE (
  page_id uuid,
  member_count bigint
)
LANGUAGE sql
STABLE
SECURITY DEFINER
SET search_path = ''
AS $counts$
  SELECT
    p.id,
    COUNT(f.user_id)::bigint
  FROM public.creatorverse_pages AS p
  LEFT JOIN public.creatorverse_follows AS f
    ON f.page_id = p.id
  WHERE cardinality(p_page_ids) BETWEEN 1 AND 25
    AND p.id = ANY(p_page_ids)
    AND p.status = 'active'
    AND creatorverse_private.read_page(p.id)

$counts$;
CREATE FUNCTION public.creatorverse_edit_post(
  p_page_id uuid,
  p_post_id uuid,
  p_content text
)
RETURNS uuid
LANGUAGE plpgsql
SECURITY DEFINER
SET search_path = ''
AS $body$
DECLARE
  v_user uuid := auth.uid();
  v_post uuid;
  v_content text := btrim(p_content);
BEGIN
  IF v_user IS NULL
     OR NOT creatorverse_private.write_enabled()
     OR NOT creatorverse_private.owns(p_page_id)
     OR NOT creatorverse_private.read_page(p_page_id)
  THEN
    RAISE EXCEPTION 'Post management unavailable'
      USING ERRCODE = '42501';
  END IF;

  PERFORM 1 FROM public.creatorverse_pages
  WHERE id = p_page_id FOR UPDATE;

  SELECT id INTO v_post
  FROM public.creatorverse_posts
  WHERE id = p_post_id
    AND page_id = p_page_id
    AND author_id = v_user
    AND status = 'published'
  FOR UPDATE;

  IF v_post IS NULL THEN
    RAISE EXCEPTION 'Post unavailable'
      USING ERRCODE = '42501';
  END IF;

  IF v_content IS NULL
     OR char_length(v_content) > 5000
     OR translate(v_content, E'\n\r\t', '') ~ '[[:cntrl:]]'
  THEN
    RAISE EXCEPTION 'Invalid post content'
      USING ERRCODE = '23514';
  END IF;

  IF v_content = '' AND NOT EXISTS (
    SELECT 1
    FROM public.creatorverse_media
    WHERE page_id = p_page_id
      AND post_id = p_post_id
      AND purpose = 'post'
      AND verification = 'verified'
  ) THEN
    RAISE EXCEPTION 'Post cannot be empty'
      USING ERRCODE = '23514';
  END IF;

  UPDATE public.creatorverse_posts
  SET content = v_content,
      updated_at = now()
  WHERE id = v_post
  RETURNING id INTO v_post;

  RETURN v_post;
END;
$body$;

-- 4. NOTIFICATION VISIBILITY AND RLS GUARDS
-- =================================================

CREATE OR REPLACE FUNCTION
creatorverse_private.community_notification_visible(
  n public.notifications
)
RETURNS boolean
LANGUAGE plpgsql
STABLE
SECURITY DEFINER
SET search_path = ''
AS $function$
DECLARE
  v_owner uuid;
  v_visibility text;
BEGIN
  -- Never expose self-activity or incomplete events.
  IF n.user_id IS NULL
     OR n.actor_id IS NULL
     OR n.user_id = n.actor_id
     OR parapost_groups_private.blocked(
       n.user_id, n.actor_id
     )
  THEN
    RETURN false;
  END IF;

  IF n.type IN (
    'creatorverse_follow',
    'creatorverse_like',
    'creatorverse_comment',
    'creatorverse_reply',
    'creatorverse_share',
    'creatorverse_comment_like'
  ) THEN

    SELECT owner_id
    INTO v_owner
    FROM public.creatorverse_pages
    WHERE id = n.creatorverse_page_id
      AND status = 'active';

    IF v_owner IS NULL THEN
      RETURN false;
    END IF;

    IF n.community_post_id IS NOT NULL
       AND NOT EXISTS (
         SELECT 1
         FROM public.creatorverse_posts p
         WHERE p.id = n.community_post_id
           AND p.page_id = n.creatorverse_page_id
           AND p.status = 'published'
       )
    THEN
      RETURN false;
    END IF;

    IF n.community_comment_id IS NOT NULL
       AND NOT EXISTS (
         SELECT 1
         FROM public.creatorverse_comments c
         WHERE c.id = n.community_comment_id
           AND c.post_id = n.community_post_id
           AND NOT c.is_deleted
       )
    THEN
      RETURN false;
    END IF;

  ELSIF n.type IN (
    'group_post',
    'group_like',
    'group_comment',
    'group_reply',
    'group_share',
    'group_comment_like'
  ) THEN

    SELECT owner_id, visibility
    INTO v_owner, v_visibility
    FROM public.groups
    WHERE id = n.group_id;

    IF v_owner IS NULL THEN
      RETURN false;
    END IF;

    IF n.community_post_id IS NOT NULL
       AND NOT EXISTS (
         SELECT 1
         FROM public.group_posts p
         WHERE p.id = n.community_post_id
           AND p.group_id = n.group_id
           AND p.published
           AND NOT p.is_deleted
       )
    THEN
      RETURN false;
    END IF;

    IF n.community_comment_id IS NOT NULL
       AND NOT EXISTS (
         SELECT 1
         FROM public.group_post_comments c
         WHERE c.id = n.community_comment_id
           AND c.post_id = n.community_post_id
       )
    THEN
      RETURN false;
    END IF;

    IF v_visibility = 'private' THEN

      IF n.user_id <> v_owner
         AND NOT EXISTS (
           SELECT 1
           FROM public.group_members m
           WHERE m.group_id = n.group_id
             AND m.user_id = n.user_id
             AND m.status = 'approved'
         )
      THEN
        RETURN false;
      END IF;

      IF n.actor_id <> v_owner
         AND NOT EXISTS (
           SELECT 1
           FROM public.group_members m
           WHERE m.group_id = n.group_id
             AND m.user_id = n.actor_id
             AND m.status = 'approved'
         )
      THEN
        RETURN false;
      END IF;

    END IF;

    IF EXISTS (
      SELECT 1
      FROM public.group_members m
      WHERE m.group_id = n.group_id
        AND m.user_id IN (n.user_id, n.actor_id)
        AND m.status IN ('removed', 'banned')
    ) THEN
      RETURN false;
    END IF;

  ELSE
    RETURN false;
  END IF;

  -- Protect relationships involving page/group owners.
  IF parapost_groups_private.blocked(n.user_id, v_owner)
     OR parapost_groups_private.blocked(
       n.actor_id, v_owner
     )
  THEN
    RETURN false;
  END IF;

  RETURN true;
END;
$function$;

-- ==================================================
-- 1. COMMUNITY LIKES, COMMENTS, REPLIES AND SHARES
-- ==================================================

CREATE OR REPLACE FUNCTION
creatorverse_private.notify_community_activity_release()
RETURNS trigger
LANGUAGE plpgsql
SECURITY DEFINER
SET search_path = ''
AS $function$
DECLARE
  r jsonb;
  actor uuid;
  post uuid;
  comment uuid;
  parent_comment uuid;
  container uuid;
  owner_id uuid;
  author_id uuid;
  comment_author uuid;
  recipient uuid;
  visibility text;
  kind text;
  action text;
  event_key text;
BEGIN
  IF TG_OP = 'DELETE' THEN
    r := to_jsonb(OLD);
  ELSE
    r := to_jsonb(NEW);
  END IF;

  kind := CASE
    WHEN TG_TABLE_NAME LIKE 'creatorverse_%'
    THEN 'creatorverse'
    ELSE 'group'
  END;

  actor := coalesce(
    r->>'user_id',
    r->>'author_id',
    r->>'sharer_id'
  )::uuid;

  IF actor IS NULL THEN
    RETURN NULL;
  END IF;

  event_key := TG_TABLE_NAME || ':' ||
    coalesce(
      r->>'id',
      r->>'comment_id',
      r->>'post_id'
    ) || ':' || actor::text;

  -- Removing interactions removes related notifications.
  IF TG_OP = 'DELETE' THEN
    DELETE FROM public.notifications
    WHERE community_event_key = event_key;

    IF TG_TABLE_NAME IN (
      'creatorverse_comments',
      'group_post_comments'
    ) THEN
      DELETE FROM public.notifications
      WHERE community_comment_id = (r->>'id')::uuid;
    END IF;

    RETURN NULL;
  END IF;

  IF TG_TABLE_NAME = 'creatorverse_comments'
     AND r->>'is_deleted' = 'true'
  THEN
    DELETE FROM public.notifications
    WHERE community_event_key = event_key
       OR community_comment_id = (r->>'id')::uuid;
    RETURN NULL;
  END IF;

  -- Editing must not generate duplicate notifications.
  IF TG_OP <> 'INSERT' THEN
    RETURN NULL;
  END IF;

  IF auth.uid() IS DISTINCT FROM actor THEN
    RETURN NULL;
  END IF;

  post := nullif(r->>'post_id', '')::uuid;

  IF TG_TABLE_NAME IN (
    'creatorverse_comments',
    'group_post_comments'
  ) THEN
    comment := (r->>'id')::uuid;
    parent_comment := nullif(r->>'parent_id', '')::uuid;

    action := CASE
      WHEN parent_comment IS NULL THEN 'comment'
      ELSE 'reply'
    END;

  ELSIF TG_TABLE_NAME IN (
    'creatorverse_comment_likes',
    'group_comment_likes'
  ) THEN
    comment := (r->>'comment_id')::uuid;
    action := 'comment_like';

  ELSIF TG_TABLE_NAME LIKE '%shares' THEN
    action := 'share';

  ELSE
    action := 'like';
  END IF;

  IF TG_TABLE_NAME = 'creatorverse_comment_likes' THEN
    SELECT c.post_id, c.author_id
    INTO post, comment_author
    FROM public.creatorverse_comments c
    WHERE c.id = comment
      AND NOT c.is_deleted;

  ELSIF TG_TABLE_NAME = 'group_comment_likes' THEN
    SELECT c.post_id, c.author_id
    INTO post, comment_author
    FROM public.group_post_comments c
    WHERE c.id = comment;
  END IF;

  IF kind = 'creatorverse' THEN

    SELECT p.page_id, p.author_id, pg.owner_id
    INTO container, author_id, owner_id
    FROM public.creatorverse_posts p
    JOIN public.creatorverse_pages pg
      ON pg.id = p.page_id
    WHERE p.id = post
      AND p.status = 'published'
      AND pg.status = 'active';

    IF parent_comment IS NOT NULL THEN
      SELECT c.author_id INTO comment_author
      FROM public.creatorverse_comments c
      WHERE c.id = parent_comment
        AND c.post_id = post
        AND NOT c.is_deleted;
    END IF;

  ELSE

    SELECT p.group_id, p.author_id,
           g.owner_id, g.visibility
    INTO container, author_id, owner_id, visibility
    FROM public.group_posts p
    JOIN public.groups g
      ON g.id = p.group_id
    WHERE p.id = post
      AND p.published
      AND NOT p.is_deleted;

    IF parent_comment IS NOT NULL THEN
      SELECT c.author_id INTO comment_author
      FROM public.group_post_comments c
      WHERE c.id = parent_comment
        AND c.post_id = post;

    ELSIF action = 'comment_like' THEN
      SELECT c.author_id INTO comment_author
      FROM public.group_post_comments c
      WHERE c.id = comment
        AND c.post_id = post;
    END IF;

  END IF;

  IF container IS NULL OR owner_id IS NULL THEN
    RETURN NULL;
  END IF;

  -- Private Groups: actors must be approved members.
  IF kind = 'group'
     AND visibility = 'private'
     AND actor <> owner_id
     AND NOT EXISTS (
       SELECT 1
       FROM public.group_members m
       WHERE m.group_id = container
         AND m.user_id = actor
         AND m.status = 'approved'
     )
  THEN
    RETURN NULL;
  END IF;

  FOR recipient IN
    SELECT DISTINCT x.user_id
    FROM unnest(
      ARRAY[owner_id, author_id, comment_author]
    ) AS x(user_id)
    WHERE x.user_id IS NOT NULL
      AND x.user_id <> actor
  LOOP

    IF parapost_groups_private.blocked(
      actor, recipient
    ) THEN
      CONTINUE;
    END IF;

    IF kind = 'group'
       AND visibility = 'private'
       AND recipient <> owner_id
       AND NOT EXISTS (
         SELECT 1
         FROM public.group_members m
         WHERE m.group_id = container
           AND m.user_id = recipient
           AND m.status = 'approved'
       )
    THEN
      CONTINUE;
    END IF;

    INSERT INTO public.notifications (
      user_id,
      actor_id,
      type,
      creatorverse_page_id,
      group_id,
      community_post_id,
      community_comment_id,
      community_event_key,
      message,
      is_read
    )
    VALUES (
      recipient,
      actor,
      kind || '_' || action,
      CASE
        WHEN kind = 'creatorverse' THEN container
      END,
      CASE
        WHEN kind = 'group' THEN container
      END,
      post,
      comment,
      event_key,
      CASE
        WHEN kind = 'creatorverse'
        THEN 'New CreatorVerse activity.'
        ELSE 'New Group activity.'
      END,
      false
    )
    ON CONFLICT DO NOTHING;

  END LOOP;

  RETURN NULL;

EXCEPTION WHEN OTHERS THEN
  RAISE WARNING
    'Community activity notification failed: %',
    SQLERRM;
  RETURN NULL;
END;
$function$;

-- ==================================================
-- 2. NEW GROUP POSTS: NOTIFY GROUP OWNER
-- ==================================================

CREATE OR REPLACE FUNCTION
parapost_groups_private.notify_owner_new_post_release()
RETURNS trigger
LANGUAGE plpgsql
SECURITY DEFINER
SET search_path = ''
AS $function$
DECLARE
  recipient_id uuid;
BEGIN
  IF TG_OP = 'DELETE' THEN
    DELETE FROM public.notifications
    WHERE group_id = OLD.group_id
      AND community_post_id = OLD.id
      AND (
        type = 'group_post'
        OR community_event_key IS NOT NULL
      );
    RETURN OLD;
  END IF;

  IF NEW.published IS NOT TRUE
     OR NEW.is_deleted IS TRUE
  THEN
    DELETE FROM public.notifications
    WHERE group_id = NEW.group_id
      AND community_post_id = NEW.id
      AND (
        type = 'group_post'
        OR community_event_key IS NOT NULL
      );
    RETURN NEW;
  END IF;

  IF TG_OP = 'UPDATE' THEN
    IF OLD.published IS TRUE
       AND OLD.is_deleted IS NOT TRUE
    THEN
      RETURN NEW;
    END IF;
  END IF;

  SELECT g.owner_id INTO recipient_id
  FROM public.groups g
  WHERE g.id = NEW.group_id;

  IF recipient_id IS NULL
     OR NEW.author_id IS NULL
     OR recipient_id = NEW.author_id
  THEN
    RETURN NEW;
  END IF;

  IF parapost_groups_private.blocked(
    recipient_id, NEW.author_id
  ) THEN
    RETURN NEW;
  END IF;

  BEGIN
    INSERT INTO public.notifications (
      user_id,
      actor_id,
      type,
      group_id,
      community_post_id,
      message,
      is_read
    )
    VALUES (
      recipient_id,
      NEW.author_id,
      'group_post',
      NEW.group_id,
      NEW.id,
      'A member published a new Group post.',
      false
    )
    ON CONFLICT DO NOTHING;

  EXCEPTION WHEN OTHERS THEN
    RAISE WARNING
      'Group owner post notification failed: %',
      SQLERRM;
  END;

  RETURN NEW;

EXCEPTION WHEN OTHERS THEN
  RAISE WARNING
    'Group owner notification failed: %',
    SQLERRM;

  IF TG_OP = 'DELETE' THEN
    RETURN OLD;
  END IF;

  RETURN NEW;
END;
$function$;

-- 3. CREATORVERSE FOLLOW NOTIFICATIONS
-- ==================================================

CREATE OR REPLACE FUNCTION
creatorverse_private.notify_owner_follow_release()
RETURNS trigger
LANGUAGE plpgsql
SECURITY DEFINER
SET search_path = ''
AS $function$
DECLARE
  owner_id uuid;
BEGIN
  IF TG_OP = 'DELETE' THEN
    DELETE FROM public.notifications
    WHERE type = 'creatorverse_follow'
      AND creatorverse_page_id = OLD.page_id
      AND actor_id = OLD.user_id;
    RETURN OLD;
  END IF;

  SELECT p.owner_id INTO owner_id
  FROM public.creatorverse_pages p
  WHERE p.id = NEW.page_id
    AND p.status = 'active';

  IF owner_id IS NULL
     OR owner_id = NEW.user_id
     OR parapost_groups_private.blocked(
       owner_id, NEW.user_id
     )
  THEN
    RETURN NEW;
  END IF;

  BEGIN
    INSERT INTO public.notifications (
      user_id,
      actor_id,
      type,
      creatorverse_page_id,
      message,
      is_read
    )
    VALUES (
      owner_id,
      NEW.user_id,
      'creatorverse_follow',
      NEW.page_id,
      'Someone followed your CreatorVerse page.',
      false
    )
    ON CONFLICT DO NOTHING;

  EXCEPTION WHEN OTHERS THEN
    RAISE WARNING
      'CreatorVerse follow notification failed: %',
      SQLERRM;
  END;

  RETURN NEW;

EXCEPTION WHEN OTHERS THEN
  RAISE WARNING
    'CreatorVerse follow handler failed: %',
    SQLERRM;

  IF TG_OP = 'DELETE' THEN
    RETURN OLD;
  END IF;

  RETURN NEW;
END;
$function$;
