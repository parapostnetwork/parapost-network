-- FUNCTION: public.creatorverse_delete_post(p_page_id uuid, p_post_id uuid)
-- SOURCE: docs/sql-drafts/creatorverse-post-management-STAGING.sql
-- SHA256: 8c7c9fe8a512c9cb2a60f0a478122e0e9d6cd2b2a4896f70cfc7609efa7d7f1c

CREATE FUNCTION public.creatorverse_delete_post(
  p_page_id uuid,
  p_post_id uuid
)
RETURNS uuid
LANGUAGE plpgsql
SECURITY DEFINER
SET search_path = ''
AS $body$
DECLARE
  v_user uuid := auth.uid();
  v_post uuid;
BEGIN
  IF v_user IS NULL
     OR NOT creatorverse_private.write_enabled()
     OR NOT creatorverse_private.owns(p_page_id)
     OR NOT creatorverse_private.read_page(p_page_id)
  THEN
    RAISE EXCEPTION 'Post management unavailable'
      USING ERRCODE = '42501';
  END IF;

  PERFORM 1 FROM public.creatorverse_pages
  WHERE id = p_page_id FOR UPDATE;

  SELECT id INTO v_post
  FROM public.creatorverse_posts
  WHERE id = p_post_id
    AND page_id = p_page_id
    AND author_id = v_user
    AND status = 'published'
  FOR UPDATE;

  IF v_post IS NULL THEN
    RAISE EXCEPTION 'Post unavailable'
      USING ERRCODE = '42501';
  END IF;

  UPDATE public.creatorverse_posts
  SET status = 'deleted',
      updated_at = now()
  WHERE id = v_post
  RETURNING id INTO v_post;

  RETURN v_post;
END;
$body$;

-- FUNCTION: public.creatorverse_directory_member_counts(p_page_ids uuid[])
-- SOURCE: docs/sql-drafts/creatorverse-member-counts-STAGING.sql
-- SHA256: 26f17dd90e50bc86baab9c5ad1e9a069e68341b9163ce65088b2121efca4f7b1

CREATE OR REPLACE FUNCTION public.creatorverse_directory_member_counts(
  p_page_ids uuid[]
)
RETURNS TABLE (
  page_id uuid,
  member_count bigint
)
LANGUAGE sql
STABLE
SECURITY DEFINER
SET search_path = ''
AS $counts$
  SELECT
    p.id,
    COUNT(f.user_id)::bigint
  FROM public.creatorverse_pages AS p
  LEFT JOIN public.creatorverse_follows AS f
    ON f.page_id = p.id
  WHERE cardinality(p_page_ids) BETWEEN 1 AND 25
    AND p.id = ANY(p_page_ids)
    AND p.status = 'active'
    AND creatorverse_private.read_page(p.id)
  GROUP BY p.id;
$counts$;

-- FUNCTION: public.creatorverse_edit_post(p_page_id uuid, p_post_id uuid, p_content text)
-- SOURCE: docs/sql-drafts/creatorverse-post-management-STAGING.sql
-- SHA256: 823902234854d62b020c60ca121d2b82eb8b7e7c2412f6bbaaa36ed142cf836e

CREATE FUNCTION public.creatorverse_edit_post(
  p_page_id uuid,
  p_post_id uuid,
  p_content text
)
RETURNS uuid
LANGUAGE plpgsql
SECURITY DEFINER
SET search_path = ''
AS $body$
DECLARE
  v_user uuid := auth.uid();
  v_post uuid;
  v_content text := btrim(p_content);
BEGIN
  IF v_user IS NULL
     OR NOT creatorverse_private.write_enabled()
     OR NOT creatorverse_private.owns(p_page_id)
     OR NOT creatorverse_private.read_page(p_page_id)
  THEN
    RAISE EXCEPTION 'Post management unavailable'
      USING ERRCODE = '42501';
  END IF;

  PERFORM 1 FROM public.creatorverse_pages
  WHERE id = p_page_id FOR UPDATE;

  SELECT id INTO v_post
  FROM public.creatorverse_posts
  WHERE id = p_post_id
    AND page_id = p_page_id
    AND author_id = v_user
    AND status = 'published'
  FOR UPDATE;

  IF v_post IS NULL THEN
    RAISE EXCEPTION 'Post unavailable'
      USING ERRCODE = '42501';
  END IF;

  IF v_content IS NULL
     OR char_length(v_content) > 5000
     OR translate(v_content, E'\n\r\t', '') ~ '[[:cntrl:]]'
  THEN
    RAISE EXCEPTION 'Invalid post content'
      USING ERRCODE = '23514';
  END IF;

  IF v_content = '' AND NOT EXISTS (
    SELECT 1
    FROM public.creatorverse_media
    WHERE page_id = p_page_id
      AND post_id = p_post_id
      AND purpose = 'post'
      AND verification = 'verified'
  ) THEN
    RAISE EXCEPTION 'Post cannot be empty'
      USING ERRCODE = '23514';
  END IF;

  UPDATE public.creatorverse_posts
  SET content = v_content,
      updated_at = now()
  WHERE id = v_post
  RETURNING id INTO v_post;

  RETURN v_post;
END;
$body$;
