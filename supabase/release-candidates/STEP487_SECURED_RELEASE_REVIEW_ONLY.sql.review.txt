CREATE TABLE public.creatorverse_pages (
  id uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  owner_id uuid NOT NULL REFERENCES auth.users(id),
  name text NOT NULL CHECK (char_length(btrim(name)) BETWEEN 3 AND 100),
  category text NOT NULL CHECK (
    category IN (
      'business','creator','investigation_team','podcast',
      'entertainment','organization','community_project','other'
    )
  ),
  short_bio text NOT NULL DEFAULT ''
    CHECK (char_length(short_bio) <= 240),
  description text NOT NULL DEFAULT ''
    CHECK (char_length(description) <= 5000),
  website text CHECK (
    website IS NULL OR (
      char_length(website) <= 2048 AND
      website ~ '^https?://[^[:space:]@/]+([/?#][^[:space:]]*)?$'
    )
  ),
  social_links jsonb NOT NULL DEFAULT '[]'::jsonb
    CHECK (
      jsonb_typeof(social_links) = 'array'
      AND jsonb_array_length(social_links) <= 8
    ),
  public_email text CHECK (
    public_email IS NULL OR (
      char_length(public_email) <= 254 AND
      public_email ~ '^[^[:space:]@]+@[^[:space:]@]+\.[^[:space:]@]+$'
    )
  ),
  location text NOT NULL DEFAULT ''
    CHECK (char_length(location) <= 160),
  logo_media_id uuid,
  cover_media_id uuid,
  status text NOT NULL DEFAULT 'draft'
    CHECK (status IN ('draft','active','suspended','deleted')),
  created_at timestamptz NOT NULL DEFAULT now(),
  updated_at timestamptz NOT NULL DEFAULT now(),
  search_document tsvector GENERATED ALWAYS AS (
    to_tsvector(
      'simple'::regconfig,
      name || ' ' || short_bio || ' ' || description
    )
  ) STORED,
  UNIQUE (id, owner_id)
);

CREATE TABLE public.creatorverse_posts (
  id uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  page_id uuid NOT NULL REFERENCES public.creatorverse_pages(id),
  author_id uuid NOT NULL REFERENCES auth.users(id),
  content text NOT NULL DEFAULT ''
    CHECK (char_length(content) <= 10000),
  status text NOT NULL DEFAULT 'draft'
    CHECK (status IN ('draft','published','deleted')),
  comments_enabled boolean NOT NULL DEFAULT true,
  created_at timestamptz NOT NULL DEFAULT now(),
  updated_at timestamptz NOT NULL DEFAULT now(),
  UNIQUE (page_id, id),
  FOREIGN KEY (page_id, author_id)
    REFERENCES public.creatorverse_pages(id, owner_id)
);

CREATE TABLE public.creatorverse_media (
  id uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  page_id uuid NOT NULL REFERENCES public.creatorverse_pages(id),
  post_id uuid,
  purpose text NOT NULL
    CHECK (purpose IN ('logo','cover','post')),
  media_type text NOT NULL
    CHECK (media_type IN ('image','video')),
  storage_path text NOT NULL UNIQUE,
  mime_type text NOT NULL CHECK (
    mime_type IN (
      'image/jpeg','image/png','image/webp',
      'video/mp4','video/webm'
    )
  ),
  verification text NOT NULL DEFAULT 'pending'
    CHECK (verification IN ('pending','verified','rejected')),
  byte_size bigint CHECK (byte_size > 0),
  duration_seconds numeric CHECK (
    duration_seconds > 0 AND duration_seconds <= 120
  ),
  display_order smallint NOT NULL DEFAULT 0
    CHECK (display_order BETWEEN 0 AND 3),
  created_at timestamptz NOT NULL DEFAULT now(),
  UNIQUE (page_id, id),
  FOREIGN KEY (page_id, post_id)
    REFERENCES public.creatorverse_posts(page_id, id),
  CHECK (storage_path LIKE page_id::text || '/%'),
  CHECK (
    char_length(storage_path) <= 240
    AND storage_path NOT LIKE '%..%'
    AND storage_path NOT LIKE '%//%'
  ),
  CHECK (
    (purpose = 'post' AND post_id IS NOT NULL)
    OR (
      purpose IN ('logo','cover')
      AND post_id IS NULL
      AND media_type = 'image'
    )
  ),
  CHECK (
    (media_type = 'image'
      AND mime_type IN ('image/jpeg','image/png','image/webp')
      AND duration_seconds IS NULL
      AND (byte_size IS NULL OR byte_size <= 8388608))
    OR
    (media_type = 'video'
      AND mime_type IN ('video/mp4','video/webm')
      AND (byte_size IS NULL OR byte_size <= 33554432))
  ),
  CHECK (
    verification <> 'verified'
    OR (
      byte_size IS NOT NULL
      AND (media_type = 'image' OR duration_seconds IS NOT NULL)
    )
  )
);

ALTER TABLE public.creatorverse_pages

CREATE OR REPLACE FUNCTION public.get_parachat_inbox_summaries()
RETURNS TABLE (
  conversation_id uuid,
  last_message_id uuid,
  last_sender_id uuid,
  last_body text,
  last_created_at timestamptz,
  last_is_read boolean,
  last_message_type text,
  last_image_path text,
  last_image_mime_type text,
  last_image_size_bytes integer,
  last_image_width integer,
  last_image_height integer,
  unread_count bigint
)
LANGUAGE sql
STABLE
SECURITY INVOKER
SET search_path = public
AS $$
  WITH my_conversations AS (
    SELECT dc.id
    FROM public.direct_conversations dc
    WHERE auth.uid() IS NOT NULL
      AND (
        dc.user_one_id = auth.uid()
        OR dc.user_two_id = auth.uid()
      )
  )
  SELECT
    mc.id AS conversation_id,
    latest.id AS last_message_id,
    latest.sender_id AS last_sender_id,
    latest.body AS last_body,
    latest.created_at AS last_created_at,
    latest.is_read AS last_is_read,
    latest.message_type AS last_message_type,
    latest.image_path AS last_image_path,
    latest.image_mime_type AS last_image_mime_type,
    latest.image_size_bytes AS last_image_size_bytes,
    latest.image_width AS last_image_width,
    latest.image_height AS last_image_height,
    COALESCE(unread.unread_count, 0)::bigint AS unread_count
  FROM my_conversations mc

  LEFT JOIN LATERAL (
    SELECT
      dm.id,
      dm.sender_id,
      dm.body,
      dm.created_at,
      dm.is_read,
      dm.message_type,
      dm.image_path,
      dm.image_mime_type,
      dm.image_size_bytes,
      dm.image_width,
      dm.image_height
    FROM public.direct_messages dm
    WHERE dm.conversation_id = mc.id
    ORDER BY dm.created_at DESC, dm.id DESC
    LIMIT 1
  ) latest ON true

  LEFT JOIN LATERAL (
    SELECT COUNT(*)::bigint AS unread_count
    FROM public.direct_messages dm
    WHERE dm.conversation_id = mc.id
      AND dm.sender_id <> auth.uid()
      AND dm.is_read = false
  ) unread ON true;
$$;

CREATE OR REPLACE FUNCTION public.create_comment_like_notification()
RETURNS trigger
LANGUAGE plpgsql
SET search_path TO 'public', 'pg_temp'
AS $function$
declare
  comment_owner_id uuid;
  comment_post_id uuid;
begin
  select
    comments.user_id,
    comments.post_id
  into
    comment_owner_id,
    comment_post_id
  from public.comments
  where comments.id = new.comment_id;

  if comment_owner_id is not null
     and comment_owner_id <> new.user_id then

    insert into public.notifications (
      user_id,
      actor_id,
      type,
      post_id,
      comment_id,
      message
    )
    values (
      comment_owner_id,
      new.user_id,
      'comment_like',
      comment_post_id,
      new.comment_id,
      'liked your comment'
    )
    on conflict do nothing;

  end if;

  return new;
end;
$function$;

CREATE OR REPLACE FUNCTION public.create_comment_reply_notification()
RETURNS trigger
LANGUAGE plpgsql
SET search_path TO 'public', 'pg_temp'
AS $function$
declare
  reply_target_id uuid;
  post_owner_id uuid;
begin
  if new.parent_comment_id is null then
    return new;
  end if;

  reply_target_id := new.reply_to_user_id;

  if reply_target_id is null then
    select comments.user_id
    into reply_target_id
    from public.comments
    where comments.id = new.parent_comment_id;
  end if;

  select posts.user_id
  into post_owner_id
  from public.posts
  where posts.id = new.post_id;

  if reply_target_id is not null
     and reply_target_id <> new.user_id then

    insert into public.notifications (
      user_id,
      actor_id,
      type,
      post_id,
      comment_id,
      message
    )
    values (
      reply_target_id,
      new.user_id,
      'comment_reply',
      new.post_id,
      new.id,
      'replied to your comment'
    )
    on conflict do nothing;

  end if;

  if post_owner_id is not null
     and post_owner_id <> new.user_id
     and post_owner_id is distinct from reply_target_id then

    insert into public.notifications (
      user_id,
      actor_id,
      type,
      post_id,
      comment_id,
      message
    )
    values (
      post_owner_id,
      new.user_id,
      'comment_reply',
      new.post_id,
      new.id,
      'replied to a comment on your post'
    )
    on conflict do nothing;

  end if;

  return new;
end;
$function$;

CREATE OR REPLACE FUNCTION public.sync_post_share_notification()
RETURNS trigger
LANGUAGE plpgsql
SECURITY DEFINER
SET search_path TO 'public', 'pg_temp'
AS $function$
DECLARE
  old_active boolean := false;
  new_active boolean := false;
  should_create boolean := false;
  post_owner_id uuid;
BEGIN

  -- ----------------------------------------------------------
  -- Determine previous active state when OLD exists.
  -- ----------------------------------------------------------

  IF TG_OP <> 'INSERT' THEN
    old_active :=
      OLD.deleted_at IS NULL
      AND COALESCE(OLD.share_destination, 'feed') = 'feed'
      AND OLD.user_id IS NOT NULL
      AND OLD.post_id IS NOT NULL;
  END IF;


  -- ----------------------------------------------------------
  -- Determine new active state when NEW exists.
  -- ----------------------------------------------------------

  IF TG_OP <> 'DELETE' THEN
    new_active :=
      NEW.deleted_at IS NULL
      AND COALESCE(NEW.share_destination, 'feed') = 'feed'
      AND NEW.user_id IS NOT NULL
      AND NEW.post_id IS NOT NULL;
  END IF;


  -- ----------------------------------------------------------
  -- HARD DELETE
  --
  -- If the removed row was the final active share for this
  -- actor + post, remove its notification.
  -- ----------------------------------------------------------

  IF TG_OP = 'DELETE' THEN

    IF old_active
       AND NOT EXISTS (
         SELECT 1
         FROM public.shares AS shares
         WHERE shares.user_id = OLD.user_id
           AND shares.post_id = OLD.post_id
           AND shares.deleted_at IS NULL
           AND COALESCE(shares.share_destination, 'feed') = 'feed'
       )
    THEN

      DELETE FROM public.notifications
      WHERE type IN ('post_share', 'share')
        AND actor_id = OLD.user_id
        AND post_id = OLD.post_id;

    END IF;

    RETURN OLD;
  END IF;


  -- ----------------------------------------------------------
  -- UPDATE
  --
  -- Clean the old notification when an active share becomes
  -- inactive or changes actor/post.
  -- ----------------------------------------------------------

  IF TG_OP = 'UPDATE' THEN

    IF old_active
       AND (
         NOT new_active
         OR OLD.user_id IS DISTINCT FROM NEW.user_id
         OR OLD.post_id IS DISTINCT FROM NEW.post_id
       )
       AND NOT EXISTS (
         SELECT 1
         FROM public.shares AS shares
         WHERE shares.user_id = OLD.user_id
           AND shares.post_id = OLD.post_id
           AND shares.deleted_at IS NULL
           AND COALESCE(shares.share_destination, 'feed') = 'feed'
       )
    THEN

      DELETE FROM public.notifications
      WHERE type IN ('post_share', 'share')
        AND actor_id = OLD.user_id
        AND post_id = OLD.post_id;

    END IF;


    IF new_active
       AND (
         NOT old_active
         OR OLD.user_id IS DISTINCT FROM NEW.user_id
         OR OLD.post_id IS DISTINCT FROM NEW.post_id
       )
    THEN
      should_create := true;
    END IF;

  END IF;


  -- ----------------------------------------------------------
  -- INSERT
  -- ----------------------------------------------------------

  IF TG_OP = 'INSERT' AND new_active THEN
    should_create := true;
  END IF;


  -- ----------------------------------------------------------
  -- CREATE NOTIFICATION
  -- ----------------------------------------------------------

CREATE OR REPLACE FUNCTION public.enforce_comment_share_context()
RETURNS trigger
LANGUAGE plpgsql
SET search_path TO 'public', 'pg_temp'
AS $function$
DECLARE
  parent_post_id uuid;
  parent_share_id uuid;
  reply_post_id uuid;
  reply_share_id uuid;
BEGIN

  -- ----------------------------------------------------------
  -- Existing comments may not be moved to another post/thread.
  -- ----------------------------------------------------------

  IF TG_OP = 'UPDATE' THEN
    IF NEW.post_id IS DISTINCT FROM OLD.post_id
       OR NEW.share_id IS DISTINCT FROM OLD.share_id
       OR NEW.parent_comment_id IS DISTINCT FROM OLD.parent_comment_id
    THEN
      RAISE EXCEPTION 'Comment discussion context cannot be changed.';
    END IF;
  END IF;


  -- ----------------------------------------------------------
  -- Replies inherit the exact parent discussion context.
  -- ----------------------------------------------------------

  IF NEW.parent_comment_id IS NOT NULL THEN

    SELECT
      comments.post_id,
      comments.share_id
    INTO
      parent_post_id,
      parent_share_id
    FROM public.comments AS comments
    WHERE comments.id = NEW.parent_comment_id;

    IF NOT FOUND THEN
      RAISE EXCEPTION 'Parent comment does not exist.';
    END IF;

    IF NEW.post_id IS DISTINCT FROM parent_post_id THEN
      RAISE EXCEPTION 'Reply post does not match parent comment.';
    END IF;

    NEW.share_id := parent_share_id;

  END IF;


  -- ----------------------------------------------------------
  -- Exact reply targets must belong to the same discussion.
  -- ----------------------------------------------------------

  IF NEW.reply_to_comment_id IS NOT NULL THEN

    SELECT
      comments.post_id,
      comments.share_id
    INTO
      reply_post_id,
      reply_share_id
    FROM public.comments AS comments
    WHERE comments.id = NEW.reply_to_comment_id;

    IF NOT FOUND THEN
      RAISE EXCEPTION 'Reply target comment does not exist.';
    END IF;

    IF NEW.post_id IS DISTINCT FROM reply_post_id
       OR NEW.share_id IS DISTINCT FROM reply_share_id
    THEN
      RAISE EXCEPTION 'Reply target belongs to another discussion.';
    END IF;

  END IF;


  -- ----------------------------------------------------------
  -- A share-specific comment must reference an active feed share
  -- for the exact same original post.
  -- ----------------------------------------------------------

  IF NEW.share_id IS NOT NULL
     AND NOT EXISTS (
       SELECT 1
       FROM public.shares AS shares
       WHERE shares.id = NEW.share_id
         AND shares.post_id = NEW.post_id
         AND shares.deleted_at IS NULL
         AND COALESCE(shares.share_destination, 'feed') = 'feed'
     )
  THEN
    RAISE EXCEPTION 'Shared post discussion is not active.';
  END IF;


  RETURN NEW;
END;
$function$;

CREATE OR REPLACE FUNCTION public.sync_post_share_notification()
RETURNS trigger
LANGUAGE plpgsql
SECURITY DEFINER
SET search_path TO 'public', 'pg_temp'
AS $function$
DECLARE
  old_active boolean := false;
  new_active boolean := false;
  should_create boolean := false;
  post_owner_id uuid;
BEGIN

  IF TG_OP <> 'INSERT' THEN

    old_active :=
      OLD.deleted_at IS NULL
      AND COALESCE(OLD.share_destination, 'feed') = 'feed'
      AND OLD.user_id IS NOT NULL
      AND OLD.post_id IS NOT NULL;

  END IF;


  IF TG_OP <> 'DELETE' THEN

    new_active :=
      NEW.deleted_at IS NULL
      AND COALESCE(NEW.share_destination, 'feed') = 'feed'
      AND NEW.user_id IS NOT NULL
      AND NEW.post_id IS NOT NULL;

  END IF;


  -- ----------------------------------------------------------
  -- HARD DELETE
  -- ----------------------------------------------------------

  IF TG_OP = 'DELETE' THEN

    IF old_active THEN

      DELETE FROM public.notifications
      WHERE type IN ('post_share', 'share')
        AND (
          share_id = OLD.id

          OR (
            share_id IS NULL
            AND actor_id = OLD.user_id
            AND post_id = OLD.post_id
          )
        );

    END IF;


    RETURN OLD;
  END IF;


  -- ----------------------------------------------------------
  -- UPDATE
  -- ----------------------------------------------------------

  IF TG_OP = 'UPDATE' THEN

    IF old_active
       AND (
         NOT new_active
         OR OLD.user_id IS DISTINCT FROM NEW.user_id
         OR OLD.post_id IS DISTINCT FROM NEW.post_id
       )
    THEN

      DELETE FROM public.notifications
      WHERE type IN ('post_share', 'share')
        AND (
          share_id = OLD.id

          OR (
            share_id IS NULL
            AND actor_id = OLD.user_id
            AND post_id = OLD.post_id
          )
        );

    END IF;


    IF new_active
       AND (
         NOT old_active
         OR OLD.user_id IS DISTINCT FROM NEW.user_id
         OR OLD.post_id IS DISTINCT FROM NEW.post_id
       )
    THEN
      should_create := true;
    END IF;

  END IF;


  -- ----------------------------------------------------------
  -- INSERT
  -- ----------------------------------------------------------

  IF TG_OP = 'INSERT'
     AND new_active
  THEN
    should_create := true;
  END IF;


  -- ----------------------------------------------------------
  -- CREATE EXACT SHARE NOTIFICATION
  -- ----------------------------------------------------------

  IF should_create THEN

    SELECT posts.user_id
    INTO post_owner_id
    FROM public.posts AS posts
    WHERE posts.id = NEW.post_id;


    IF post_owner_id IS NOT NULL
       AND post_owner_id <> NEW.user_id
    THEN

      INSERT INTO public.notifications (
        user_id,
        actor_id,
        type,
        post_id,
        share_id,
        message,
        is_read
      )
      VALUES (
        post_owner_id,
        NEW.user_id,
        'post_share',
        NEW.post_id,
        NEW.id,
        'shared your post.',
        false
      )
      ON CONFLICT DO NOTHING;

    END IF;

  END IF;


  RETURN NEW;
END;

CREATE OR REPLACE FUNCTION public.sync_reel_share_count()
RETURNS trigger
LANGUAGE plpgsql
SECURITY DEFINER
SET search_path TO ''
AS $$
BEGIN
  IF TG_OP = 'INSERT' THEN
    UPDATE public.reels
    SET shares = coalesce(shares, 0) + 1
    WHERE id = NEW.reel_id;

    RETURN NEW;
  END IF;

  IF TG_OP = 'DELETE' THEN
    UPDATE public.reels
    SET shares = greatest(coalesce(shares, 0) - 1, 0)
    WHERE id = OLD.reel_id;

    RETURN OLD;
  END IF;

  -- Only a Reel move changes the number of shares belonging
  -- to either Reel. Other UPDATE operations leave counts alone.
  IF OLD.reel_id IS DISTINCT FROM NEW.reel_id THEN
    UPDATE public.reels
    SET shares = greatest(coalesce(shares, 0) - 1, 0)
    WHERE id = OLD.reel_id;

    UPDATE public.reels
    SET shares = coalesce(shares, 0) + 1
    WHERE id = NEW.reel_id;
  END IF;

  RETURN NEW;
END;
$$;

CREATE OR REPLACE FUNCTION public.parapost_guard_message_updates()
RETURNS trigger
LANGUAGE plpgsql
SET search_path = ''
AS $$
DECLARE
  v_user_id uuid;
BEGIN
  -- Allow trusted service-role operations.
  IF auth.role() = 'service_role' THEN
    RETURN NEW;
  END IF;

  -- Reject anonymous or unauthenticated updates.
  IF auth.role() IS DISTINCT FROM 'authenticated' THEN
    RAISE EXCEPTION 'Authentication required';
  END IF;

  v_user_id := auth.uid();

  IF v_user_id IS NULL THEN
    RAISE EXCEPTION 'Authentication required';
  END IF;

  -- The original sender may edit the message body only.
  IF OLD.sender_id = v_user_id THEN
    IF (to_jsonb(NEW) - 'body') IS DISTINCT FROM
       (to_jsonb(OLD) - 'body') THEN
      RAISE EXCEPTION 'Message sender may update body only';
    END IF;

    RETURN NEW;
  END IF;

  -- A recipient may mark an incoming message as read only.
  IF public.is_direct_conversation_participant(
       OLD.conversation_id,
       v_user_id
     )
     AND OLD.sender_id <> v_user_id
     AND OLD.is_read IS DISTINCT FROM TRUE
     AND NEW.is_read IS TRUE
     AND (to_jsonb(NEW) - 'is_read') IS NOT DISTINCT FROM
         (to_jsonb(OLD) - 'is_read') THEN
    RETURN NEW;
  END IF;

  RAISE EXCEPTION 'Message update not permitted';
END;

CREATE FUNCTION parapost_groups_private.request_membership(
  p_group_id uuid
)
RETURNS text
LANGUAGE plpgsql
SECURITY DEFINER
SET search_path = ''
AS $function$
DECLARE
  v_user_id uuid := auth.uid();
  v_visibility text;
  v_owner_id uuid;
  v_status text;
BEGIN
  IF v_user_id IS NULL THEN
    RAISE EXCEPTION 'Authentication required'
      USING ERRCODE = '28000';
  END IF;

  SELECT g.visibility, g.owner_id
    INTO v_visibility, v_owner_id
  FROM public.groups g
  WHERE g.id = p_group_id
  FOR SHARE;

  IF NOT FOUND THEN
    RAISE EXCEPTION 'Group unavailable';
  END IF;

  IF EXISTS (
    SELECT 1
    FROM public.blocked_users b
    WHERE
      (b.blocker_id = v_user_id
       AND b.blocked_id = v_owner_id)
      OR
      (b.blocked_id = v_user_id
       AND b.blocker_id = v_owner_id)
  ) THEN
    RAISE EXCEPTION 'Membership unavailable';
  END IF;

  INSERT INTO public.group_members (
    group_id,
    user_id,
    role,
    status,
    approved_at
  )
  VALUES (
    p_group_id,
    v_user_id,
    'member',
    CASE
      WHEN v_visibility = 'public'
      THEN 'approved'
      ELSE 'pending'
    END,
    CASE
      WHEN v_visibility = 'public'
      THEN now()
      ELSE NULL
    END
  )
  ON CONFLICT (group_id, user_id)
  DO NOTHING;

  SELECT gm.status
    INTO v_status
  FROM public.group_members gm
  WHERE gm.group_id = p_group_id
    AND gm.user_id = v_user_id;

  IF v_status NOT IN ('approved', 'pending')
     OR v_status IS NULL THEN
    RAISE EXCEPTION 'Membership unavailable';
  END IF;

  RETURN v_status;
END;
$function$;

CREATE FUNCTION parapost_groups_private.review_membership(
  p_group_id uuid,
  p_target_user_id uuid,
  p_decision text
)
RETURNS text
LANGUAGE plpgsql
SECURITY DEFINER
SET search_path = ''
AS $function$
DECLARE
  v_reviewer uuid := auth.uid();
  v_owner_id uuid;
  v_reviewer_role text;
  v_final_status text;
BEGIN
  IF v_reviewer IS NULL THEN
    RAISE EXCEPTION 'Authentication required';
  END IF;

  IF p_group_id IS NULL
     OR p_target_user_id IS NULL
     OR p_decision IS NULL
     OR p_decision NOT IN ('approved', 'rejected') THEN
    RAISE EXCEPTION 'Invalid membership review';
  END IF;

  SELECT g.owner_id
    INTO v_owner_id
  FROM public.groups g
  WHERE g.id = p_group_id
  FOR SHARE;

  IF NOT FOUND THEN
    RAISE EXCEPTION 'Group unavailable';
  END IF;

  SELECT gm.role
    INTO v_reviewer_role
  FROM public.group_members gm
  WHERE gm.group_id = p_group_id
    AND gm.user_id = v_reviewer
    AND gm.status = 'approved'
    AND gm.role IN ('owner', 'admin')
  FOR SHARE;

  IF NOT FOUND THEN
    RAISE EXCEPTION 'Not authorized to review membership';
  END IF;

  IF p_decision = 'approved' AND EXISTS (
    SELECT 1
    FROM public.blocked_users b
    WHERE
      (
        b.blocker_id = p_target_user_id
        AND b.blocked_id IN (v_owner_id, v_reviewer)
      )
      OR
      (
        b.blocked_id = p_target_user_id
        AND b.blocker_id IN (v_owner_id, v_reviewer)
      )
  ) THEN
    RAISE EXCEPTION 'Membership unavailable';
  END IF;

  UPDATE public.group_members gm
  SET
    status = p_decision,
    approved_at = CASE
      WHEN p_decision = 'approved'
      THEN now()
      ELSE NULL
    END
  WHERE gm.group_id = p_group_id
    AND gm.user_id = p_target_user_id
    AND gm.role = 'member'
    AND gm.status = 'pending'
  RETURNING gm.status INTO v_final_status;

  IF NOT FOUND THEN
    RAISE EXCEPTION 'Pending membership unavailable';
  END IF;

  RETURN v_final_status;
END;
$function$;

CREATE FUNCTION parapost_groups_private.command(action text, gid uuid, payload jsonb)
RETURNS jsonb LANGUAGE plpgsql SECURITY DEFINER SET search_path = '' AS $$
DECLARE
 uid uuid := auth.uid(); g public.groups; m public.group_members; p public.group_posts;
 target uuid; pid uuid; mid uuid; path text; mime text; role_name text; result jsonb; item record;
BEGIN
 IF NOT parapost_groups_private.enabled() THEN RAISE EXCEPTION 'Groups unavailable' USING ERRCODE='42501'; END IF;
 IF action='create' THEN
   IF (SELECT count(*) FROM public.groups WHERE owner_id=uid)>=100 THEN RAISE EXCEPTION 'Group creation limit reached'; END IF;
   INSERT INTO public.groups(owner_id,name,description,visibility)
   VALUES(uid,btrim(payload->>'name'),coalesce(payload->>'description',''),coalesce(payload->>'visibility','public')) RETURNING * INTO g;
   RETURN jsonb_build_object('id',g.id);
 END IF;
 -- Group lock serializes ALL writes, including membership removal vs publish/role changes.
 SELECT * INTO g FROM public.groups WHERE id=gid FOR UPDATE;
 IF NOT FOUND OR NOT parapost_groups_private.allowed(gid,'directory') THEN RAISE EXCEPTION 'Group unavailable' USING ERRCODE='42501'; END IF;
 SELECT * INTO m FROM public.group_members WHERE group_id=gid AND user_id=uid;
 IF action='join' THEN
   IF m.status IS NOT NULL THEN
     IF m.status NOT IN ('pending','approved') THEN RAISE EXCEPTION 'Membership unavailable'; END IF;
   ELSE
     INSERT INTO public.group_members(group_id,user_id,role,status,approved_at)
     VALUES(gid,uid,'member',CASE WHEN g.visibility='public' THEN 'approved' ELSE 'pending' END,CASE WHEN g.visibility='public' THEN now() END);
   END IF;
 ELSIF action='leave' THEN
   IF m.role='owner' THEN RAISE EXCEPTION 'Owner must delete the Group; transfer is not supported'; END IF;
   -- Only voluntary departures remove the row. Removed/banned/rejected rows persist.
   DELETE FROM public.group_members WHERE group_id=gid AND user_id=uid AND status IN ('pending','approved');
 ELSIF action IN ('approve','reject','remove','ban','role','restore') THEN
   target := (payload->>'user_id')::uuid;
   IF NOT parapost_groups_private.allowed(gid,'admin') OR target=uid OR target=g.owner_id THEN RAISE EXCEPTION 'Not authorized' USING ERRCODE='42501'; END IF;
   SELECT * INTO m FROM public.group_members WHERE group_id=gid AND user_id=target;
   IF NOT FOUND THEN RAISE EXCEPTION 'Membership unavailable'; END IF;
   IF NOT parapost_groups_private.allowed(gid,'owner') AND (m.role <> 'member' OR action IN ('role','restore')) THEN RAISE EXCEPTION 'Owner permission required'; END IF;
   IF action IN ('approve','restore') AND parapost_groups_private.blocked(uid,target) THEN RAISE EXCEPTION 'Membership unavailable'; END IF;
   IF action IN ('approve','restore') AND parapost_groups_private.blocked(g.owner_id,target) THEN RAISE EXCEPTION 'Membership unavailable'; END IF;
   IF action IN ('approve','reject') THEN
     IF m.status <> 'pending' THEN RAISE EXCEPTION 'Request is no longer pending'; END IF;
     UPDATE public.group_members SET status=CASE WHEN action='approve' THEN 'approved' ELSE 'rejected' END,
       approved_at=CASE WHEN action='approve' THEN now() END WHERE group_id=gid AND user_id=target;
   ELSIF action='role' THEN
     role_name := payload->>'role';
     IF m.status <> 'approved' OR role_name IS NULL OR role_name NOT IN ('admin','moderator','member') THEN RAISE EXCEPTION 'Invalid role change'; END IF;
     UPDATE public.group_members SET role=role_name WHERE group_id=gid AND user_id=target;
   ELSIF action='restore' THEN
     IF m.status NOT IN ('removed','rejected','banned') THEN RAISE EXCEPTION 'Invalid restoration'; END IF;
     -- Restoration requires owner action AND a fresh approval, never immediate access.
     UPDATE public.group_members SET role='member',status='pending',approved_at=NULL WHERE group_id=gid AND user_id=target;
   ELSE
     UPDATE public.group_members SET role='member',status=CASE WHEN action='ban' THEN 'banned' ELSE 'removed' END,approved_at=NULL WHERE group_id=gid AND user_id=target;
   END IF;
 ELSIF action='edit' THEN
   IF NOT parapost_groups_private.allowed(gid,'admin') THEN RAISE EXCEPTION 'Not authorized'; END IF;
   -- Privacy changes are owner-only. Existing approvals survive; pending requests stay pending.
   IF payload->>'visibility' IS DISTINCT FROM g.visibility AND NOT parapost_groups_private.allowed(gid,'owner') THEN RAISE EXCEPTION 'Owner permission required'; END IF;
   UPDATE public.groups SET name=btrim(payload->>'name'),description=coalesce(payload->>'description',''),visibility=payload->>'visibility',updated_at=now() WHERE id=gid;
 ELSIF action='delete' THEN
   IF NOT parapost_groups_private.allowed(gid,'owner') OR payload->>'confirmation' IS DISTINCT FROM g.name THEN RAISE EXCEPTION 'Owner and Group name confirmation required'; END IF;
   DELETE FROM public.groups WHERE id=gid;
 ELSIF action='draft' THEN
   IF NOT parapost_groups_private.allowed(gid,'member') THEN RAISE EXCEPTION 'Approved membership required'; END IF;
   IF (SELECT count(*) FROM public.group_posts WHERE group_id=gid AND author_id=uid AND NOT published)>=10 THEN RAISE EXCEPTION 'Too many unfinished uploads'; END IF;
   INSERT INTO public.group_posts(group_id,author_id,content) VALUES(gid,uid,coalesce(payload->>'content','')) RETURNING id INTO pid;
   RETURN jsonb_build_object('id',pid);
 ELSIF action IN ('reserve_media','publish','edit_post','delete_post') THEN
   pid := (payload->>'post_id')::uuid;
   SELECT * INTO p FROM public.group_posts WHERE id=pid AND group_id=gid;
   IF NOT FOUND OR p.is_deleted THEN RAISE EXCEPTION 'Post unavailable'; END IF;
   IF NOT parapost_groups_private.allowed(gid,'member') OR (p.author_id<>uid AND NOT (action='delete_post' AND parapost_groups_private.allowed(gid,'moderate'))) THEN RAISE EXCEPTION 'Not authorized'; END IF;
   IF action='delete_post' THEN
     DELETE FROM public.group_posts WHERE id=pid;
   ELSIF action='edit_post' THEN
     IF char_length(btrim(coalesce(payload->>'content','')))=0 AND NOT EXISTS(SELECT 1 FROM public.group_post_media WHERE post_id=pid) THEN RAISE EXCEPTION 'Post cannot be empty'; END IF;
     UPDATE public.group_posts SET content=coalesce(payload->>'content',''),updated_at=now() WHERE id=pid;
   ELSIF action='reserve_media' THEN
     IF p.published OR (SELECT count(*) FROM public.group_post_media WHERE post_id=pid)>=4 THEN RAISE EXCEPTION 'Media limit or published post'; END IF;
     mime := payload->>'mime_type';
     IF mime IS NULL OR mime NOT IN ('image/jpeg','image/png','image/webp','video/mp4','video/webm') THEN RAISE EXCEPTION 'Unsupported media'; END IF;
     mid := gen_random_uuid();
     path := gid::text || '/posts/' || pid::text || '/' || uid::text || '/' || mid::text;
     INSERT INTO public.group_post_media(id,group_id,post_id,uploader_id,media_type,storage_path,mime_type,display_order)
     VALUES(mid,gid,pid,uid,CASE WHEN mime LIKE 'image/%' THEN 'image' ELSE 'video' END,path,mime,(SELECT count(*) FROM public.group_post_media WHERE post_id=pid));
     RETURN jsonb_build_object('path',path);
   ELSE
     IF p.published THEN RETURN jsonb_build_object('id',pid); END IF;
     IF char_length(btrim(p.content))=0 AND NOT EXISTS(SELECT 1 FROM public.group_post_media WHERE post_id=pid) THEN RAISE EXCEPTION 'Post cannot be empty'; END IF;
     FOR item IN SELECT * FROM public.group_post_media WHERE post_id=pid LOOP
       IF NOT EXISTS(SELECT 1 FROM storage.objects o WHERE o.bucket_id='group-media' AND o.name=item.storage_path
         AND o.metadata->>'mimetype'=item.mime_type AND (o.metadata->>'size')::bigint BETWEEN 1 AND CASE WHEN item.media_type='image' THEN 8388608 ELSE 33554432 END) THEN RAISE EXCEPTION 'Media upload incomplete or invalid'; END IF;
     END LOOP;
     UPDATE public.group_posts SET published=true,updated_at=now() WHERE id=pid;
   END IF;
 ELSIF action='remove_cover' THEN
   IF NOT parapost_groups_private.allowed(gid,'admin') THEN RAISE EXCEPTION 'Not authorized'; END IF;
   UPDATE public.groups SET cover_storage_path=NULL,updated_at=now() WHERE id=gid;
 ELSIF action='reserve_cover' THEN
   IF NOT parapost_groups_private.allowed(gid,'admin') THEN RAISE EXCEPTION 'Not authorized'; END IF;
   IF (SELECT count(*) FROM parapost_groups_private.cover_uploads WHERE group_id=gid AND uploader_id=uid)>=10 THEN RAISE EXCEPTION 'Too many unfinished covers'; END IF;
   -- Covers are intentionally directory metadata, including those for private Groups.
   path := gid::text || '/covers/' || gen_random_uuid()::text;
   INSERT INTO parapost_groups_private.cover_uploads(storage_path,group_id,uploader_id) VALUES(path,gid,uid);
   RETURN jsonb_build_object('path',path);
 ELSIF action='set_cover' THEN
   path := payload->>'path';
   IF NOT parapost_groups_private.allowed(gid,'admin') OR NOT EXISTS(SELECT 1 FROM parapost_groups_private.cover_uploads c JOIN storage.objects o ON o.name=c.storage_path AND o.bucket_id='group-media'
     WHERE c.storage_path=path AND c.group_id=gid AND c.uploader_id=uid AND o.metadata->>'mimetype' IN ('image/jpeg','image/png','image/webp') AND (o.metadata->>'size')::bigint BETWEEN 1 AND 8388608) THEN RAISE EXCEPTION 'Invalid cover'; END IF;
   UPDATE public.groups SET cover_storage_path=path,updated_at=now() WHERE id=gid;
   DELETE FROM parapost_groups_private.cover_uploads WHERE storage_path=path;
 ELSE RAISE EXCEPTION 'Unknown Group action';
 END IF;
 RETURN jsonb_build_object('id',gid);
END; $$;

  p_group_id uuid, p_post_id uuid, p_action text,
  p_content text DEFAULT NULL, p_parent_id uuid DEFAULT NULL
) RETURNS jsonb
LANGUAGE plpgsql SECURITY DEFINER SET search_path='' AS $$
DECLARE
  v_uid uuid := auth.uid();
  v_group public.groups;
  v_post public.group_posts;
  v_parent public.group_post_comments;
  v_id uuid;
BEGIN
  IF v_uid IS NULL THEN
    RAISE EXCEPTION 'Sign in required' USING ERRCODE='42501';
  END IF;

  -- Serialize interactions with Group membership/privacy changes.
  SELECT * INTO v_group FROM public.groups
    WHERE id=p_group_id FOR SHARE;

  IF NOT FOUND OR NOT parapost_groups_private.allowed(p_group_id,'member') THEN
    RAISE EXCEPTION 'Approved membership required' USING ERRCODE='42501';
  END IF;

  SELECT * INTO v_post FROM public.group_posts
    WHERE id=p_post_id AND group_id=p_group_id
      AND published AND NOT is_deleted;

  IF NOT FOUND THEN
    RAISE EXCEPTION 'Post unavailable' USING ERRCODE='42501';
  END IF;

  -- Block interactions with posts from blocked accounts.
  IF p_action IN ('like','comment','reply','like_comment')
     AND parapost_groups_private.blocked(v_uid,v_post.author_id) THEN
    RAISE EXCEPTION 'Interaction unavailable'
      USING ERRCODE='42501';
  END IF;

  IF p_action='like' THEN
    INSERT INTO public.group_post_likes(post_id,user_id)
      VALUES(p_post_id,v_uid)
      ON CONFLICT (post_id,user_id) DO NOTHING;
    RETURN jsonb_build_object('liked',true);
  ELSIF p_action='unlike' THEN
    DELETE FROM public.group_post_likes
      WHERE post_id=p_post_id AND user_id=v_uid;
    RETURN jsonb_build_object('liked',false);
  ELSIF p_action IN ('like_comment','unlike_comment') THEN
    IF p_parent_id IS NULL THEN
      RAISE EXCEPTION 'Comment required';
    END IF;

    SELECT * INTO v_parent
      FROM public.group_post_comments
      WHERE id=p_parent_id AND post_id=p_post_id;

    IF NOT FOUND THEN
      RAISE EXCEPTION 'Comment unavailable'
        USING ERRCODE='42501';
    END IF;

    IF p_action='like_comment' THEN
      IF parapost_groups_private.blocked(
        v_uid,v_parent.author_id
      ) THEN
        RAISE EXCEPTION 'Comment Like unavailable'
          USING ERRCODE='42501';
      END IF;

      INSERT INTO public.group_comment_likes
        (comment_id,post_id,user_id)
      VALUES (p_parent_id,p_post_id,v_uid)
      ON CONFLICT (comment_id,user_id) DO NOTHING;

      RETURN jsonb_build_object('liked',true);
    ELSE
      DELETE FROM public.group_comment_likes
      WHERE comment_id=p_parent_id AND user_id=v_uid;

      RETURN jsonb_build_object('liked',false);
    END IF;
  ELSIF p_action IN ('comment','reply') THEN
    IF p_content IS NULL OR
       char_length(btrim(p_content)) NOT BETWEEN 1 AND 1200 THEN
      RAISE EXCEPTION 'Invalid comment';
    END IF;

    IF p_action='comment' AND p_parent_id IS NOT NULL THEN
      RAISE EXCEPTION 'Invalid comment parent';
    END IF;

    IF p_action='reply' THEN
      IF p_parent_id IS NULL THEN
        RAISE EXCEPTION 'Reply parent required';
      END IF;
      SELECT * INTO v_parent FROM public.group_post_comments
        WHERE id=p_parent_id AND post_id=p_post_id;
      IF NOT FOUND OR v_parent.parent_id IS NOT NULL THEN
        RAISE EXCEPTION 'Invalid reply parent';
      END IF;
      -- Do not allow replies to blocked comment authors.
      IF parapost_groups_private.blocked(v_uid,v_parent.author_id) THEN
        RAISE EXCEPTION 'Reply unavailable'
          USING ERRCODE='42501';
      END IF;
    END IF;

    INSERT INTO public.group_post_comments
      (post_id,author_id,parent_id,content)
    VALUES (p_post_id,v_uid,p_parent_id,btrim(p_content))
    RETURNING id INTO v_id;

    RETURN jsonb_build_object('id',v_id);
  ELSE
    RAISE EXCEPTION 'Unknown interaction action';
  END IF;
END;
$$;
CREATE FUNCTION public.groups_comment_like_stats(
  p_group_id uuid,
  p_post_id uuid
)

  like_count bigint,
  liked boolean
)
LANGUAGE sql

SECURITY DEFINER

AS $$

    SELECT c.id, c.created_at
    FROM public.group_post_comments c
    JOIN public.group_posts p ON p.id = c.post_id
    WHERE c.post_id = p_post_id

CREATE FUNCTION public.groups_share_post(
  p_group_id uuid,
  p_post_id uuid,
  p_caption text DEFAULT NULL
)
RETURNS jsonb
LANGUAGE plpgsql
SECURITY DEFINER
SET search_path = ''
AS $$
DECLARE
  v_uid uuid := auth.uid();
  v_group public.groups%ROWTYPE;
  v_post public.group_posts%ROWTYPE;
  v_share_id uuid;
BEGIN
  IF v_uid IS NULL THEN
    RAISE EXCEPTION 'Sign in required'
      USING ERRCODE = '42501';
  END IF;

  SELECT * INTO v_group
  FROM public.groups
  WHERE id = p_group_id
  FOR SHARE;

  IF NOT FOUND
     OR v_group.visibility <> 'public'
     OR NOT parapost_groups_private.allowed(
       p_group_id, 'member'
     )
  THEN
    RAISE EXCEPTION 'Public Group membership required'
      USING ERRCODE = '42501';
  END IF;

  SELECT * INTO v_post
  FROM public.group_posts
  WHERE id = p_post_id
    AND group_id = p_group_id
    AND published
    AND NOT is_deleted;

  IF NOT FOUND THEN
    RAISE EXCEPTION 'Group post unavailable'
      USING ERRCODE = '42501';
  END IF;

  IF parapost_groups_private.blocked(
    v_uid, v_post.author_id
  ) THEN
    RAISE EXCEPTION 'Sharing unavailable'
      USING ERRCODE = '42501';
  END IF;

  IF p_caption IS NOT NULL
     AND char_length(p_caption) > 1000 THEN
    RAISE EXCEPTION 'Share caption too long';
  END IF;

  INSERT INTO public.group_post_shares(
    group_id, post_id, user_id, caption
  )
  VALUES (
    p_group_id,
    p_post_id,
    v_uid,
    NULLIF(btrim(p_caption), '')
  )
  ON CONFLICT (post_id, user_id) DO NOTHING
  RETURNING id INTO v_share_id;

  IF v_share_id IS NULL THEN
    SELECT id INTO v_share_id
    FROM public.group_post_shares
    WHERE post_id = p_post_id
      AND user_id = v_uid;
  END IF;

  RETURN jsonb_build_object('id', v_share_id);
END;
$$;
CREATE FUNCTION public.groups_shared_feed(
  p_sharer_id uuid DEFAULT NULL,
  p_limit integer DEFAULT 60,
  p_offset integer DEFAULT 0
)
RETURNS TABLE (
  share_id uuid,
  group_id uuid,
  post_id uuid,
  sharer_id uuid,
  caption text,
  shared_at timestamptz,
  group_name text,
  original_author_id uuid,
  original_content text,
  original_created_at timestamptz,
  media jsonb
)
LANGUAGE sql
STABLE
SECURITY DEFINER
SET search_path = ''
AS $$
  SELECT
    s.id,
    s.group_id,
    s.post_id,
    s.user_id,
    s.caption,
    s.created_at,
    g.name,
    p.author_id,
    p.content,
    p.created_at,
    COALESCE(media_items.items, '[]'::jsonb)
  FROM public.group_post_shares s
  JOIN public.groups g
    ON g.id = s.group_id
  JOIN public.group_posts p
    ON p.id = s.post_id
   AND p.group_id = s.group_id
  LEFT JOIN LATERAL (
    SELECT jsonb_agg(
      jsonb_build_object(
        'media_type', m.media_type,
        'storage_path', m.storage_path,
        'display_order', m.display_order
      )
      ORDER BY m.display_order, m.id
    ) AS items
    FROM public.group_post_media m
    WHERE m.post_id = p.id
      AND m.group_id = g.id
  ) media_items ON true
  WHERE auth.uid() IS NOT NULL
    AND g.visibility = 'public'
    AND p.published
    AND NOT p.is_deleted
    AND parapost_groups_private.allowed(
      g.id, 'read'
    )
    AND NOT parapost_groups_private.blocked(
      auth.uid(), s.user_id
    )
    AND NOT parapost_groups_private.blocked(
      auth.uid(), p.author_id
    )
    AND (
      p_sharer_id IS NULL
      OR s.user_id = p_sharer_id
    )
  ORDER BY s.created_at DESC, s.id DESC
  LIMIT LEAST(GREATEST(COALESCE(p_limit,60),1),100)
  OFFSET LEAST(GREATEST(COALESCE(p_offset,0),0),10000);
$$;

CREATE OR REPLACE FUNCTION public.groups_shared_post_stats(
  p_share_id uuid
)
RETURNS jsonb
LANGUAGE plpgsql
STABLE
SECURITY DEFINER
SET search_path = ''
AS $function$
DECLARE
  v_uid uuid := auth.uid();
  v_group_id uuid;
  v_sharer_id uuid;
  v_author_id uuid;
  v_comments jsonb;
BEGIN
  IF v_uid IS NULL THEN
    RAISE EXCEPTION 'Sign in required'
      USING ERRCODE = '42501';
  END IF;

  SELECT s.group_id, s.user_id, p.author_id
  INTO v_group_id, v_sharer_id, v_author_id
  FROM public.group_post_shares s
  JOIN public.groups g ON g.id = s.group_id
  JOIN public.group_posts p
    ON p.id = s.post_id AND p.group_id = s.group_id
  WHERE s.id = p_share_id
    AND g.visibility = 'public'
    AND NOT p.is_deleted;

  IF NOT FOUND THEN
    RAISE EXCEPTION 'Share unavailable'
      USING ERRCODE = '42501';
  END IF;

  IF NOT parapost_groups_private.allowed(v_group_id, 'read')
     OR parapost_groups_private.blocked(v_uid, v_sharer_id)
     OR parapost_groups_private.blocked(v_uid, v_author_id)
  THEN
    RAISE EXCEPTION 'Share unavailable'
      USING ERRCODE = '42501';
  END IF;

  SELECT COALESCE(
    jsonb_agg(
      jsonb_build_object(
        'id', c.id,
        'author_id', c.author_id,
        'parent_id', c.parent_id,
        'content', c.content,
        'created_at', c.created_at,
        'updated_at', c.updated_at,
        'like_count', c.like_count,
        'liked', c.liked
      )
      ORDER BY c.created_at, c.id
    ),
    '[]'::jsonb
  )
  INTO v_comments
  FROM (
    SELECT
      cm.*,
      (
        SELECT count(*)::integer
        FROM public.group_post_share_comment_likes cl
        WHERE cl.comment_id = cm.id
          AND NOT parapost_groups_private.blocked(v_uid, cl.user_id)
      ) AS like_count,
      EXISTS (
        SELECT 1
        FROM public.group_post_share_comment_likes cl
        WHERE cl.comment_id = cm.id
          AND cl.user_id = v_uid
      ) AS liked
    FROM public.group_post_share_comments cm
    WHERE cm.share_id = p_share_id
      AND NOT parapost_groups_private.blocked(v_uid, cm.author_id)
      AND (
        cm.parent_id IS NULL
        OR EXISTS (
          SELECT 1
          FROM public.group_post_share_comments parent
          WHERE parent.id = cm.parent_id
            AND parent.share_id = cm.share_id
            AND parent.parent_id IS NULL
            AND NOT parapost_groups_private.blocked(
              v_uid, parent.author_id
            )
        )
      )
    ORDER BY cm.created_at, cm.id
    LIMIT 150
  ) c;

  RETURN jsonb_build_object(
    'liked', EXISTS (
      SELECT 1
      FROM public.group_post_share_likes l
      WHERE l.share_id = p_share_id
        AND l.user_id = v_uid
    ),
    'like_count', (
      SELECT count(*)::integer
      FROM public.group_post_share_likes l
      WHERE l.share_id = p_share_id
        AND NOT parapost_groups_private.blocked(v_uid, l.user_id)
    ),
    'comment_count', (
      SELECT count(*)::integer
      FROM public.group_post_share_comments c
      WHERE c.share_id = p_share_id
        AND NOT parapost_groups_private.blocked(v_uid, c.author_id)
        AND (
          c.parent_id IS NULL
          OR EXISTS (
            SELECT 1
            FROM public.group_post_share_comments parent
            WHERE parent.id = c.parent_id
              AND parent.share_id = c.share_id
              AND parent.parent_id IS NULL
              AND NOT parapost_groups_private.blocked(
                v_uid, parent.author_id
              )
          )
        )
    ),
    'comments_limited', (
      SELECT count(*) > 150
      FROM public.group_post_share_comments c
      WHERE c.share_id = p_share_id
        AND NOT parapost_groups_private.blocked(v_uid, c.author_id)
        AND (
          c.parent_id IS NULL
          OR EXISTS (
            SELECT 1
            FROM public.group_post_share_comments parent
            WHERE parent.id = c.parent_id
              AND parent.share_id = c.share_id
              AND parent.parent_id IS NULL
              AND NOT parapost_groups_private.blocked(
                v_uid, parent.author_id
              )
          )
        )
    ),
    'comments', v_comments
  );
END;
$function$;
CREATE OR REPLACE FUNCTION public.groups_shared_post_interact(
  p_share_id uuid,
  p_action text,
  p_content text DEFAULT NULL,
  p_parent_id uuid DEFAULT NULL
)
RETURNS jsonb
LANGUAGE plpgsql
SECURITY DEFINER
SET search_path = ''
AS $function$
DECLARE
  v_uid uuid := auth.uid();
  v_share public.group_post_shares%ROWTYPE;
  v_group public.groups%ROWTYPE;
  v_post public.group_posts%ROWTYPE;
  v_parent public.group_post_share_comments%ROWTYPE;
  v_id uuid;
  v_limit_kind text;
  v_limit_max integer;
  v_limit_now timestamptz;
  v_limit_count integer;
BEGIN
  IF v_uid IS NULL THEN
    RAISE EXCEPTION 'Sign in required'
      USING ERRCODE = '42501';
  END IF;

  SELECT * INTO v_share
  FROM public.group_post_shares
  WHERE id = p_share_id;

  IF NOT FOUND THEN
    RAISE EXCEPTION 'Share unavailable'
      USING ERRCODE = '42501';
  END IF;

  SELECT * INTO v_group
  FROM public.groups
  WHERE id = v_share.group_id
  FOR SHARE;

  IF NOT FOUND
     OR v_group.visibility <> 'public'
     OR NOT parapost_groups_private.allowed(
       v_share.group_id, 'read'
     ) THEN
    RAISE EXCEPTION 'Group unavailable'
      USING ERRCODE = '42501';
  END IF;

  SELECT * INTO v_post
  FROM public.group_posts
  WHERE id = v_share.post_id
    AND group_id = v_share.group_id
    AND NOT is_deleted;

  IF NOT FOUND THEN
    RAISE EXCEPTION 'Post unavailable'
      USING ERRCODE = '42501';
  END IF;

  IF parapost_groups_private.blocked(v_uid, v_share.user_id)
     OR parapost_groups_private.blocked(
       v_uid, v_post.author_id
     ) THEN
    RAISE EXCEPTION 'Interaction unavailable'
      USING ERRCODE = '42501';
  END IF;

  -- Atomic per-user limits across all shared Group posts.
  IF p_action IN ('comment', 'reply') THEN
    v_limit_kind := 'comment';
    v_limit_max := 10;
  ELSIF p_action IN (
    'like', 'unlike', 'like_comment', 'unlike_comment'
  ) THEN
    v_limit_kind := 'reaction';
    v_limit_max := 60;
  END IF;

  IF v_limit_kind IS NOT NULL THEN
    v_limit_now := clock_timestamp();

    INSERT INTO parapost_groups_private.shared_post_interaction_rate_limits AS rl
      (user_id, action_kind, window_started_at, action_count)
    VALUES (v_uid, v_limit_kind, v_limit_now, 1)
    ON CONFLICT (user_id, action_kind)
    DO UPDATE SET
      window_started_at = CASE
        WHEN rl.window_started_at <=
             v_limit_now - interval '60 seconds'
        THEN v_limit_now
        ELSE rl.window_started_at
      END,
      action_count = CASE
        WHEN rl.window_started_at <=
             v_limit_now - interval '60 seconds'
        THEN 1
        ELSE rl.action_count + 1
      END
    WHERE rl.window_started_at <=
          v_limit_now - interval '60 seconds'
       OR rl.action_count < v_limit_max
    RETURNING action_count INTO v_limit_count;

    IF NOT FOUND THEN
      RAISE EXCEPTION 'Rate limit exceeded. Try again shortly.'
        USING ERRCODE = 'P0001';
    END IF;
  END IF;

  IF p_action = 'like' THEN
    INSERT INTO public.group_post_share_likes
      (share_id, user_id)
    VALUES (p_share_id, v_uid)
    ON CONFLICT (share_id, user_id) DO NOTHING;

    RETURN jsonb_build_object('liked', true);

  ELSIF p_action = 'unlike' THEN
    DELETE FROM public.group_post_share_likes
    WHERE share_id = p_share_id
      AND user_id = v_uid;

    RETURN jsonb_build_object('liked', false);

  ELSIF p_action = 'comment' THEN
    IF p_parent_id IS NOT NULL
       OR p_content IS NULL
       OR char_length(btrim(p_content)) NOT BETWEEN 1 AND 1200 THEN
      RAISE EXCEPTION 'Invalid comment';
    END IF;

    INSERT INTO public.group_post_share_comments
      (share_id, author_id, content)
    VALUES (p_share_id, v_uid, btrim(p_content))
    RETURNING id INTO v_id;

    RETURN jsonb_build_object('id', v_id);

  ELSIF p_action IN (
    'reply', 'like_comment', 'unlike_comment',
    'edit_comment', 'delete_comment'
  ) THEN
    IF p_parent_id IS NULL THEN
      RAISE EXCEPTION 'Comment required';
    END IF;

    SELECT * INTO v_parent
    FROM public.group_post_share_comments
    WHERE id = p_parent_id
      AND share_id = p_share_id;

    IF NOT FOUND THEN
      RAISE EXCEPTION 'Comment unavailable'
        USING ERRCODE = '42501';
    END IF;

    IF p_action = 'reply' THEN
      IF v_parent.parent_id IS NOT NULL
         OR parapost_groups_private.blocked(
           v_uid, v_parent.author_id
         )
         OR p_content IS NULL
         OR char_length(btrim(p_content)) NOT BETWEEN 1 AND 1200 THEN
        RAISE EXCEPTION 'Reply unavailable'
          USING ERRCODE = '42501';
      END IF;

      INSERT INTO public.group_post_share_comments
        (share_id, author_id, parent_id, content)
      VALUES (
        p_share_id, v_uid, p_parent_id, btrim(p_content)
      )
      RETURNING id INTO v_id;

      RETURN jsonb_build_object('id', v_id);

    ELSIF p_action = 'like_comment' THEN
      IF parapost_groups_private.blocked(
        v_uid, v_parent.author_id
      )
      OR (
        v_parent.parent_id IS NOT NULL
        AND NOT EXISTS (
          SELECT 1
          FROM public.group_post_share_comments parent
          WHERE parent.id = v_parent.parent_id
            AND parent.share_id = p_share_id
            AND parent.parent_id IS NULL
            AND NOT parapost_groups_private.blocked(
              v_uid, parent.author_id
            )
        )
      ) THEN
        RAISE EXCEPTION 'Comment Like unavailable'
          USING ERRCODE = '42501';
      END IF;

      INSERT INTO public.group_post_share_comment_likes
        (comment_id, share_id, user_id)
      VALUES (p_parent_id, p_share_id, v_uid)
      ON CONFLICT (comment_id, user_id) DO NOTHING;

      RETURN jsonb_build_object('liked', true);

    ELSIF p_action = 'unlike_comment' THEN
      DELETE FROM public.group_post_share_comment_likes
      WHERE comment_id = p_parent_id
        AND share_id = p_share_id
        AND user_id = v_uid;

      RETURN jsonb_build_object('liked', false);

    ELSIF p_action = 'edit_comment' THEN
      IF v_parent.author_id <> v_uid
         OR p_content IS NULL
         OR char_length(btrim(p_content)) NOT BETWEEN 1 AND 1200 THEN
        RAISE EXCEPTION 'Edit unavailable'
          USING ERRCODE = '42501';
      END IF;

      UPDATE public.group_post_share_comments
      SET content = btrim(p_content),
          updated_at = now()
      WHERE id = p_parent_id
        AND share_id = p_share_id
        AND author_id = v_uid;

      RETURN jsonb_build_object('updated', true);

    ELSIF p_action = 'delete_comment' THEN
      IF v_parent.author_id <> v_uid THEN
        RAISE EXCEPTION 'Delete unavailable'
          USING ERRCODE = '42501';
      END IF;

      DELETE FROM public.group_post_share_comments
      WHERE id = p_parent_id
        AND share_id = p_share_id
        AND author_id = v_uid;

      RETURN jsonb_build_object('deleted', true);
    END IF;
  END IF;

  RAISE EXCEPTION 'Unknown interaction action';
END;
$function$;

CREATE FUNCTION creatorverse_private.validate_live_page_owner()
RETURNS trigger
LANGUAGE plpgsql
SECURITY DEFINER
SET search_path = ''
AS $function$
BEGIN
  -- Existing personal broadcasts require no new permission.
  IF NEW.creatorverse_page_id IS NULL THEN
    RETURN NEW;
  END IF;

  -- Only the authenticated broadcaster can associate a page.
  IF auth.uid() IS DISTINCT FROM NEW.user_id THEN
    RAISE EXCEPTION
      'Unauthorized CreatorVerse livestream association'
      USING ERRCODE = '42501';
  END IF;

  -- The page must be active and belong to the broadcaster.
  IF NOT EXISTS (
    SELECT 1
    FROM public.creatorverse_pages AS p
    WHERE p.id = NEW.creatorverse_page_id
      AND p.owner_id = NEW.user_id
      AND p.status = 'active'
  ) THEN
    RAISE EXCEPTION
      'CreatorVerse page is not owned or active'
      USING ERRCODE = '42501';
  END IF;

  RETURN NEW;
END;
$function$;

CREATE FUNCTION public.creatorverse_edit_post(
  p_page_id uuid,
  p_post_id uuid,
  p_content text
)
RETURNS uuid
LANGUAGE plpgsql
SECURITY DEFINER
SET search_path = ''
AS $body$
DECLARE
  v_user uuid := auth.uid();
  v_post uuid;
  v_content text := btrim(p_content);
BEGIN
  IF v_user IS NULL
     OR NOT creatorverse_private.write_enabled()
     OR NOT creatorverse_private.owns(p_page_id)
     OR NOT creatorverse_private.read_page(p_page_id)
  THEN
    RAISE EXCEPTION 'Post management unavailable'
      USING ERRCODE = '42501';
  END IF;

  PERFORM 1 FROM public.creatorverse_pages
  WHERE id = p_page_id FOR UPDATE;

  SELECT id INTO v_post
  FROM public.creatorverse_posts
  WHERE id = p_post_id
    AND page_id = p_page_id
    AND author_id = v_user
    AND status = 'published'
  FOR UPDATE;

  IF v_post IS NULL THEN
    RAISE EXCEPTION 'Post unavailable'
      USING ERRCODE = '42501';
  END IF;

  IF v_content IS NULL
     OR char_length(v_content) > 5000
     OR translate(v_content, E'\n\r\t', '') ~ '[[:cntrl:]]'
  THEN
    RAISE EXCEPTION 'Invalid post content'
      USING ERRCODE = '23514';
  END IF;

  IF v_content = '' AND NOT EXISTS (
    SELECT 1
    FROM public.creatorverse_media
    WHERE page_id = p_page_id
      AND post_id = p_post_id
      AND purpose = 'post'
      AND verification = 'verified'
  ) THEN
    RAISE EXCEPTION 'Post cannot be empty'
      USING ERRCODE = '23514';
  END IF;

  UPDATE public.creatorverse_posts
  SET content = v_content,
      updated_at = now()
  WHERE id = v_post
  RETURNING id INTO v_post;

  RETURN v_post;
END;
$body$;

-- creatorverse_private.guard_published_media
-- Source: /Users/parapost/Documents/parapost-share-count-staging/docs/sql-drafts/creatorverse-atomic-publishing-DRAFT.sql
CREATE FUNCTION creatorverse_private.guard_published_media()
RETURNS trigger
LANGUAGE plpgsql
SECURITY DEFINER
SET search_path = ''
AS $guard$
DECLARE
  v_post_id uuid;
  v_page_id uuid;
  v_status text;
BEGIN
  IF TG_OP = 'DELETE' THEN
    v_post_id := OLD.post_id;
    v_page_id := OLD.page_id;
  ELSE
    v_post_id := NEW.post_id;
    v_page_id := NEW.page_id;

    IF TG_OP = 'UPDATE'
       AND (
         OLD.post_id IS DISTINCT FROM NEW.post_id
         OR OLD.page_id IS DISTINCT FROM NEW.page_id
       )
    THEN
      RAISE EXCEPTION 'Media association is immutable';
    END IF;
  END IF;

  IF v_post_id IS NOT NULL THEN
    SELECT status INTO v_status
    FROM public.creatorverse_posts
    WHERE id = v_post_id
      AND page_id = v_page_id
    FOR SHARE;

    IF v_status = 'published' THEN
      RAISE EXCEPTION
        'Published post media cannot be modified';
    END IF;
  END IF;

  IF TG_OP = 'DELETE' THEN
    RETURN OLD;
  END IF;

  RETURN NEW;
END;
$guard$;

-- creatorverse_private.immutable_identity
-- Source: /Users/parapost/Documents/parapost-share-count-staging/docs/sql-drafts/creatorverse-foundation-schema.sql
CREATE FUNCTION creatorverse_private.immutable_identity() RETURNS trigger
LANGUAGE plpgsql SET search_path='' AS $$
BEGIN
  IF NEW.id IS DISTINCT FROM OLD.id OR NEW.created_at IS DISTINCT FROM OLD.created_at THEN
    RAISE EXCEPTION 'Immutable identity';
  END IF;
  IF TG_TABLE_NAME='creatorverse_pages' THEN
    IF NEW.owner_id IS DISTINCT FROM OLD.owner_id THEN RAISE EXCEPTION 'Ownership transfers are not supported'; END IF;
  ELSIF TG_TABLE_NAME='creatorverse_posts' THEN
    IF NEW.page_id IS DISTINCT FROM OLD.page_id OR NEW.author_id IS DISTINCT FROM OLD.author_id THEN RAISE EXCEPTION 'Immutable post identity'; END IF;
  ELSIF TG_TABLE_NAME='creatorverse_comments' THEN
    IF NEW.post_id IS DISTINCT FROM OLD.post_id OR NEW.author_id IS DISTINCT FROM OLD.author_id OR NEW.parent_id IS DISTINCT FROM OLD.parent_id THEN RAISE EXCEPTION 'Immutable comment identity'; END IF;
  END IF;
  NEW.updated_at=now(); RETURN NEW;
END; $$;

-- creatorverse_private.interact
-- Source: /Users/parapost/Documents/parapost-share-count-staging/docs/sql-drafts/creatorverse-phase3-integration-DRAFT.sql
CREATE FUNCTION creatorverse_private.interact(page uuid, post uuid, action text, target uuid DEFAULT NULL, body text DEFAULT '', request uuid DEFAULT NULL)
RETURNS uuid LANGUAGE plpgsql VOLATILE SECURITY DEFINER SET search_path='' AS $$
DECLARE result uuid; previous public.creatorverse_shares%ROWTYPE;
BEGIN
 PERFORM 1 FROM public.creatorverse_pages WHERE id=page FOR UPDATE;
 PERFORM 1 FROM public.creatorverse_posts WHERE id=post AND page_id=page FOR UPDATE;
 IF action NOT IN ('like','unlike','comment','delete_comment','share')
    OR NOT creatorverse_private.operation_allowed(page,post,action,target)
 THEN RAISE EXCEPTION 'Operation unavailable' USING ERRCODE='42501'; END IF;
 IF action='like' THEN
   INSERT INTO public.creatorverse_post_likes(post_id,user_id) VALUES(post,auth.uid()) ON CONFLICT DO NOTHING;
   RETURN post;
 ELSIF action='unlike' THEN
   DELETE FROM public.creatorverse_post_likes WHERE post_id=post AND user_id=auth.uid(); RETURN post;
 ELSIF action='comment' THEN
   IF NOT creatorverse_private.valid_text(body,1200,1) THEN RAISE EXCEPTION 'Invalid comment' USING ERRCODE='22023'; END IF;
   INSERT INTO public.creatorverse_comments(post_id,author_id,parent_id,content)
   VALUES(post,auth.uid(),target,btrim(body)) RETURNING id INTO result;
 ELSIF action='delete_comment' THEN
   UPDATE public.creatorverse_comments SET is_deleted=true,content='[deleted]'
   WHERE post_id=post AND (id=target OR parent_id=target); result:=target;
 ELSE
   IF request IS NULL OR NOT creatorverse_private.valid_text(body,1000)
   THEN RAISE EXCEPTION 'Invalid share' USING ERRCODE='22023'; END IF;
   -- Serialize one sharer's idempotency requests even across different pages.
   PERFORM pg_catalog.pg_advisory_xact_lock(pg_catalog.hashtextextended(auth.uid()::text, 0));
   SELECT * INTO previous FROM public.creatorverse_shares WHERE sharer_id=auth.uid() AND request_id=request;
   IF FOUND THEN
     IF previous.post_id<>post OR previous.caption<>btrim(body)
     THEN RAISE EXCEPTION 'Request already used' USING ERRCODE='23505'; END IF;
     RETURN previous.id;
   END IF;
   -- Existing foundation has one canonical share per sharer/post. No Group cooldown changes.
   INSERT INTO public.creatorverse_shares(post_id,sharer_id,caption,request_id)
   VALUES(post,auth.uid(),btrim(body),request) RETURNING id INTO result;
 END IF;
 RETURN result;
END; $$;

-- creatorverse_private.interact_once
-- Source: /Users/parapost/Documents/parapost-share-count-staging/docs/sql-drafts/creatorverse-phase4-social-DRAFT.sql
CREATE FUNCTION creatorverse_private.interact_once(page uuid, post uuid, action text, target uuid, body text, request uuid)
RETURNS uuid LANGUAGE plpgsql VOLATILE SECURITY DEFINER SET search_path='' AS $$
DECLARE previous creatorverse_private.social_requests%ROWTYPE; result uuid; submitted jsonb;
BEGIN
 IF request IS NULL OR action NOT IN ('like','unlike','comment','delete_comment','share')
 THEN RAISE EXCEPTION 'Invalid interaction request' USING ERRCODE='22023'; END IF;
 PERFORM 1 FROM public.creatorverse_pages WHERE id=page FOR UPDATE;
 PERFORM 1 FROM public.creatorverse_posts WHERE id=post AND page_id=page FOR UPDATE;
 -- Recheck visibility/eligibility even for retries. Deleted-comment retries are
 -- allowed only to the original actor through the ledger after general access checks.
 IF NOT creatorverse_private.write_enabled() OR NOT creatorverse_private.read_post(post)
    OR NOT EXISTS(SELECT 1 FROM public.creatorverse_posts WHERE id=post AND page_id=page)
 THEN RAISE EXCEPTION 'Operation unavailable' USING ERRCODE='42501'; END IF;
 PERFORM pg_catalog.pg_advisory_xact_lock(pg_catalog.hashtextextended(auth.uid()::text||request::text,0));
 submitted:=jsonb_build_object('page',page,'post',post,'action',action,'target',target,'body',body);
 SELECT * INTO previous FROM creatorverse_private.social_requests WHERE actor_id=auth.uid() AND request_id=request;
 IF FOUND THEN
   IF previous.payload<>submitted THEN RAISE EXCEPTION 'Request already used' USING ERRCODE='23505'; END IF;
   RETURN previous.result_id;
 END IF;
 result:=creatorverse_private.interact(page,post,action,target,body,request);
 INSERT INTO creatorverse_private.social_requests(actor_id,request_id,payload,result_id) VALUES(auth.uid(),request,submitted,result);
 RETURN result;
END; $$;

-- creatorverse_private.mutate_post
-- Source: /Users/parapost/Documents/parapost-share-count-staging/docs/sql-drafts/creatorverse-phase3-integration-DRAFT.sql
CREATE FUNCTION creatorverse_private.mutate_post(page uuid, post uuid, action text, body text DEFAULT '')
RETURNS uuid LANGUAGE plpgsql VOLATILE SECURITY DEFINER SET search_path='' AS $$
DECLARE result uuid; current_status text;
BEGIN
 PERFORM 1 FROM public.creatorverse_pages WHERE id=page FOR UPDATE;
 SELECT status INTO current_status FROM public.creatorverse_posts WHERE id=post AND page_id=page FOR UPDATE;
 IF action NOT IN ('create_post','edit_post','delete_post') OR NOT creatorverse_private.operation_allowed(page,post,action)
 THEN RAISE EXCEPTION 'Operation unavailable' USING ERRCODE='42501'; END IF;
 IF NOT creatorverse_private.valid_text(body,5000) THEN RAISE EXCEPTION 'Invalid content' USING ERRCODE='22023'; END IF;
 IF action='create_post' THEN
   INSERT INTO public.creatorverse_posts(page_id,author_id,content,status)
   VALUES(page,auth.uid(),btrim(body),'draft') RETURNING id INTO result;
 ELSIF action='edit_post' THEN
   IF current_status='published' AND btrim(body)='' AND NOT EXISTS
     (SELECT 1 FROM public.creatorverse_media WHERE post_id=post AND page_id=page AND verification='verified')
   THEN RAISE EXCEPTION 'Published post cannot be empty' USING ERRCODE='23514'; END IF;
   UPDATE public.creatorverse_posts SET content=btrim(body) WHERE id=post AND page_id=page RETURNING id INTO result;
 ELSE
   -- Tombstone first; all projections and storage read policies recheck status.
   UPDATE public.creatorverse_posts SET status='deleted' WHERE id=post AND page_id=page RETURNING id INTO result;
   INSERT INTO creatorverse_private.media_cleanup(storage_path,next_attempt_at)
     SELECT storage_path,now()+interval '20 minutes' FROM public.creatorverse_media WHERE post_id=post AND page_id=page
     ON CONFLICT(storage_path) DO UPDATE SET next_attempt_at=greatest(creatorverse_private.media_cleanup.next_attempt_at,excluded.next_attempt_at);
   DELETE FROM creatorverse_private.upload_reservations WHERE page_id=page AND media_id IN
     (SELECT id FROM public.creatorverse_media WHERE post_id=post AND page_id=page);
 END IF;
 RETURN result;
END; $$;

-- creatorverse_private.operation_allowed
-- Source: /Users/parapost/Documents/parapost-share-count-staging/docs/sql-drafts/creatorverse-phase3-integration-DRAFT.sql
CREATE FUNCTION creatorverse_private.operation_allowed(page uuid, post uuid, action text, target uuid DEFAULT NULL)
RETURNS boolean LANGUAGE sql STABLE SECURITY DEFINER SET search_path='' AS $$
 SELECT creatorverse_private.write_enabled() AND CASE
  WHEN action='create_post' THEN post IS NULL AND creatorverse_private.owns(page)
  WHEN action IN ('edit_post','delete_post','reserve_media','delete_media') THEN
    creatorverse_private.owns(page) AND EXISTS (
      SELECT 1 FROM public.creatorverse_posts p WHERE p.id=post AND p.page_id=page
      AND p.author_id=auth.uid() AND p.status IN ('draft','published')
      AND (action NOT IN ('reserve_media','delete_media') OR p.status='draft')
      AND (action<>'delete_media' OR EXISTS (SELECT 1 FROM public.creatorverse_media m
        WHERE m.id=target AND m.page_id=page AND m.post_id=post AND m.purpose='post')))
  WHEN action IN ('like','unlike','comment','delete_comment','share') THEN
    creatorverse_private.read_post(post) AND EXISTS (
      SELECT 1 FROM public.creatorverse_posts p WHERE p.id=post AND p.page_id=page)
    AND CASE WHEN action='comment' THEN creatorverse_private.comment_allowed(post,target)
      AND (target IS NULL OR EXISTS (SELECT 1 FROM public.creatorverse_comments c
        WHERE c.id=target AND c.post_id=post AND c.parent_id IS NULL AND NOT c.is_deleted))
    WHEN action='delete_comment' THEN EXISTS (SELECT 1 FROM public.creatorverse_comments c
      WHERE c.id=target AND c.post_id=post AND NOT c.is_deleted
      AND NOT creatorverse_private.blocked(c.author_id)
      AND (c.author_id=auth.uid() OR creatorverse_private.owns(page)))
    ELSE true END
  ELSE false END;
$$;

-- creatorverse_private.page_image_storage_allowed
-- Source: /Users/parapost/Documents/parapost-share-count-staging/docs/sql-drafts/creatorverse-page-image-storage-INSTALL-STAGING.sql
CREATE FUNCTION creatorverse_private.page_image_storage_allowed(
  p_path text,
  p_operation text
)
RETURNS boolean
LANGUAGE plpgsql
STABLE
SECURITY DEFINER
SET search_path TO ''
AS $function$
DECLARE
  v_user uuid := auth.uid();
BEGIN
  IF p_path IS NULL
     OR p_operation IS NULL
  THEN
    RETURN false;
  END IF;

  -- Uploads require an authenticated page owner,
  -- an active reservation, and an unverified image.
  IF p_operation = 'upload' THEN

    IF v_user IS NULL
       OR NOT creatorverse_private.write_enabled()
    THEN
      RETURN false;
    END IF;

    RETURN EXISTS (
      SELECT 1
      FROM public.creatorverse_media m
      JOIN public.creatorverse_pages p
        ON p.id = m.page_id
      JOIN creatorverse_private.upload_reservations r
        ON r.page_id = m.page_id
       AND r.media_id = m.id
      WHERE m.storage_path = p_path
        AND m.purpose IN ('logo', 'cover')
        AND m.post_id IS NULL
        AND m.media_type = 'image'
        AND m.mime_type IN (
          'image/jpeg',
          'image/png',
          'image/webp'
        )
        AND m.verification = 'pending'
        AND m.storage_path = (
          m.page_id::text || '/' ||
          m.purpose || '/' ||
          m.id::text || '.' ||
          CASE m.mime_type
            WHEN 'image/jpeg' THEN 'jpg'
            WHEN 'image/png' THEN 'png'
            WHEN 'image/webp' THEN 'webp'
          END
        )
        AND p.owner_id = v_user
        AND p.status IN ('draft', 'active')
        AND creatorverse_private.owns(p.id)
        AND r.owner_id = v_user
        AND r.consumed_at IS NULL
        AND r.expires_at >
          pg_catalog.statement_timestamp()
    );

  ELSIF p_operation = 'read' THEN

    -- Public access only to verified images
    -- permanently attached to active pages.
    RETURN EXISTS (
      SELECT 1
      FROM public.creatorverse_media m
      JOIN public.creatorverse_pages p
        ON p.id = m.page_id
      WHERE m.storage_path = p_path
        AND m.purpose IN ('logo', 'cover')
        AND m.post_id IS NULL
        AND m.media_type = 'image'
        AND m.verification = 'verified'
        AND p.status = 'active'
        AND (
          (m.purpose = 'logo'
           AND p.logo_media_id = m.id)
          OR
          (m.purpose = 'cover'
           AND p.cover_media_id = m.id)
        )
        AND creatorverse_private.read_page(p.id)
    );

  END IF;

  RETURN false;
END;
$function$;

-- creatorverse_private.post_counts
-- Source: /Users/parapost/Documents/parapost-share-count-staging/docs/sql-drafts/creatorverse-phase3-integration-DRAFT.sql
CREATE FUNCTION creatorverse_private.post_counts(post uuid)
RETURNS jsonb LANGUAGE sql STABLE SECURITY DEFINER SET search_path='' AS $$
 SELECT CASE WHEN creatorverse_private.read_post(post) THEN jsonb_build_object(
  'likes',(SELECT count(*) FROM public.creatorverse_post_likes WHERE post_id=post AND NOT creatorverse_private.blocked(user_id)),
  'comments',(SELECT count(*) FROM public.creatorverse_comments WHERE post_id=post AND creatorverse_private.read_comment(id)),
  'shares',(SELECT count(*) FROM public.creatorverse_shares WHERE post_id=post AND NOT creatorverse_private.blocked(sharer_id)),
  'liked',(SELECT EXISTS(SELECT 1 FROM public.creatorverse_post_likes WHERE post_id=post AND user_id=auth.uid()))
 ) ELSE NULL END;
$$;

-- creatorverse_private.publish_post_draft
-- Source: /Users/parapost/Documents/parapost-share-count-staging/docs/sql-drafts/creatorverse-atomic-publishing-DRAFT.sql
CREATE FUNCTION creatorverse_private.publish_post_draft(
  p_page_id uuid,
  p_post_id uuid
)
RETURNS uuid
LANGUAGE plpgsql
VOLATILE
SECURITY DEFINER
SET search_path = ''
AS $publish$
DECLARE
  v_user_id uuid := auth.uid();
  v_content text;
  v_media_count integer;
  v_bad_media integer;
  v_result uuid;
BEGIN
  IF v_user_id IS NULL
     OR NOT creatorverse_private.write_enabled()
     OR NOT creatorverse_private.owns(p_page_id)
     OR NOT creatorverse_private.read_page(p_page_id)
  THEN
    RAISE EXCEPTION 'Publishing unavailable'
      USING ERRCODE = '42501';
  END IF;

  -- Lock the post. Concurrent media attachment through
  -- its foreign key must wait for publication to finish.
  SELECT content INTO v_content
  FROM public.creatorverse_posts
  WHERE id = p_post_id
    AND page_id = p_page_id
    AND author_id = v_user_id
    AND status = 'draft'
  FOR UPDATE;

  IF NOT FOUND THEN
    RAISE EXCEPTION 'Draft not available'
      USING ERRCODE = '42501';
  END IF;

  -- Lock existing media against concurrent verification
  -- or metadata changes while inspecting its readiness.
  PERFORM 1
  FROM public.creatorverse_media
  WHERE page_id = p_page_id
    AND post_id = p_post_id
  FOR SHARE;

  SELECT COUNT(*)::integer,
         COUNT(*) FILTER (
           WHERE verification <> 'verified'
              OR purpose <> 'post'
              OR byte_size IS NULL
              OR (media_type = 'video'
                  AND duration_seconds IS NULL)
         )::integer
  INTO v_media_count, v_bad_media
  FROM public.creatorverse_media
  WHERE page_id = p_page_id
    AND post_id = p_post_id;

  IF v_media_count > 4 OR v_bad_media > 0 THEN
    RAISE EXCEPTION 'Post media is not ready'
      USING ERRCODE = '23514';
  END IF;

  IF (
    char_length(btrim(v_content)) = 0
    AND v_media_count = 0
  ) OR char_length(v_content) > 5000 THEN
    RAISE EXCEPTION 'Invalid post content'
      USING ERRCODE = '23514';
  END IF;

  UPDATE public.creatorverse_posts
  SET status = 'published',
      updated_at = now()
  WHERE id = p_post_id
    AND page_id = p_page_id
    AND author_id = v_user_id
    AND status = 'draft'
  RETURNING id INTO v_result;

  IF v_result IS NULL THEN
    RAISE EXCEPTION 'Publication failed'
      USING ERRCODE = '42501';
  END IF;

  RETURN v_result;
END;
$publish$;

CREATE FUNCTION creatorverse_private.reserve_media(page uuid, post uuid, mime text, bytes bigint, slot integer)
RETURNS jsonb LANGUAGE plpgsql VOLATILE SECURITY DEFINER SET search_path='' AS $$
DECLARE media uuid:=gen_random_uuid(); path text; kind text; reservation uuid; expiry timestamptz:=now()+interval '15 minutes';
BEGIN
 PERFORM 1 FROM public.creatorverse_pages WHERE id=page FOR UPDATE;
 PERFORM 1 FROM public.creatorverse_posts WHERE id=post AND page_id=page FOR UPDATE;
 IF NOT creatorverse_private.operation_allowed(page,post,'reserve_media')
 THEN RAISE EXCEPTION 'Operation unavailable' USING ERRCODE='42501'; END IF;
 IF mime IS NULL OR mime NOT IN ('image/jpeg','image/png','image/webp','video/mp4','video/webm')
   OR bytes IS NULL OR bytes<=0 OR bytes>(CASE WHEN mime LIKE 'image/%' THEN 8388608 ELSE 33554432 END)
   OR slot IS NULL OR slot NOT BETWEEN 0 AND 3
 THEN RAISE EXCEPTION 'Invalid media' USING ERRCODE='22023'; END IF;
 IF (SELECT count(*) FROM public.creatorverse_media WHERE page_id=page AND post_id=post)>=4
 THEN RAISE EXCEPTION 'Attachment limit' USING ERRCODE='23514'; END IF;
 kind:=CASE WHEN mime LIKE 'image/%' THEN 'image' ELSE 'video' END;
 path:=page::text||'/'||media::text;
 INSERT INTO public.creatorverse_media(id,page_id,post_id,purpose,media_type,storage_path,mime_type,byte_size,display_order)
 VALUES(media,page,post,'post',kind,path,mime,bytes,slot);
 INSERT INTO creatorverse_private.upload_reservations(page_id,media_id,owner_id,expires_at)
 VALUES(page,media,auth.uid(),expiry) RETURNING id INTO reservation;
 RETURN jsonb_build_object('media_id',media,'reservation_id',reservation,'bucket','creatorverse-private','path',path,'expires_at',expiry);
END; $$;

-- creatorverse_private.storage_allowed
-- Source: /Users/parapost/Documents/parapost-share-count-staging/docs/sql-drafts/creatorverse-page-image-storage-INSTALL-STAGING.sql
CREATE OR REPLACE FUNCTION creatorverse_private.storage_allowed(
  p_path text,
  p_operation text
)
RETURNS boolean
LANGUAGE plpgsql
STABLE
SECURITY DEFINER
SET search_path TO ''
AS $function$
DECLARE
  v_user uuid := auth.uid();
BEGIN
  IF v_user IS NULL
     OR p_path IS NULL
     OR p_operation IS NULL
  THEN
    RETURN false;
  END IF;

  IF p_operation = 'upload' THEN

    IF NOT creatorverse_private.write_enabled() THEN
      RETURN false;
    END IF;

    -- Existing post-media upload authorization.
    IF EXISTS (
      SELECT 1
      FROM public.creatorverse_media m
      JOIN public.creatorverse_posts p
        ON p.id = m.post_id
       AND p.page_id = m.page_id
      JOIN public.creatorverse_pages page
        ON page.id = m.page_id
      JOIN creatorverse_private.upload_reservations r
        ON r.media_id = m.id
       AND r.page_id = m.page_id
      WHERE m.storage_path = p_path
        AND m.purpose = 'post'
        AND m.verification = 'pending'
        AND p.status = 'draft'
        AND p.author_id = v_user
        AND page.owner_id = v_user
        AND page.status IN ('draft', 'active')
        AND creatorverse_private.owns(page.id)
        AND r.owner_id = v_user
        AND r.consumed_at IS NULL
        AND r.expires_at >
          pg_catalog.statement_timestamp()
    ) THEN
      RETURN true;
    END IF;

    -- New reserved profile and cover image authorization.
    RETURN creatorverse_private.page_image_storage_allowed(
      p_path, 'upload'
    );

  ELSIF p_operation = 'read' THEN

    -- Existing published post-media read authorization.
    IF EXISTS (
      SELECT 1
      FROM public.creatorverse_media m
      JOIN public.creatorverse_posts p
        ON p.id = m.post_id
       AND p.page_id = m.page_id
      JOIN public.creatorverse_pages page
        ON page.id = m.page_id
      WHERE m.storage_path = p_path
        AND m.purpose = 'post'
        AND m.verification = 'verified'
        AND p.status = 'published'
        AND page.status = 'active'
        AND creatorverse_private.read_post(p.id)
    ) THEN
      RETURN true;
    END IF;

    -- New verified and attached page-image read authorization.
    RETURN creatorverse_private.page_image_storage_allowed(
      p_path, 'read'
    );

  END IF;

  RETURN false;
END;
$function$;

-- parapost_groups_private.promotion_interact
-- Source: /Users/parapost/Documents/parapost-share-count-staging/docs/sql-drafts/groups-promotional-engagement-staging.sql
CREATE OR REPLACE FUNCTION parapost_groups_private.promotion_interact(
  p_share_id uuid,
  p_action text,
  p_content text DEFAULT NULL,
  p_parent_id uuid DEFAULT NULL
)
RETURNS jsonb
LANGUAGE plpgsql
SECURITY DEFINER
SET search_path = ''
AS $function$
DECLARE
  v_uid uuid := auth.uid();
  v_share public.group_promotional_shares%ROWTYPE;
  v_group public.groups%ROWTYPE;
  v_parent public.group_promotion_share_comments%ROWTYPE;
  v_id uuid;
  v_limit_kind text;
  v_limit_max integer;
  v_limit_now timestamptz;
  v_limit_count integer;
BEGIN
  IF v_uid IS NULL THEN
    RAISE EXCEPTION 'Sign in required'
      USING ERRCODE = '42501';
  END IF;

  SELECT * INTO v_share FROM public.group_promotional_shares WHERE id = p_share_id;
  IF NOT FOUND THEN
    RAISE EXCEPTION 'Share unavailable' USING ERRCODE = '42501';
  END IF;
  -- Match the Group lock used by privacy/membership commands, then lock the
  -- share against concurrent deletion. Never authorize using client metadata.
  SELECT * INTO v_group FROM public.groups WHERE id = v_share.group_id FOR SHARE;
  PERFORM 1 FROM public.group_promotional_shares WHERE id = p_share_id FOR SHARE;
  IF NOT FOUND OR parapost_groups_private.promotion_preview(v_share.group_id, p_share_id) IS NULL THEN
    RAISE EXCEPTION 'Share unavailable' USING ERRCODE = '42501';
  END IF;

  -- Reuse the existing per-member limits across both kinds of shared cards.
  IF p_action IN ('comment', 'reply') THEN
    v_limit_kind := 'comment';
    v_limit_max := 10;
  ELSIF p_action IN (
    'like', 'unlike', 'like_comment', 'unlike_comment'
  ) THEN
    v_limit_kind := 'reaction';
    v_limit_max := 60;
  END IF;

  IF v_limit_kind IS NOT NULL THEN
    v_limit_now := clock_timestamp();

    INSERT INTO parapost_groups_private.shared_post_interaction_rate_limits AS rl
      (user_id, action_kind, window_started_at, action_count)
    VALUES (v_uid, v_limit_kind, v_limit_now, 1)
    ON CONFLICT (user_id, action_kind)
    DO UPDATE SET
      window_started_at = CASE
        WHEN rl.window_started_at <=
             v_limit_now - interval '60 seconds'
        THEN v_limit_now
        ELSE rl.window_started_at
      END,
      action_count = CASE
        WHEN rl.window_started_at <=
             v_limit_now - interval '60 seconds'
        THEN 1
        ELSE rl.action_count + 1
      END
    WHERE rl.window_started_at <=
          v_limit_now - interval '60 seconds'
       OR rl.action_count < v_limit_max
    RETURNING action_count INTO v_limit_count;

    IF NOT FOUND THEN
      RAISE EXCEPTION 'Rate limit exceeded. Try again shortly.'
        USING ERRCODE = 'P0001';
    END IF;
  END IF;

  IF p_action = 'like' THEN
    INSERT INTO public.group_promotion_share_likes
      (share_id, user_id)
    VALUES (p_share_id, v_uid)
    ON CONFLICT (share_id, user_id) DO NOTHING;

    RETURN jsonb_build_object('liked', true);

  ELSIF p_action = 'unlike' THEN
    DELETE FROM public.group_promotion_share_likes
    WHERE share_id = p_share_id
      AND user_id = v_uid;

    RETURN jsonb_build_object('liked', false);

  ELSIF p_action = 'comment' THEN
    IF p_parent_id IS NOT NULL
       OR p_content IS NULL
       OR char_length(btrim(p_content)) NOT BETWEEN 1 AND 1200 THEN
      RAISE EXCEPTION 'Invalid comment';
    END IF;

    INSERT INTO public.group_promotion_share_comments
      (share_id, author_id, content)
    VALUES (p_share_id, v_uid, btrim(p_content))
    RETURNING id INTO v_id;

    RETURN jsonb_build_object('id', v_id);

  ELSIF p_action IN (
    'reply', 'like_comment', 'unlike_comment',
    'edit_comment', 'delete_comment'
  ) THEN
    IF p_parent_id IS NULL THEN
      RAISE EXCEPTION 'Comment required';
    END IF;

    SELECT * INTO v_parent
    FROM public.group_promotion_share_comments
    WHERE id = p_parent_id
      AND share_id = p_share_id;

    IF NOT FOUND THEN
      RAISE EXCEPTION 'Comment unavailable'
        USING ERRCODE = '42501';
    END IF;

    IF parapost_groups_private.blocked(v_uid, v_parent.author_id)
       OR (v_parent.parent_id IS NOT NULL AND NOT EXISTS (
         SELECT 1 FROM public.group_promotion_share_comments parent
         WHERE parent.id = v_parent.parent_id AND parent.share_id = p_share_id
           AND parent.parent_id IS NULL
           AND NOT parapost_groups_private.blocked(v_uid, parent.author_id)
       )) THEN
      RAISE EXCEPTION 'Comment unavailable' USING ERRCODE = '42501';
    END IF;

    IF p_action = 'reply' THEN
      IF v_parent.parent_id IS NOT NULL
         OR parapost_groups_private.blocked(
           v_uid, v_parent.author_id
         )
         OR p_content IS NULL
         OR char_length(btrim(p_content)) NOT BETWEEN 1 AND 1200 THEN
        RAISE EXCEPTION 'Reply unavailable'
          USING ERRCODE = '42501';
      END IF;

      INSERT INTO public.group_promotion_share_comments
        (share_id, author_id, parent_id, content)
      VALUES (
        p_share_id, v_uid, p_parent_id, btrim(p_content)
      )
      RETURNING id INTO v_id;

      RETURN jsonb_build_object('id', v_id);

    ELSIF p_action = 'like_comment' THEN
      IF parapost_groups_private.blocked(
        v_uid, v_parent.author_id
      )
      OR (
        v_parent.parent_id IS NOT NULL
        AND NOT EXISTS (
          SELECT 1
          FROM public.group_promotion_share_comments parent
          WHERE parent.id = v_parent.parent_id
            AND parent.share_id = p_share_id
            AND parent.parent_id IS NULL
            AND NOT parapost_groups_private.blocked(
              v_uid, parent.author_id
            )
        )
      ) THEN
        RAISE EXCEPTION 'Comment Like unavailable'
          USING ERRCODE = '42501';
      END IF;

      INSERT INTO public.group_promotion_share_comment_likes
        (comment_id, share_id, user_id)
      VALUES (p_parent_id, p_share_id, v_uid)
      ON CONFLICT (comment_id, user_id) DO NOTHING;

      RETURN jsonb_build_object('liked', true);

    ELSIF p_action = 'unlike_comment' THEN
      DELETE FROM public.group_promotion_share_comment_likes
      WHERE comment_id = p_parent_id
        AND share_id = p_share_id
        AND user_id = v_uid;

      RETURN jsonb_build_object('liked', false);

    ELSIF p_action = 'edit_comment' THEN
      IF v_parent.author_id <> v_uid
         OR p_content IS NULL
         OR char_length(btrim(p_content)) NOT BETWEEN 1 AND 1200 THEN
        RAISE EXCEPTION 'Edit unavailable'
          USING ERRCODE = '42501';
      END IF;

      UPDATE public.group_promotion_share_comments
      SET content = btrim(p_content),
          updated_at = now()
      WHERE id = p_parent_id
        AND share_id = p_share_id
        AND author_id = v_uid;

      RETURN jsonb_build_object('updated', true);

    ELSIF p_action = 'delete_comment' THEN
      IF v_parent.author_id <> v_uid THEN
        RAISE EXCEPTION 'Delete unavailable'
          USING ERRCODE = '42501';
      END IF;

      DELETE FROM public.group_promotion_share_comments
      WHERE id = p_parent_id
        AND share_id = p_share_id
        AND author_id = v_uid;

      RETURN jsonb_build_object('deleted', true);
    END IF;
  END IF;

  RAISE EXCEPTION 'Unknown interaction action';
END;
$function$;

-- parapost_groups_private.promotion_preview
-- Source: /Users/parapost/Documents/parapost-share-count-staging/docs/sql-drafts/groups-promotional-sharing.sql
CREATE FUNCTION parapost_groups_private.promotion_preview(gid uuid, source_id uuid DEFAULT NULL) RETURNS jsonb
LANGUAGE plpgsql STABLE SECURITY DEFINER SET search_path='' AS $$
DECLARE g public.groups; s parapost_groups_private.promotion_settings; m public.group_members;
        source public.group_promotional_shares; post public.group_posts; expose boolean;
BEGIN
 IF NOT parapost_groups_private.enabled() THEN RETURN NULL; END IF;
 SELECT * INTO g FROM public.groups WHERE id=gid;
 IF NOT FOUND OR parapost_groups_private.blocked(auth.uid(),g.owner_id) THEN RETURN NULL; END IF;
 SELECT * INTO m FROM public.group_members WHERE group_id=gid AND user_id=auth.uid();
 IF m.status IN ('rejected','removed','banned') THEN RETURN NULL; END IF;
 SELECT * INTO s FROM parapost_groups_private.promotion_settings WHERE group_id=gid;
 IF g.visibility='private' AND NOT coalesce(s.enabled,false) THEN RETURN NULL; END IF;
 IF source_id IS NOT NULL THEN
   SELECT * INTO source FROM public.group_promotional_shares WHERE id=source_id AND group_id=gid;
   IF NOT FOUND OR parapost_groups_private.blocked(auth.uid(),source.user_id)
     OR NOT parapost_groups_private.promotion_actor(gid,source.user_id) THEN RETURN NULL; END IF;
   IF source.post_id IS NOT NULL THEN
     SELECT * INTO post FROM public.group_posts WHERE id=source.post_id AND group_id=gid AND published AND NOT is_deleted;
     IF NOT FOUND OR parapost_groups_private.blocked(auth.uid(),post.author_id)
       OR parapost_groups_private.blocked(source.user_id,post.author_id)
       OR NOT EXISTS(SELECT 1 FROM public.group_members WHERE group_id=gid AND user_id=source.user_id AND status='approved') THEN RETURN NULL; END IF;
   END IF;
 END IF;
 expose := g.visibility='public';
 RETURN jsonb_build_object(
   'group_id',gid,'visibility',g.visibility,
   'name',CASE WHEN expose OR s.show_name THEN g.name ELSE 'Private Group' END,
   'description',CASE WHEN expose OR s.show_description THEN g.description ELSE NULL END,
   'member_count',CASE WHEN expose THEN (SELECT count(*) FROM public.group_members WHERE group_id=gid AND status='approved') ELSE NULL END,
   'cover_key',CASE WHEN (expose OR s.show_cover) AND g.cover_storage_path IS NOT NULL THEN md5(g.cover_storage_path) ELSE NULL END,
   'can_share',parapost_groups_private.promotion_actor(gid,auth.uid()) AND
     (source.post_id IS NULL OR (m.status='approved' AND NOT parapost_groups_private.blocked(auth.uid(),post.author_id))),
   'approved',coalesce(m.status='approved',false),
   'post_id',CASE WHEN m.status='approved' THEN source.post_id ELSE NULL END,
   'kind',CASE WHEN source.post_id IS NULL THEN 'group' ELSE 'invitation' END
 );
END;
$$;

CREATE FUNCTION parapost_groups_private.promotion_settings_command(gid uuid, changes jsonb DEFAULT NULL) RETURNS jsonb
LANGUAGE plpgsql SECURITY DEFINER SET search_path='' AS $$

 -- Same Group row lock as the existing membership/visibility commands.

-- parapost_groups_private.promotion_share
-- Source: /Users/parapost/Documents/parapost-share-count-staging/docs/sql-drafts/groups-promotional-sharing.sql
CREATE FUNCTION parapost_groups_private.promotion_share(gid uuid, pid uuid DEFAULT NULL, message text DEFAULT NULL, source_id uuid DEFAULT NULL) RETURNS jsonb
LANGUAGE plpgsql SECURITY DEFINER SET search_path='' AS $$
DECLARE sid uuid; source public.group_promotional_shares; author uuid;
BEGIN
 PERFORM 1 FROM public.groups WHERE id=gid FOR UPDATE;
 IF NOT parapost_groups_private.enabled() OR NOT parapost_groups_private.promotion_actor(gid,auth.uid()) THEN RAISE EXCEPTION 'Sharing unavailable' USING ERRCODE='42501'; END IF;
 IF char_length(message)>1000 THEN RAISE EXCEPTION 'Message too long'; END IF;
 IF source_id IS NOT NULL THEN
   IF parapost_groups_private.promotion_preview(gid,source_id) IS NULL THEN RAISE EXCEPTION 'Share unavailable' USING ERRCODE='42501'; END IF;
   SELECT * INTO source FROM public.group_promotional_shares WHERE id=source_id AND group_id=gid;
   IF NOT FOUND THEN RAISE EXCEPTION 'Share unavailable' USING ERRCODE='42501'; END IF;
   pid := source.post_id; -- Ignore any client-supplied target when resharing.
 END IF;
 IF pid IS NOT NULL THEN
   SELECT author_id INTO author FROM public.group_posts WHERE id=pid AND group_id=gid AND published AND NOT is_deleted;
   IF NOT FOUND OR NOT parapost_groups_private.allowed(gid,'member') OR parapost_groups_private.blocked(auth.uid(),author) THEN RAISE EXCEPTION 'Post unavailable' USING ERRCODE='42501'; END IF;
   -- Public posts retain the established single-record public-post sharing system.
   IF (SELECT visibility FROM public.groups WHERE id=gid)='public' THEN
     RETURN public.groups_share_post(gid,pid,message);
   END IF;
 END IF;
 INSERT INTO public.group_promotional_shares(group_id,post_id,user_id,caption) VALUES(gid,pid,auth.uid(),nullif(btrim(message),''))
 ON CONFLICT DO NOTHING RETURNING id INTO sid;
 IF sid IS NULL THEN SELECT id INTO sid FROM public.group_promotional_shares WHERE group_id=gid AND post_id IS NOT DISTINCT FROM pid AND user_id=auth.uid(); END IF;
 RETURN jsonb_build_object('id',sid);
END;
$$;

-- parapost_groups_private.promotion_stats
-- Source: /Users/parapost/Documents/parapost-share-count-staging/docs/sql-drafts/groups-promotional-engagement-staging.sql
CREATE OR REPLACE FUNCTION parapost_groups_private.promotion_stats(
  p_share_id uuid
)
RETURNS jsonb
LANGUAGE plpgsql
STABLE
SECURITY DEFINER
SET search_path = ''
AS $function$
DECLARE
  v_uid uuid := auth.uid();
  v_group_id uuid;
  v_sharer_id uuid;
  v_comments jsonb;
BEGIN
  IF v_uid IS NULL THEN
    RAISE EXCEPTION 'Sign in required'
      USING ERRCODE = '42501';
  END IF;

  SELECT group_id, user_id INTO v_group_id, v_sharer_id
  FROM public.group_promotional_shares WHERE id = p_share_id;
  IF NOT FOUND OR parapost_groups_private.promotion_preview(v_group_id, p_share_id) IS NULL THEN
    RAISE EXCEPTION 'Share unavailable' USING ERRCODE = '42501';
  END IF;

  SELECT COALESCE(
    jsonb_agg(
      jsonb_build_object(
        'id', c.id,
        'author_id', c.author_id,
        'parent_id', c.parent_id,
        'content', c.content,
        'created_at', c.created_at,
        'updated_at', c.updated_at,
        'like_count', c.like_count,
        'liked', c.liked
      )
      ORDER BY c.created_at, c.id
    ),
    '[]'::jsonb
  )
  INTO v_comments
  FROM (
    SELECT
      cm.*,
      (
        SELECT count(*)::integer
        FROM public.group_promotion_share_comment_likes cl
        WHERE cl.comment_id = cm.id
          AND NOT parapost_groups_private.blocked(v_uid, cl.user_id)
      ) AS like_count,
      EXISTS (
        SELECT 1
        FROM public.group_promotion_share_comment_likes cl
        WHERE cl.comment_id = cm.id
          AND cl.user_id = v_uid
      ) AS liked
    FROM public.group_promotion_share_comments cm
    WHERE cm.share_id = p_share_id
      AND NOT parapost_groups_private.blocked(v_uid, cm.author_id)
      AND (
        cm.parent_id IS NULL
        OR EXISTS (
          SELECT 1
          FROM public.group_promotion_share_comments parent
          WHERE parent.id = cm.parent_id
            AND parent.share_id = cm.share_id
            AND parent.parent_id IS NULL
            AND NOT parapost_groups_private.blocked(
              v_uid, parent.author_id
            )
        )
      )
    ORDER BY cm.created_at, cm.id
    LIMIT 150
  ) c;

  RETURN jsonb_build_object(
    'liked', EXISTS (
      SELECT 1
      FROM public.group_promotion_share_likes l
      WHERE l.share_id = p_share_id
        AND l.user_id = v_uid
    ),
    'like_count', (
      SELECT count(*)::integer
      FROM public.group_promotion_share_likes l
      WHERE l.share_id = p_share_id
        AND NOT parapost_groups_private.blocked(v_uid, l.user_id)
    ),
    'comment_count', (
      SELECT count(*)::integer
      FROM public.group_promotion_share_comments c
      WHERE c.share_id = p_share_id
        AND NOT parapost_groups_private.blocked(v_uid, c.author_id)
        AND (
          c.parent_id IS NULL
          OR EXISTS (
            SELECT 1
            FROM public.group_promotion_share_comments parent
            WHERE parent.id = c.parent_id
              AND parent.share_id = c.share_id
              AND parent.parent_id IS NULL
              AND NOT parapost_groups_private.blocked(
                v_uid, parent.author_id
              )
          )
        )
    ),
    'comments_limited', (
      SELECT count(*) > 150
      FROM public.group_promotion_share_comments c
      WHERE c.share_id = p_share_id
        AND NOT parapost_groups_private.blocked(v_uid, c.author_id)
        AND (
          c.parent_id IS NULL
          OR EXISTS (
            SELECT 1
            FROM public.group_promotion_share_comments parent
            WHERE parent.id = c.parent_id
              AND parent.share_id = c.share_id
              AND parent.parent_id IS NULL
              AND NOT parapost_groups_private.blocked(
                v_uid, parent.author_id
              )
          )
        )
    ),
    'comments', v_comments
  );
END;
$function$;

-- parapost_groups_private.raise_share_cooldown
-- Source: /Users/parapost/Documents/parapost-share-count-staging/docs/sql-drafts/groups-sharing-cooldown-response-staging.sql
CREATE FUNCTION parapost_groups_private.raise_share_cooldown(retry_at timestamptz, target_kind text)
RETURNS void LANGUAGE plpgsql SECURITY INVOKER SET search_path = '' AS $$
DECLARE remaining_seconds integer;
BEGIN
  IF auth.uid() IS NULL OR NOT parapost_groups_private.enabled() THEN
    RAISE EXCEPTION 'Sharing unavailable' USING ERRCODE = '42501';
  END IF;
  IF target_kind NOT IN ('group','post') OR target_kind IS NULL OR retry_at IS NULL OR NOT isfinite(retry_at) THEN
    RAISE EXCEPTION 'Invalid cooldown response';
  END IF;
  remaining_seconds := ceil(extract(epoch FROM (retry_at - clock_timestamp())))::integer;
  -- Never fabricate a duration when the caller has no active cooldown.
  IF remaining_seconds <= 0 OR remaining_seconds > CASE WHEN target_kind='group' THEN 259200 ELSE 604800 END THEN
    RAISE EXCEPTION 'Invalid cooldown response';
  END IF;
  RAISE EXCEPTION 'GROUP_SHARE_COOLDOWN' USING ERRCODE = 'P0001',
    DETAIL = jsonb_build_object('kind',target_kind,'retry_after_seconds',remaining_seconds)::text;
END;
$$;

-- parapost_groups_private.record_share_cooldown
-- Source: /Users/parapost/Documents/parapost-staging-review-20261027/docs/sql-drafts/groups-cooldown-error-contract-review.sql
CREATE OR REPLACE FUNCTION parapost_groups_private.record_share_cooldown(
  p_target_type text,
  p_target_id uuid
)
RETURNS void
LANGUAGE plpgsql
SET search_path TO ''
AS $function$
DECLARE
  v_previous timestamptz;
  v_gap interval;
  v_remaining_seconds integer;
BEGIN
  IF auth.uid() IS NULL
     OR p_target_id IS NULL
     OR p_target_type IS NULL
     OR p_target_type NOT IN ('group', 'post') THEN
    RAISE EXCEPTION 'Sharing unavailable'
      USING ERRCODE = '42501';
  END IF;

  v_gap := CASE
    WHEN p_target_type = 'group'
      THEN interval '72 hours'
    ELSE interval '7 days'
  END;

  SELECT last_shared_at INTO v_previous
  FROM parapost_groups_private.group_share_cooldowns
  WHERE user_id = auth.uid()
    AND target_type = p_target_type
    AND target_id = p_target_id
  FOR UPDATE;

  IF v_previous IS NOT NULL
     AND now() < v_previous + v_gap THEN

    v_remaining_seconds := CEIL(
      EXTRACT(EPOCH FROM (v_previous + v_gap - now()))
    )::integer;

    RAISE EXCEPTION 'GROUP_SHARE_COOLDOWN'
      USING
        ERRCODE = 'P0001',
        DETAIL = jsonb_build_object(
          'kind', p_target_type,
          'retry_after_seconds', v_remaining_seconds
        )::text;
  END IF;

  INSERT INTO parapost_groups_private.group_share_cooldowns
    (user_id, target_type, target_id, last_shared_at)
  VALUES (auth.uid(), p_target_type, p_target_id, now())
  ON CONFLICT (user_id, target_type, target_id)
  DO UPDATE SET last_shared_at = EXCLUDED.last_shared_at;
END;
$function$;

-- public.creatorverse_cancel_page_media
-- Source: /Users/parapost/Documents/parapost-share-count-staging/docs/sql-drafts/creatorverse-page-image-cancellation-STAGING.sql
CREATE FUNCTION public.creatorverse_cancel_page_media(
  p_page_id uuid,
  p_media_id uuid,
  p_owner_id uuid
)
RETURNS boolean
LANGUAGE plpgsql
SECURITY INVOKER
SET search_path TO ''
AS $function$
DECLARE
  v_media record;
BEGIN
  IF current_user <> 'service_role'
     OR p_page_id IS NULL
     OR p_media_id IS NULL
     OR p_owner_id IS NULL
  THEN
    RAISE EXCEPTION 'Not authorized'
      USING ERRCODE = '42501';
  END IF;

  -- Keep lock order consistent with page ownership.
  PERFORM 1
  FROM public.creatorverse_pages
  WHERE id = p_page_id
    AND owner_id = p_owner_id
    AND status IN ('draft', 'active')
  FOR UPDATE;

  IF NOT FOUND THEN
    RETURN false;
  END IF;

  SELECT id, purpose, verification
  INTO v_media
  FROM public.creatorverse_media
  WHERE id = p_media_id
    AND page_id = p_page_id
    AND purpose IN ('logo', 'cover')
    AND post_id IS NULL
    AND media_type = 'image'
  FOR UPDATE;

  IF NOT FOUND OR v_media.verification <> 'pending' THEN
    RETURN false;
  END IF;

  -- Prevent any further upload through the reservation.
  UPDATE creatorverse_private.upload_reservations
  SET expires_at = pg_catalog.statement_timestamp()
  WHERE page_id = p_page_id
    AND media_id = p_media_id
    AND owner_id = p_owner_id
    AND consumed_at IS NULL;

  UPDATE public.creatorverse_media
  SET verification = 'rejected'
  WHERE id = p_media_id
    AND page_id = p_page_id
    AND verification = 'pending';

  IF NOT FOUND THEN
    RETURN false;
  END IF;

  -- Never remove storage.objects here.
  -- Cleanup must account for late uploads and retries.
  RETURN true;
END;
$function$;

-- public.creatorverse_finalize_page_media
-- Source: /Users/parapost/Documents/parapost-share-count-staging/docs/sql-drafts/creatorverse-page-image-finalization-STAGING.sql
CREATE FUNCTION public.creatorverse_finalize_page_media(
  p_media_id uuid,
  p_verified_bytes bigint,
  p_detected_mime_type text
)
RETURNS boolean
LANGUAGE plpgsql
SECURITY INVOKER
SET search_path TO ''
AS $function$
DECLARE
  v_media record;
  v_owner uuid;
  v_page_status text;
  v_reservation uuid;
  v_extension text;
BEGIN
  IF current_user <> 'service_role' THEN
    RAISE EXCEPTION 'Not authorized'
      USING ERRCODE = '42501';
  END IF;

  -- Read the page identifier without locking media.
  -- It is immutable because of the existing foreign key.
  SELECT page_id
  INTO v_media
  FROM public.creatorverse_media
  WHERE id = p_media_id
    AND purpose IN ('logo', 'cover')
    AND post_id IS NULL
    AND media_type = 'image';

  IF NOT FOUND THEN
    RAISE EXCEPTION 'Pending page image not found';
  END IF;

  -- Acquire locks consistently: page, media, reservation.
  SELECT owner_id, status
  INTO v_owner, v_page_status
  FROM public.creatorverse_pages
  WHERE id = v_media.page_id
  FOR UPDATE;

  IF NOT FOUND
     OR v_page_status NOT IN ('draft', 'active')
  THEN
    RAISE EXCEPTION 'Page unavailable';
  END IF;

  SELECT page_id, purpose, mime_type,
         storage_path, verification
  INTO v_media
  FROM public.creatorverse_media
  WHERE id = p_media_id
    AND page_id = v_media.page_id
    AND purpose IN ('logo', 'cover')
    AND post_id IS NULL
    AND media_type = 'image'
  FOR UPDATE;

  IF NOT FOUND OR v_media.verification <> 'pending' THEN
    RAISE EXCEPTION 'Pending page image not found';
  END IF;

  SELECT id
  INTO v_reservation
  FROM creatorverse_private.upload_reservations
  WHERE media_id = p_media_id
    AND page_id = v_media.page_id
    AND owner_id = v_owner
    AND consumed_at IS NULL
    AND expires_at > pg_catalog.statement_timestamp()
  FOR UPDATE;

  IF NOT FOUND THEN
    RAISE EXCEPTION 'Reservation expired or consumed';
  END IF;

  v_extension := CASE v_media.mime_type
    WHEN 'image/jpeg' THEN 'jpg'
    WHEN 'image/png' THEN 'png'
    WHEN 'image/webp' THEN 'webp'
    ELSE NULL
  END;

  IF v_extension IS NULL
     OR p_detected_mime_type IS DISTINCT FROM v_media.mime_type
     OR p_verified_bytes IS NULL
     OR p_verified_bytes NOT BETWEEN 1 AND 8388608
  THEN
    RAISE EXCEPTION 'Invalid image verification';
  END IF;

  IF v_media.storage_path IS DISTINCT FROM (
    v_media.page_id::text || '/' ||
    v_media.purpose || '/' ||
    p_media_id::text || '.' || v_extension
  ) THEN
    RAISE EXCEPTION 'Invalid image storage path';
  END IF;

  IF NOT EXISTS (
    SELECT 1
    FROM storage.objects
    WHERE bucket_id = 'creatorverse-media'
      AND name = v_media.storage_path
  ) THEN
    RAISE EXCEPTION 'Uploaded image missing';
  END IF;

  UPDATE creatorverse_private.upload_reservations
  SET consumed_at = pg_catalog.statement_timestamp()
  WHERE id = v_reservation
    AND consumed_at IS NULL;

  IF NOT FOUND THEN
    RAISE EXCEPTION 'Reservation consumption failed';
  END IF;

  UPDATE public.creatorverse_media
  SET verification = 'verified',
      byte_size = p_verified_bytes
  WHERE id = p_media_id
    AND verification = 'pending';

  IF NOT FOUND THEN
    RAISE EXCEPTION 'Image finalization failed';
  END IF;

  IF v_media.purpose = 'logo' THEN
    UPDATE public.creatorverse_pages
    SET logo_media_id = p_media_id
    WHERE id = v_media.page_id
      AND owner_id = v_owner
      AND status IN ('draft', 'active');
  ELSE
    UPDATE public.creatorverse_pages
    SET cover_media_id = p_media_id
    WHERE id = v_media.page_id
      AND owner_id = v_owner
      AND status IN ('draft', 'active');
  END IF;

  IF NOT FOUND THEN
    RAISE EXCEPTION 'Image attachment failed';
  END IF;

  RETURN true;
END;
$function$;

-- public.creatorverse_reserve_page_media
-- Source: /Users/parapost/Documents/parapost-share-count-staging/docs/sql-drafts/creatorverse-page-image-reservation-IMPLEMENTATION-STAGING.sql
CREATE OR REPLACE FUNCTION public.creatorverse_reserve_page_media(
  p_page_id uuid,
  p_purpose text,
  p_mime_type text,
  p_byte_size bigint
)
RETURNS jsonb
LANGUAGE plpgsql
SECURITY DEFINER
SET search_path TO ''
AS $function$
DECLARE
  v_user uuid := auth.uid();
  v_id uuid;
  v_extension text;
  v_path text;
  v_now timestamptz := pg_catalog.statement_timestamp();
BEGIN
  IF v_user IS NULL
     OR NOT creatorverse_private.write_enabled()
     OR NOT creatorverse_private.owns(p_page_id)
  THEN
    RAISE EXCEPTION 'Page image reservation unavailable'
      USING ERRCODE = '42501';
  END IF;

  -- Lock page to serialize owner reservations.
  PERFORM 1
  FROM public.creatorverse_pages
  WHERE id = p_page_id
    AND owner_id = v_user
    AND status IN ('draft', 'active')
  FOR UPDATE;

  IF NOT FOUND THEN
    RAISE EXCEPTION 'Page unavailable'
      USING ERRCODE = '42501';
  END IF;

  IF p_purpose NOT IN ('logo', 'cover')
     OR p_purpose IS NULL
  THEN
    RAISE EXCEPTION 'Invalid page image purpose'
      USING ERRCODE = '22023';
  END IF;

  v_extension := CASE p_mime_type
    WHEN 'image/jpeg' THEN 'jpg'
    WHEN 'image/png' THEN 'png'
    WHEN 'image/webp' THEN 'webp'
    ELSE NULL
  END;

  IF v_extension IS NULL
     OR p_byte_size IS NULL
     OR p_byte_size NOT BETWEEN 1 AND 8388608
  THEN
    RAISE EXCEPTION 'Invalid image format or size'
      USING ERRCODE = '22023';
  END IF;

  -- Limit pending unexpired reservations for each image purpose.
  IF (
    SELECT count(*)
    FROM public.creatorverse_media m
    JOIN creatorverse_private.upload_reservations r
      ON r.media_id = m.id
     AND r.page_id = m.page_id
    WHERE m.page_id = p_page_id
      AND m.purpose = p_purpose
      AND m.verification = 'pending'
      AND r.owner_id = v_user
      AND r.consumed_at IS NULL
      AND r.expires_at > v_now
  ) >= 3 THEN
    RAISE EXCEPTION 'Too many pending page image uploads'
      USING ERRCODE = '42901';
  END IF;

  -- Bound reservation attempts within one hour.
  IF (
    SELECT count(*)
    FROM public.creatorverse_media
    WHERE page_id = p_page_id
      AND purpose IN ('logo', 'cover')
      AND created_at > v_now - interval '1 hour'
  ) >= 12 THEN
    RAISE EXCEPTION 'Page image reservation limit reached'
      USING ERRCODE = '42901';
  END IF;

  v_id := pg_catalog.gen_random_uuid();

  v_path :=
    p_page_id::text || '/' ||
    p_purpose || '/' ||
    v_id::text || '.' || v_extension;

  INSERT INTO public.creatorverse_media (
    id,
    page_id,
    post_id,
    purpose,
    media_type,
    mime_type,
    storage_path,
    verification,
    display_order
  )
  VALUES (
    v_id,
    p_page_id,
    NULL,
    p_purpose,
    'image',
    p_mime_type,
    v_path,
    'pending',
    0
  );

  INSERT INTO creatorverse_private.upload_reservations (
    page_id,
    media_id,
    owner_id,
    expires_at
  )
  VALUES (
    p_page_id,
    v_id,
    v_user,
    v_now + interval '30 minutes'
  );

  RETURN pg_catalog.jsonb_build_object(
    'media_id', v_id,
    'storage_path', v_path,
    'verification', 'pending'
  );
END;
$function$;

-- =================================================
-- 4. NOTIFICATION VISIBILITY AND RLS GUARDS
-- =================================================

CREATE OR REPLACE FUNCTION
creatorverse_private.community_notification_visible(
  n public.notifications
)
RETURNS boolean
LANGUAGE plpgsql
STABLE
SECURITY DEFINER
SET search_path = ''
AS $function$
DECLARE
  v_owner uuid;
  v_visibility text;
BEGIN
  -- Never expose self-activity or incomplete events.
  IF n.user_id IS NULL
     OR n.actor_id IS NULL
     OR n.user_id = n.actor_id
     OR parapost_groups_private.blocked(
       n.user_id, n.actor_id
     )
  THEN
    RETURN false;
  END IF;

  IF n.type IN (
    'creatorverse_follow',
    'creatorverse_like',
    'creatorverse_comment',
    'creatorverse_reply',
    'creatorverse_share',
    'creatorverse_comment_like'
  ) THEN

    SELECT owner_id
    INTO v_owner
    FROM public.creatorverse_pages
    WHERE id = n.creatorverse_page_id
      AND status = 'active';

    IF v_owner IS NULL THEN
      RETURN false;
    END IF;

    IF n.community_post_id IS NOT NULL
       AND NOT EXISTS (
         SELECT 1
         FROM public.creatorverse_posts p
         WHERE p.id = n.community_post_id
           AND p.page_id = n.creatorverse_page_id
           AND p.status = 'published'
       )
    THEN
      RETURN false;
    END IF;

    IF n.community_comment_id IS NOT NULL
       AND NOT EXISTS (
         SELECT 1
         FROM public.creatorverse_comments c
         WHERE c.id = n.community_comment_id
           AND c.post_id = n.community_post_id
           AND NOT c.is_deleted
       )
    THEN
      RETURN false;
    END IF;

  ELSIF n.type IN (
    'group_post',
    'group_like',
    'group_comment',
    'group_reply',
    'group_share',
    'group_comment_like'
  ) THEN

    SELECT owner_id, visibility
    INTO v_owner, v_visibility
    FROM public.groups
    WHERE id = n.group_id;

    IF v_owner IS NULL THEN
      RETURN false;
    END IF;

    IF n.community_post_id IS NOT NULL
       AND NOT EXISTS (
         SELECT 1
         FROM public.group_posts p
         WHERE p.id = n.community_post_id
           AND p.group_id = n.group_id
           AND p.published
           AND NOT p.is_deleted
       )
    THEN
      RETURN false;
    END IF;

    IF n.community_comment_id IS NOT NULL
       AND NOT EXISTS (
         SELECT 1
         FROM public.group_post_comments c
         WHERE c.id = n.community_comment_id
           AND c.post_id = n.community_post_id
       )
    THEN
      RETURN false;
    END IF;

    IF v_visibility = 'private' THEN

      IF n.user_id <> v_owner
         AND NOT EXISTS (
           SELECT 1
           FROM public.group_members m
           WHERE m.group_id = n.group_id
             AND m.user_id = n.user_id
             AND m.status = 'approved'
         )
      THEN
        RETURN false;
      END IF;

      IF n.actor_id <> v_owner
         AND NOT EXISTS (
           SELECT 1
           FROM public.group_members m
           WHERE m.group_id = n.group_id
             AND m.user_id = n.actor_id
             AND m.status = 'approved'
         )
      THEN
        RETURN false;
      END IF;

    END IF;

    IF EXISTS (
      SELECT 1
      FROM public.group_members m
      WHERE m.group_id = n.group_id
        AND m.user_id IN (n.user_id, n.actor_id)
        AND m.status IN ('removed', 'banned')
    ) THEN
      RETURN false;
    END IF;

  ELSE
    RETURN false;
  END IF;

  -- Protect relationships involving page/group owners.
  IF parapost_groups_private.blocked(n.user_id, v_owner)
     OR parapost_groups_private.blocked(
       n.actor_id, v_owner
     )
  THEN
    RETURN false;
  END IF;

  RETURN true;
END;
$function$;

      'creatorverse_reply',
      'creatorverse_share',
      'creatorverse_comment_like',
      'group_post',
      'group_like',
      'group_comment',
      'group_reply',
      'group_share',
      'group_comment_like'
    ]
  )
  AND creatorverse_page_id IS NULL
  AND group_id IS NULL
  AND community_post_id IS NULL
  AND community_comment_id IS NULL
  AND community_event_key IS NULL

-- ==================================================
-- 1. COMMUNITY LIKES, COMMENTS, REPLIES AND SHARES
-- ==================================================

CREATE OR REPLACE FUNCTION
creatorverse_private.notify_community_activity_release()
RETURNS trigger
LANGUAGE plpgsql
SECURITY DEFINER
SET search_path = ''
AS $function$
DECLARE
  r jsonb;
  actor uuid;
  post uuid;
  comment uuid;
  parent_comment uuid;
  container uuid;
  owner_id uuid;
  author_id uuid;
  comment_author uuid;
  recipient uuid;
  visibility text;
  kind text;
  action text;
  event_key text;
BEGIN
  IF TG_OP = 'DELETE' THEN
    r := to_jsonb(OLD);
  ELSE
    r := to_jsonb(NEW);
  END IF;

  kind := CASE
    WHEN TG_TABLE_NAME LIKE 'creatorverse_%'
    THEN 'creatorverse'
    ELSE 'group'
  END;

  actor := coalesce(
    r->>'user_id',
    r->>'author_id',
    r->>'sharer_id'
  )::uuid;

  IF actor IS NULL THEN
    RETURN NULL;
  END IF;

  event_key := TG_TABLE_NAME || ':' ||
    coalesce(
      r->>'id',
      r->>'comment_id',
      r->>'post_id'
    ) || ':' || actor::text;

  -- Removing interactions removes related notifications.
  IF TG_OP = 'DELETE' THEN
    DELETE FROM public.notifications
    WHERE community_event_key = event_key;

    IF TG_TABLE_NAME IN (
      'creatorverse_comments',
      'group_post_comments'
    ) THEN
      DELETE FROM public.notifications
      WHERE community_comment_id = (r->>'id')::uuid;
    END IF;

    RETURN NULL;
  END IF;

  IF TG_TABLE_NAME = 'creatorverse_comments'
     AND r->>'is_deleted' = 'true'
  THEN
    DELETE FROM public.notifications
    WHERE community_event_key = event_key
       OR community_comment_id = (r->>'id')::uuid;
    RETURN NULL;
  END IF;

  -- Editing must not generate duplicate notifications.
  IF TG_OP <> 'INSERT' THEN
    RETURN NULL;
  END IF;

  IF auth.uid() IS DISTINCT FROM actor THEN
    RETURN NULL;
  END IF;

  post := nullif(r->>'post_id', '')::uuid;

  IF TG_TABLE_NAME IN (
    'creatorverse_comments',
    'group_post_comments'
  ) THEN
    comment := (r->>'id')::uuid;
    parent_comment := nullif(r->>'parent_id', '')::uuid;

    action := CASE
      WHEN parent_comment IS NULL THEN 'comment'
      ELSE 'reply'
    END;

  ELSIF TG_TABLE_NAME IN (
    'creatorverse_comment_likes',
    'group_comment_likes'
  ) THEN
    comment := (r->>'comment_id')::uuid;
    action := 'comment_like';

  ELSIF TG_TABLE_NAME LIKE '%shares' THEN
    action := 'share';

  ELSE
    action := 'like';
  END IF;

  IF TG_TABLE_NAME = 'creatorverse_comment_likes' THEN
    SELECT c.post_id, c.author_id
    INTO post, comment_author
    FROM public.creatorverse_comments c
    WHERE c.id = comment
      AND NOT c.is_deleted;

  ELSIF TG_TABLE_NAME = 'group_comment_likes' THEN
    SELECT c.post_id, c.author_id
    INTO post, comment_author
    FROM public.group_post_comments c
    WHERE c.id = comment;
  END IF;

  IF kind = 'creatorverse' THEN

    SELECT p.page_id, p.author_id, pg.owner_id
    INTO container, author_id, owner_id
    FROM public.creatorverse_posts p
    JOIN public.creatorverse_pages pg
      ON pg.id = p.page_id
    WHERE p.id = post
      AND p.status = 'published'
      AND pg.status = 'active';

    IF parent_comment IS NOT NULL THEN
      SELECT c.author_id INTO comment_author
      FROM public.creatorverse_comments c
      WHERE c.id = parent_comment
        AND c.post_id = post
        AND NOT c.is_deleted;
    END IF;

  ELSE

    SELECT p.group_id, p.author_id,
           g.owner_id, g.visibility
    INTO container, author_id, owner_id, visibility
    FROM public.group_posts p
    JOIN public.groups g
      ON g.id = p.group_id
    WHERE p.id = post
      AND p.published
      AND NOT p.is_deleted;

    IF parent_comment IS NOT NULL THEN
      SELECT c.author_id INTO comment_author
      FROM public.group_post_comments c
      WHERE c.id = parent_comment
        AND c.post_id = post;

    ELSIF action = 'comment_like' THEN
      SELECT c.author_id INTO comment_author
      FROM public.group_post_comments c
      WHERE c.id = comment
        AND c.post_id = post;
    END IF;

  END IF;

  IF container IS NULL OR owner_id IS NULL THEN
    RETURN NULL;
  END IF;

  -- Private Groups: actors must be approved members.
  IF kind = 'group'
     AND visibility = 'private'
     AND actor <> owner_id
     AND NOT EXISTS (
       SELECT 1
       FROM public.group_members m
       WHERE m.group_id = container
         AND m.user_id = actor
         AND m.status = 'approved'
     )
  THEN
    RETURN NULL;
  END IF;

  FOR recipient IN
    SELECT DISTINCT x.user_id
    FROM unnest(
      ARRAY[owner_id, author_id, comment_author]
    ) AS x(user_id)
    WHERE x.user_id IS NOT NULL
      AND x.user_id <> actor
  LOOP

    IF parapost_groups_private.blocked(
      actor, recipient
    ) THEN
      CONTINUE;
    END IF;

    IF kind = 'group'
       AND visibility = 'private'
       AND recipient <> owner_id
       AND NOT EXISTS (
         SELECT 1
         FROM public.group_members m
         WHERE m.group_id = container
           AND m.user_id = recipient
           AND m.status = 'approved'
       )
    THEN
      CONTINUE;
    END IF;

    INSERT INTO public.notifications (
      user_id,
      actor_id,
      type,
      creatorverse_page_id,
      group_id,
      community_post_id,
      community_comment_id,
      community_event_key,
      message,
      is_read
    )
    VALUES (
      recipient,
      actor,
      kind || '_' || action,
      CASE
        WHEN kind = 'creatorverse' THEN container
      END,
      CASE
        WHEN kind = 'group' THEN container
      END,
      post,
      comment,
      event_key,
      CASE
        WHEN kind = 'creatorverse'
        THEN 'New CreatorVerse activity.'
        ELSE 'New Group activity.'
      END,
      false
    )
    ON CONFLICT DO NOTHING;

  END LOOP;

  RETURN NULL;

EXCEPTION WHEN OTHERS THEN
  RAISE WARNING
    'Community activity notification failed: %',
    SQLERRM;
  RETURN NULL;
END;
$function$;

-- ==================================================
-- 2. NEW GROUP POSTS: NOTIFY GROUP OWNER
-- ==================================================

CREATE OR REPLACE FUNCTION
parapost_groups_private.notify_owner_new_post_release()
RETURNS trigger
LANGUAGE plpgsql
SECURITY DEFINER
SET search_path = ''
AS $function$
DECLARE
  recipient_id uuid;
BEGIN
  IF TG_OP = 'DELETE' THEN
    DELETE FROM public.notifications
    WHERE group_id = OLD.group_id
      AND community_post_id = OLD.id
      AND (
        type = 'group_post'
        OR community_event_key IS NOT NULL
      );
    RETURN OLD;
  END IF;

  IF NEW.published IS NOT TRUE
     OR NEW.is_deleted IS TRUE
  THEN
    DELETE FROM public.notifications
    WHERE group_id = NEW.group_id
      AND community_post_id = NEW.id
      AND (
        type = 'group_post'
        OR community_event_key IS NOT NULL
      );
    RETURN NEW;
  END IF;

  IF TG_OP = 'UPDATE' THEN
    IF OLD.published IS TRUE
       AND OLD.is_deleted IS NOT TRUE
    THEN
      RETURN NEW;
    END IF;
  END IF;

  SELECT g.owner_id INTO recipient_id
  FROM public.groups g
  WHERE g.id = NEW.group_id;

  IF recipient_id IS NULL
     OR NEW.author_id IS NULL
     OR recipient_id = NEW.author_id
  THEN
    RETURN NEW;
  END IF;

  IF parapost_groups_private.blocked(
    recipient_id, NEW.author_id
  ) THEN
    RETURN NEW;
  END IF;

  BEGIN
    INSERT INTO public.notifications (
      user_id,
      actor_id,
      type,
      group_id,
      community_post_id,
      message,
      is_read
    )
    VALUES (
      recipient_id,
      NEW.author_id,
      'group_post',
      NEW.group_id,
      NEW.id,
      'A member published a new Group post.',
      false
    )
    ON CONFLICT DO NOTHING;

  EXCEPTION WHEN OTHERS THEN
    RAISE WARNING
      'Group owner post notification failed: %',
      SQLERRM;
  END;

  RETURN NEW;

EXCEPTION WHEN OTHERS THEN
  RAISE WARNING
    'Group owner notification failed: %',
    SQLERRM;

  IF TG_OP = 'DELETE' THEN
    RETURN OLD;
  END IF;

  RETURN NEW;
END;
$function$;

CREATE OR REPLACE FUNCTION
creatorverse_private.notify_owner_follow_release()
RETURNS trigger
LANGUAGE plpgsql
SECURITY DEFINER
SET search_path = ''
AS $function$
DECLARE
  owner_id uuid;
BEGIN
  IF TG_OP = 'DELETE' THEN
    DELETE FROM public.notifications
    WHERE type = 'creatorverse_follow'
      AND creatorverse_page_id = OLD.page_id
      AND actor_id = OLD.user_id;
    RETURN OLD;
  END IF;

  SELECT p.owner_id INTO owner_id
  FROM public.creatorverse_pages p
  WHERE p.id = NEW.page_id
    AND p.status = 'active';

  IF owner_id IS NULL
     OR owner_id = NEW.user_id
     OR parapost_groups_private.blocked(
       owner_id, NEW.user_id
     )
  THEN
    RETURN NEW;
  END IF;

  BEGIN
    INSERT INTO public.notifications (
      user_id,
      actor_id,
      type,
      creatorverse_page_id,
      message,
      is_read
    )
    VALUES (
      owner_id,
      NEW.user_id,
      'creatorverse_follow',
      NEW.page_id,
      'Someone followed your CreatorVerse page.',
      false
    )
    ON CONFLICT DO NOTHING;

  EXCEPTION WHEN OTHERS THEN
    RAISE WARNING
      'CreatorVerse follow notification failed: %',
      SQLERRM;
  END;

  RETURN NEW;

EXCEPTION WHEN OTHERS THEN
  RAISE WARNING
    'CreatorVerse follow handler failed: %',
    SQLERRM;

  IF TG_OP = 'DELETE' THEN
    RETURN OLD;
  END IF;

  RETURN NEW;
END;
$function$;
