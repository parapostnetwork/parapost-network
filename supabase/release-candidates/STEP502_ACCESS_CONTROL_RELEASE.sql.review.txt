CREATE TABLE public.creatorverse_pages (
  id uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  owner_id uuid NOT NULL REFERENCES auth.users(id),
  name text NOT NULL CHECK (char_length(btrim(name)) BETWEEN 3 AND 100),
  category text NOT NULL CHECK (
    category IN (
      'business','creator','investigation_team','podcast',
      'entertainment','organization','community_project','other'
    )
  ),
  short_bio text NOT NULL DEFAULT ''
    CHECK (char_length(short_bio) <= 240),
  description text NOT NULL DEFAULT ''
    CHECK (char_length(description) <= 5000),
  website text CHECK (
    website IS NULL OR (
      char_length(website) <= 2048 AND
      website ~ '^https?://[^[:space:]@/]+([/?#][^[:space:]]*)?$'
    )
  ),
  social_links jsonb NOT NULL DEFAULT '[]'::jsonb
    CHECK (
      jsonb_typeof(social_links) = 'array'
      AND jsonb_array_length(social_links) <= 8
    ),
  public_email text CHECK (
    public_email IS NULL OR (
      char_length(public_email) <= 254 AND
      public_email ~ '^[^[:space:]@]+@[^[:space:]@]+\.[^[:space:]@]+$'
    )
  ),
  location text NOT NULL DEFAULT ''
    CHECK (char_length(location) <= 160),
  logo_media_id uuid,
  cover_media_id uuid,
  status text NOT NULL DEFAULT 'draft'
    CHECK (status IN ('draft','active','suspended','deleted')),
  created_at timestamptz NOT NULL DEFAULT now(),
  updated_at timestamptz NOT NULL DEFAULT now(),
  search_document tsvector GENERATED ALWAYS AS (
    to_tsvector(
      'simple'::regconfig,
      name || ' ' || short_bio || ' ' || description
    )
  ) STORED,
  UNIQUE (id, owner_id)
);

CREATE TABLE public.creatorverse_posts (
  id uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  page_id uuid NOT NULL REFERENCES public.creatorverse_pages(id),
  author_id uuid NOT NULL REFERENCES auth.users(id),
  content text NOT NULL DEFAULT ''
    CHECK (char_length(content) <= 10000),
  status text NOT NULL DEFAULT 'draft'
    CHECK (status IN ('draft','published','deleted')),
  comments_enabled boolean NOT NULL DEFAULT true,
  created_at timestamptz NOT NULL DEFAULT now(),
  updated_at timestamptz NOT NULL DEFAULT now(),
  UNIQUE (page_id, id),
  FOREIGN KEY (page_id, author_id)
    REFERENCES public.creatorverse_pages(id, owner_id)
);

CREATE TABLE public.creatorverse_media (
  id uuid PRIMARY KEY DEFAULT gen_random_uuid(),
  page_id uuid NOT NULL REFERENCES public.creatorverse_pages(id),
  post_id uuid,
  purpose text NOT NULL
    CHECK (purpose IN ('logo','cover','post')),
  media_type text NOT NULL
    CHECK (media_type IN ('image','video')),
  storage_path text NOT NULL UNIQUE,
  mime_type text NOT NULL CHECK (
    mime_type IN (
      'image/jpeg','image/png','image/webp',
      'video/mp4','video/webm'
    )
  ),
  verification text NOT NULL DEFAULT 'pending'
    CHECK (verification IN ('pending','verified','rejected')),
  byte_size bigint CHECK (byte_size > 0),
  duration_seconds numeric CHECK (
    duration_seconds > 0 AND duration_seconds <= 120
  ),
  display_order smallint NOT NULL DEFAULT 0
    CHECK (display_order BETWEEN 0 AND 3),
  created_at timestamptz NOT NULL DEFAULT now(),
  UNIQUE (page_id, id),
  FOREIGN KEY (page_id, post_id)
    REFERENCES public.creatorverse_posts(page_id, id),
  CHECK (storage_path LIKE page_id::text || '/%'),
  CHECK (
    char_length(storage_path) <= 240
    AND storage_path NOT LIKE '%..%'
    AND storage_path NOT LIKE '%//%'
  ),
  CHECK (
    (purpose = 'post' AND post_id IS NOT NULL)
    OR (
      purpose IN ('logo','cover')
      AND post_id IS NULL
      AND media_type = 'image'
    )
  ),
  CHECK (
    (media_type = 'image'
      AND mime_type IN ('image/jpeg','image/png','image/webp')
      AND duration_seconds IS NULL
      AND (byte_size IS NULL OR byte_size <= 8388608))
    OR
    (media_type = 'video'
      AND mime_type IN ('video/mp4','video/webm')
      AND (byte_size IS NULL OR byte_size <= 33554432))
  ),
  CHECK (
    verification <> 'verified'
    OR (
      byte_size IS NOT NULL
      AND (media_type = 'image' OR duration_seconds IS NOT NULL)
    )
  )
);

ALTER TABLE public.creatorverse_pages

CREATE OR REPLACE FUNCTION creatorverse_private.activate_page_draft(p_page_id uuid)
 RETURNS uuid
 LANGUAGE plpgsql
 SECURITY DEFINER
 SET search_path TO ''
AS $function$
DECLARE
  v_page_id uuid;
BEGIN
  IF auth.uid() IS NULL
     OR NOT creatorverse_private.write_enabled()
  THEN
    RAISE EXCEPTION 'Page activation unavailable'
      USING ERRCODE = '42501';
  END IF;

  UPDATE public.creatorverse_pages AS p
  SET status = 'active',
      updated_at = now()
  WHERE p.id = p_page_id
    AND p.owner_id = auth.uid()
    AND p.status = 'draft'

    AND (
      p.logo_media_id IS NULL
      OR EXISTS (
        SELECT 1
        FROM public.creatorverse_media AS m
        WHERE m.id = p.logo_media_id
          AND m.page_id = p.id
          AND m.purpose = 'logo'
          AND m.media_type = 'image'
          AND m.verification = 'verified'
          AND m.post_id IS NULL
      )
    )

    AND (
      p.cover_media_id IS NULL
      OR EXISTS (
        SELECT 1
        FROM public.creatorverse_media AS m
        WHERE m.id = p.cover_media_id
          AND m.page_id = p.id
          AND m.purpose = 'cover'
          AND m.media_type = 'image'
          AND m.verification = 'verified'
          AND m.post_id IS NULL
      )
    )

  RETURNING p.id INTO v_page_id;

  IF v_page_id IS NULL THEN
    RAISE EXCEPTION 'Page not eligible for activation'
      USING ERRCODE = '42501';
  END IF;

  RETURN v_page_id;
END;
$function$;

CREATE OR REPLACE FUNCTION creatorverse_private.blocked(other_id uuid)

    OR EXISTS (SELECT 1 FROM public.user_blocks WHERE (blocker_id=auth.uid() AND blocked_id=other_id) OR (blocker_id=other_id AND blocked_id=auth.uid()))
  );
$function$;

CREATE OR REPLACE FUNCTION creatorverse_private.community_notification_visible(n public.notifications)
 RETURNS boolean
 LANGUAGE plpgsql
 STABLE SECURITY DEFINER
 SET search_path TO ''
AS $function$
DECLARE
    v_owner uuid;
    v_visibility text;
BEGIN
    IF n.user_id IS NULL
       OR n.actor_id IS NULL
       OR n.user_id = n.actor_id
       OR parapost_groups_private.blocked(n.user_id, n.actor_id)
    THEN
        RETURN false;
    END IF;

    IF n.type IN ('creatorverse_follow', 'creatorverse_like', 'creatorverse_comment', 'creatorverse_reply', 'creatorverse_share', 'creatorverse_comment_like') THEN

        SELECT owner_id INTO v_owner
        FROM public.creatorverse_pages
        WHERE id = n.creatorverse_page_id
          AND status = 'active';

        IF v_owner IS NULL THEN
            RETURN false;
        END IF;

        IF n.community_post_id IS NOT NULL
           AND NOT EXISTS (
               SELECT 1
               FROM public.creatorverse_posts p
               WHERE p.id = n.community_post_id
                 AND p.page_id = n.creatorverse_page_id
                 AND p.status = 'published'
           )
        THEN
            RETURN false;
        END IF;

        IF n.community_comment_id IS NOT NULL
           AND NOT EXISTS (
               SELECT 1
               FROM public.creatorverse_comments c
               WHERE c.id = n.community_comment_id
                 AND c.post_id = n.community_post_id
                 AND NOT c.is_deleted
           )
        THEN
            RETURN false;
        END IF;

    ELSIF n.type IN ('group_post', 'group_like', 'group_comment', 'group_reply', 'group_share', 'group_comment_like') THEN

        SELECT owner_id, visibility
        INTO v_owner, v_visibility
        FROM public.groups
        WHERE id = n.group_id;

        IF v_owner IS NULL THEN
            RETURN false;
        END IF;

        IF n.community_post_id IS NOT NULL
           AND NOT EXISTS (
               SELECT 1
               FROM public.group_posts p
               WHERE p.id = n.community_post_id
                 AND p.group_id = n.group_id
                 AND p.published
                 AND NOT p.is_deleted
           )
        THEN
            RETURN false;
        END IF;

        IF n.community_comment_id IS NOT NULL
           AND NOT EXISTS (
               SELECT 1
               FROM public.group_post_comments c
               WHERE c.id = n.community_comment_id
                 AND c.post_id = n.community_post_id
           )
        THEN
            RETURN false;
        END IF;

        IF v_visibility = 'private' THEN

            IF n.user_id <> v_owner
               AND NOT EXISTS (
                   SELECT 1 FROM public.group_members m
                   WHERE m.group_id = n.group_id
                     AND m.user_id = n.user_id
                     AND m.status = 'approved'
               )
            THEN
                RETURN false;
            END IF;

            IF n.actor_id <> v_owner
               AND NOT EXISTS (
                   SELECT 1 FROM public.group_members m
                   WHERE m.group_id = n.group_id
                     AND m.user_id = n.actor_id
                     AND m.status = 'approved'
               )
            THEN
                RETURN false;
            END IF;

        END IF;

        IF EXISTS (
            SELECT 1 FROM public.group_members m
            WHERE m.group_id = n.group_id
              AND m.user_id IN (n.user_id, n.actor_id)
              AND m.status IN ('removed', 'banned')
        )
        THEN
            RETURN false;
        END IF;

    ELSE
        RETURN true;
    END IF;

    IF parapost_groups_private.blocked(n.user_id, v_owner)
       OR parapost_groups_private.blocked(n.actor_id, v_owner)
    THEN
        RETURN false;
    END IF;

    RETURN true;
END;
$function$;

CREATE OR REPLACE FUNCTION creatorverse_private.create_page_draft(p_name text, p_category text, p_short_bio text DEFAULT ''::text, p_description text DEFAULT ''::text)
 RETURNS uuid
 LANGUAGE plpgsql
 SECURITY DEFINER
 SET search_path TO ''
AS $function$
DECLARE
  v_id uuid;
  v_name text := btrim(p_name);
BEGIN
  IF auth.uid() IS NULL
     OR NOT creatorverse_private.write_enabled()
  THEN
    RAISE EXCEPTION 'Page creation unavailable'
      USING ERRCODE = '42501';
  END IF;

  IF v_name IS NULL
     OR char_length(v_name) NOT BETWEEN 3 AND 100
     OR p_category IS NULL
     OR p_category NOT IN (
       'business', 'creator', 'investigation_team',
       'podcast', 'entertainment', 'organization',
       'community_project', 'other'
     )
     OR char_length(coalesce(p_short_bio, '')) > 240
     OR char_length(coalesce(p_description, '')) > 5000
  THEN
    RAISE EXCEPTION 'Invalid page details'
      USING ERRCODE = '22023';
  END IF;

  INSERT INTO public.creatorverse_pages (
    owner_id, name, category, short_bio,
    description, status
  )
  VALUES (
    auth.uid(), v_name, p_category,
    coalesce(p_short_bio, ''),
    coalesce(p_description, ''),
    'draft'
  )
  RETURNING id INTO v_id;

  RETURN v_id;
END;
$function$;

CREATE OR REPLACE FUNCTION creatorverse_private.create_text_draft(p_page_id uuid, p_content text)
 RETURNS uuid
 LANGUAGE plpgsql
 SECURITY DEFINER
 SET search_path TO ''
AS $function$
DECLARE
  v_id uuid;
BEGIN
  IF auth.uid() IS NULL
     OR NOT creatorverse_private.write_enabled()
     OR NOT creatorverse_private.owns(p_page_id)
  THEN
    RAISE EXCEPTION 'Posting unavailable'
      USING ERRCODE = '42501';
  END IF;

  IF p_content IS NULL
     OR char_length(btrim(p_content)) > 5000
  THEN
    RAISE EXCEPTION 'Invalid post content'
      USING ERRCODE = '22023';
  END IF;

  PERFORM 1
  FROM public.creatorverse_pages
  WHERE id = p_page_id
    AND owner_id = auth.uid()
    AND status IN ('draft', 'active')
  FOR UPDATE;

  IF NOT FOUND THEN
    RAISE EXCEPTION 'Page unavailable'
      USING ERRCODE = '42501';
  END IF;

  INSERT INTO public.creatorverse_posts(
    page_id, author_id, content, status, comments_enabled
  )
  VALUES (
    p_page_id, auth.uid(), btrim(p_content), 'draft', true
  )
  RETURNING id INTO v_id;

  RETURN v_id;
END;
$function$;

CREATE OR REPLACE FUNCTION creatorverse_private.enforce_post_share_cooldown()
 RETURNS trigger
 LANGUAGE plpgsql
 SECURITY DEFINER
 SET search_path TO ''
AS $function$
DECLARE
  v_page_id uuid;
  v_owner_id uuid;
  v_existing_post uuid;
  v_accepted integer;
BEGIN
  IF auth.uid() IS NULL
     OR NEW.sharer_id IS DISTINCT FROM auth.uid()
     OR NEW.post_id IS NULL
     OR NEW.request_id IS NULL THEN
    RAISE EXCEPTION 'Sharing unauthorized'
      USING ERRCODE = '42501';
  END IF;

  IF NOT creatorverse_private.write_enabled() THEN
    RAISE EXCEPTION 'Sharing disabled'
      USING ERRCODE = '42501';
  END IF;

  SELECT p.page_id, page.owner_id
  INTO v_page_id, v_owner_id
  FROM public.creatorverse_posts p
  JOIN public.creatorverse_pages page
    ON page.id = p.page_id
  WHERE p.id = NEW.post_id
    AND p.status = 'published'
    AND page.status = 'active';

  IF NOT FOUND
     OR NOT creatorverse_private.read_post(NEW.post_id)
     OR creatorverse_private.blocked(v_owner_id) THEN
    RAISE EXCEPTION 'Post unavailable'
      USING ERRCODE = '42501';
  END IF;

  IF NOT (
    creatorverse_private.owns(v_page_id)
    OR EXISTS (
      SELECT 1
      FROM public.creatorverse_follows
      WHERE page_id = v_page_id
        AND user_id = NEW.sharer_id
    )
  ) THEN
    RAISE EXCEPTION 'Follow this page to share'
      USING ERRCODE = '42501';
  END IF;

  -- Serialize attempts for the same member and post.
  PERFORM pg_catalog.pg_advisory_xact_lock(
    pg_catalog.hashtextextended(
      NEW.sharer_id::text || ':' || NEW.post_id::text,
      0
    )
  );

  -- A retry of the same request must not
  -- consume another seven-day cooldown.
  SELECT post_id INTO v_existing_post
  FROM public.creatorverse_shares
  WHERE sharer_id = NEW.sharer_id
    AND request_id = NEW.request_id;

  IF FOUND THEN
    IF v_existing_post <> NEW.post_id THEN
      RAISE EXCEPTION 'Request ID already used'
        USING ERRCODE = '22023';
    END IF;

    RETURN NULL;
  END IF;

  INSERT INTO creatorverse_private.post_share_cooldowns
    (sharer_id, post_id, last_shared_at)
  VALUES
    (NEW.sharer_id, NEW.post_id, pg_catalog.now())
  ON CONFLICT (sharer_id, post_id)
  DO UPDATE SET
    last_shared_at = EXCLUDED.last_shared_at
  WHERE
    post_share_cooldowns.last_shared_at
      <= EXCLUDED.last_shared_at - interval '7 days'
  RETURNING 1 INTO v_accepted;

  IF v_accepted IS DISTINCT FROM 1 THEN
    RAISE EXCEPTION 'CREATORVERSE_SHARE_COOLDOWN'
      USING ERRCODE = 'P0001';
  END IF;

  NEW.created_at := pg_catalog.now();
  RETURN NEW;
END;
$function$;

CREATE OR REPLACE FUNCTION creatorverse_private.enqueue_rejected_post_media()
 RETURNS trigger
 LANGUAGE plpgsql
 SECURITY DEFINER
 SET search_path TO ''
AS $function$
DECLARE
  extension text;
BEGIN
  IF NEW.verification IS DISTINCT FROM 'rejected'
     OR OLD.verification = 'rejected'
     OR NEW.purpose IS DISTINCT FROM 'post'
     OR NEW.post_id IS NULL
  THEN
    RETURN NEW;
  END IF;

  extension := CASE NEW.mime_type
    WHEN 'image/jpeg' THEN 'jpg'
    WHEN 'image/png' THEN 'png'
    WHEN 'image/webp' THEN 'webp'
    WHEN 'video/mp4' THEN 'mp4'
    WHEN 'video/webm' THEN 'webm'
    ELSE NULL
  END;

  IF extension IS NULL
     OR NEW.storage_path IS DISTINCT FROM (
       NEW.page_id::text || '/' ||
       NEW.post_id::text || '/' ||
       NEW.id::text || '.' || extension
     )
  THEN
    RAISE EXCEPTION 'Invalid media cleanup path';
  END IF;

  INSERT INTO creatorverse_private.media_cleanup_jobs (
    media_id,
    page_id,
    post_id,
    storage_path,
    eligible_after
  )
  VALUES (
    NEW.id,
    NEW.page_id,
    NEW.post_id,
    NEW.storage_path,
    pg_catalog.statement_timestamp()
      + interval '30 minutes'
  )
  ON CONFLICT (media_id) DO NOTHING;

  RETURN NEW;
END;
$function$;

CREATE OR REPLACE FUNCTION creatorverse_private.guard_published_media()
 RETURNS trigger
 LANGUAGE plpgsql
 SECURITY DEFINER
 SET search_path TO ''
AS $function$
DECLARE
  v_post_id uuid;
  v_page_id uuid;
  v_status text;
BEGIN
  IF TG_OP = 'DELETE' THEN
    v_post_id := OLD.post_id;
    v_page_id := OLD.page_id;
  ELSE
    v_post_id := NEW.post_id;
    v_page_id := NEW.page_id;

    IF TG_OP = 'UPDATE'
       AND (
         OLD.post_id IS DISTINCT FROM NEW.post_id
         OR OLD.page_id IS DISTINCT FROM NEW.page_id
       )
    THEN
      RAISE EXCEPTION 'Media association is immutable'
        USING ERRCODE = '23514';
    END IF;
  END IF;

  IF v_post_id IS NOT NULL THEN
    SELECT status
    INTO v_status
    FROM public.creatorverse_posts
    WHERE id = v_post_id
      AND page_id = v_page_id
    FOR SHARE;

    IF NOT FOUND THEN
      RAISE EXCEPTION 'Referenced post unavailable'
        USING ERRCODE = '23503';
    END IF;

    IF v_status = 'published' THEN
      RAISE EXCEPTION
        'Published post media cannot be modified'
        USING ERRCODE = '23514';
    END IF;
  END IF;

  IF TG_OP = 'DELETE' THEN
    RETURN OLD;
  END IF;

  RETURN NEW;
END;
$function$;

CREATE OR REPLACE FUNCTION creatorverse_private.immutable_identity()
 RETURNS trigger
 LANGUAGE plpgsql
 SET search_path TO ''
AS $function$
BEGIN
  IF NEW.id IS DISTINCT FROM OLD.id OR NEW.created_at IS DISTINCT FROM OLD.created_at THEN
    RAISE EXCEPTION 'Immutable identity';
  END IF;
  IF TG_TABLE_NAME='creatorverse_pages' THEN
    IF NEW.owner_id IS DISTINCT FROM OLD.owner_id THEN RAISE EXCEPTION 'Ownership transfers are not supported'; END IF;
  ELSIF TG_TABLE_NAME='creatorverse_posts' THEN
    IF NEW.page_id IS DISTINCT FROM OLD.page_id OR NEW.author_id IS DISTINCT FROM OLD.author_id THEN RAISE EXCEPTION 'Immutable post identity'; END IF;
  ELSIF TG_TABLE_NAME='creatorverse_comments' THEN
    IF NEW.post_id IS DISTINCT FROM OLD.post_id OR NEW.author_id IS DISTINCT FROM OLD.author_id OR NEW.parent_id IS DISTINCT FROM OLD.parent_id THEN RAISE EXCEPTION 'Immutable comment identity'; END IF;
  END IF;
  NEW.updated_at=now(); RETURN NEW;
END; $function$;

CREATE OR REPLACE FUNCTION creatorverse_private.notify_community_activity_release()
 RETURNS trigger
 LANGUAGE plpgsql
 SECURITY DEFINER
 SET search_path TO ''
AS $function$
DECLARE
  r jsonb;
  actor uuid;
  post uuid;
  comment uuid;
  parent_comment uuid;
  container uuid;
  owner uuid;
  author uuid;
  comment_author uuid;
  recipient uuid;
  visibility text;
  kind text;
  action text;
  event_key text;
BEGIN
  IF TG_OP = 'DELETE' THEN
    r := to_jsonb(OLD);
  ELSE
    r := to_jsonb(NEW);
  END IF;

  kind := CASE
    WHEN TG_TABLE_NAME LIKE 'creatorverse_%'
      THEN 'creatorverse'
    ELSE 'group'
  END;

  actor := COALESCE(
    r->>'user_id',
    r->>'author_id',
    r->>'sharer_id'
  )::uuid;

  IF actor IS NULL THEN
    RETURN NULL;
  END IF;

  event_key := TG_TABLE_NAME || ':' ||
    COALESCE(r->>'id', r->>'comment_id', r->>'post_id')
    || ':' || actor::text;

  IF TG_OP = 'DELETE' THEN
    DELETE FROM public.notifications
    WHERE community_event_key = event_key;

    IF TG_TABLE_NAME IN (
      'creatorverse_comments', 'group_post_comments'
    ) THEN
      DELETE FROM public.notifications
      WHERE community_comment_id = (r->>'id')::uuid;
    END IF;

    RETURN NULL;
  END IF;

  IF TG_TABLE_NAME = 'creatorverse_comments'
     AND r->>'is_deleted' = 'true' THEN
    DELETE FROM public.notifications
    WHERE community_event_key = event_key
       OR community_comment_id = (r->>'id')::uuid;
    RETURN NULL;
  END IF;

  -- Edits must not create new notifications.
  IF TG_OP <> 'INSERT' THEN
    RETURN NULL;
  END IF;

  -- The database-authenticated actor must match the event.
  IF auth.uid() IS DISTINCT FROM actor THEN
    RETURN NULL;
  END IF;

  post := NULLIF(r->>'post_id', '')::uuid;

  IF TG_TABLE_NAME IN (
    'creatorverse_comments', 'group_post_comments'
  ) THEN
    comment := (r->>'id')::uuid;
    parent_comment := NULLIF(r->>'parent_id', '')::uuid;
    action := CASE
      WHEN parent_comment IS NULL THEN 'comment'
      ELSE 'reply'
    END;

  ELSIF TG_TABLE_NAME IN (
    'creatorverse_comment_likes', 'group_comment_likes'
  ) THEN
    comment := (r->>'comment_id')::uuid;
    action := 'comment_like';

  ELSIF TG_TABLE_NAME LIKE '%shares' THEN
    action := 'share';

  ELSE
    action := 'like';
  END IF;

  IF TG_TABLE_NAME = 'creatorverse_comment_likes' THEN
    SELECT c.post_id, c.author_id
    INTO post, comment_author
    FROM public.creatorverse_comments c
    WHERE c.id = comment
      AND NOT c.is_deleted;
  ELSIF TG_TABLE_NAME = 'group_comment_likes' THEN
    SELECT c.post_id, c.author_id
    INTO post, comment_author
    FROM public.group_post_comments c
    WHERE c.id = comment;
  END IF;

  IF kind = 'creatorverse' THEN
    SELECT p.page_id, p.author_id, pg.owner_id
    INTO container, author, owner
    FROM public.creatorverse_posts p
    JOIN public.creatorverse_pages pg ON pg.id = p.page_id
    WHERE p.id = post
      AND p.status = 'published'
      AND pg.status = 'active';

    IF parent_comment IS NOT NULL THEN
      SELECT author_id INTO comment_author
      FROM public.creatorverse_comments
      WHERE id = parent_comment
        AND post_id = post
        AND NOT is_deleted;
    END IF;

  ELSE
    SELECT p.group_id, p.author_id, g.owner_id,
           g.visibility
    INTO container, author, owner, visibility
    FROM public.group_posts p
    JOIN public.groups g ON g.id = p.group_id
    WHERE p.id = post
      AND p.published
      AND NOT p.is_deleted;

    IF parent_comment IS NOT NULL THEN
      SELECT author_id INTO comment_author
      FROM public.group_post_comments
      WHERE id = parent_comment AND post_id = post;

    ELSIF action = 'comment_like' THEN
      SELECT author_id INTO comment_author
      FROM public.group_post_comments
      WHERE id = comment AND post_id = post;
    END IF;
  END IF;

  IF container IS NULL OR owner IS NULL THEN
    RETURN NULL;
  END IF;

  -- Private Group members must be approved at the time of the interaction.
  IF kind = 'group' AND visibility = 'private' AND actor <> owner
     AND NOT EXISTS (
       SELECT 1 FROM public.group_members m
       WHERE m.group_id = container
         AND m.user_id = actor
         AND m.status = 'approved'
     ) THEN
    RETURN NULL;
  END IF;

  FOR recipient IN
    SELECT DISTINCT x.user_id
    FROM unnest(
      ARRAY[owner, author, comment_author]
    ) AS x(user_id)
    WHERE x.user_id IS NOT NULL
      AND x.user_id <> actor
  LOOP

    IF parapost_groups_private.blocked(actor, recipient)
       THEN CONTINUE;
    END IF;

    IF kind = 'group' AND visibility = 'private'
       AND recipient <> owner
       AND NOT EXISTS (
         SELECT 1 FROM public.group_members m
         WHERE m.group_id = container
           AND m.user_id = recipient
           AND m.status = 'approved'
       ) THEN
      CONTINUE;
    END IF;

    INSERT INTO public.notifications (
      user_id, actor_id, type,
      creatorverse_page_id, group_id,
      community_post_id, community_comment_id,
      community_event_key, message, is_read
    )
    VALUES (
      recipient, actor, kind || '_' || action,
      CASE WHEN kind = 'creatorverse' THEN container END,
      CASE WHEN kind = 'group' THEN container END,
      post, comment, event_key,
      CASE
        WHEN kind = 'creatorverse'
          THEN 'New CreatorVerse activity.'
        ELSE 'New Group activity.'
      END,
      false
    )
    ON CONFLICT DO NOTHING;

  END LOOP;

  RETURN NULL;
EXCEPTION WHEN OTHERS THEN
  -- Notification errors must not cancel an actual like/comment/share.
  RAISE WARNING 'Community activity notification failed: %', SQLERRM;
  RETURN NULL;
END;
$function$;

CREATE OR REPLACE FUNCTION creatorverse_private.notify_owner_follow_release()
 RETURNS trigger
 LANGUAGE plpgsql
 SECURITY DEFINER
 SET search_path TO ''
AS $function$
DECLARE
    v_owner_id uuid;
BEGIN
    IF TG_OP = 'DELETE' THEN
        DELETE FROM public.notifications
        WHERE type = 'creatorverse_follow'
          AND creatorverse_page_id = OLD.page_id
          AND actor_id = OLD.user_id;
        RETURN OLD;
    END IF;

    SELECT owner_id
    INTO v_owner_id
    FROM public.creatorverse_pages
    WHERE id = NEW.page_id
      AND status = 'active';

    IF v_owner_id IS NULL
       OR v_owner_id = NEW.user_id
    THEN
        RETURN NEW;
    END IF;

    IF EXISTS (
        SELECT 1
        FROM public.blocked_users b
        WHERE (
            b.blocker_id = v_owner_id
            AND b.blocked_id = NEW.user_id
        ) OR (
            b.blocker_id = NEW.user_id
            AND b.blocked_id = v_owner_id
        )
    ) OR EXISTS (
        SELECT 1
        FROM public.user_blocks b
        WHERE (
            b.blocker_id = v_owner_id
            AND b.blocked_id = NEW.user_id
        ) OR (
            b.blocker_id = NEW.user_id
            AND b.blocked_id = v_owner_id
        )
    ) THEN
        RETURN NEW;
    END IF;

    BEGIN
        INSERT INTO public.notifications (
            user_id,
            actor_id,
            type,
            creatorverse_page_id,
            message,
            is_read
        )
        VALUES (
            v_owner_id,
            NEW.user_id,
            'creatorverse_follow',
            NEW.page_id,
            'Someone followed your CreatorVerse page.',
            false
        )
        ON CONFLICT DO NOTHING;
    EXCEPTION WHEN OTHERS THEN
        RAISE WARNING 'CreatorVerse follow notification insert failed: %', SQLERRM;
    END;

    RETURN NEW;
EXCEPTION WHEN OTHERS THEN
    RAISE WARNING 'CreatorVerse follow notification failed: %', SQLERRM;
    IF TG_OP = 'DELETE' THEN RETURN OLD; END IF;
    RETURN NEW;
END;
$function$;

CREATE OR REPLACE FUNCTION creatorverse_private.page_image_storage_allowed(p_path text, p_operation text)
 RETURNS boolean
 LANGUAGE plpgsql
 STABLE SECURITY DEFINER
 SET search_path TO ''
AS $function$
DECLARE
  v_user uuid := auth.uid();
BEGIN
  IF p_path IS NULL
     OR p_operation IS NULL
  THEN
    RETURN false;
  END IF;

  -- Uploads require an authenticated page owner,
  -- an active reservation, and an unverified image.
  IF p_operation = 'upload' THEN

    IF v_user IS NULL
       OR NOT creatorverse_private.write_enabled()
    THEN
      RETURN false;
    END IF;

    RETURN EXISTS (
      SELECT 1
      FROM public.creatorverse_media m
      JOIN public.creatorverse_pages p
        ON p.id = m.page_id
      JOIN creatorverse_private.upload_reservations r
        ON r.page_id = m.page_id
       AND r.media_id = m.id
      WHERE m.storage_path = p_path
        AND m.purpose IN ('logo', 'cover')
        AND m.post_id IS NULL
        AND m.media_type = 'image'
        AND m.mime_type IN (
          'image/jpeg',
          'image/png',
          'image/webp'
        )
        AND m.verification = 'pending'
        AND m.storage_path = (
          m.page_id::text || '/' ||
          m.purpose || '/' ||
          m.id::text || '.' ||
          CASE m.mime_type
            WHEN 'image/jpeg' THEN 'jpg'
            WHEN 'image/png' THEN 'png'
            WHEN 'image/webp' THEN 'webp'
          END
        )
        AND p.owner_id = v_user
        AND p.status IN ('draft', 'active')
        AND creatorverse_private.owns(p.id)
        AND r.owner_id = v_user
        AND r.consumed_at IS NULL
        AND r.expires_at >
          pg_catalog.statement_timestamp()
    );

  ELSIF p_operation = 'read' THEN

    -- Public access only to verified images
    -- permanently attached to active pages.
    RETURN EXISTS (
      SELECT 1
      FROM public.creatorverse_media m
      JOIN public.creatorverse_pages p
        ON p.id = m.page_id
      WHERE m.storage_path = p_path
        AND m.purpose IN ('logo', 'cover')
        AND m.post_id IS NULL
        AND m.media_type = 'image'
        AND m.verification = 'verified'
        AND p.status = 'active'
        AND (
          (m.purpose = 'logo'
           AND p.logo_media_id = m.id)
          OR
          (m.purpose = 'cover'
           AND p.cover_media_id = m.id)
        )
        AND creatorverse_private.read_page(p.id)
    );

  END IF;

  RETURN false;
END;
$function$;

CREATE OR REPLACE FUNCTION creatorverse_private.publish_text_draft(p_page_id uuid, p_post_id uuid)
 RETURNS uuid
 LANGUAGE plpgsql
 SECURITY DEFINER
 SET search_path TO ''
AS $function$
DECLARE
  v_content text;
  v_verified_count integer;
BEGIN
  -- Preserve existing authentication and ownership checks.
  IF auth.uid() IS NULL
     OR NOT creatorverse_private.write_enabled()
     OR NOT creatorverse_private.owns(p_page_id)
  THEN
    RAISE EXCEPTION 'Publishing unavailable'
      USING ERRCODE = '42501';
  END IF;

  -- Lock page before post. All media write paths must
  -- follow this same order.
  PERFORM 1
  FROM public.creatorverse_pages
  WHERE id = p_page_id
    AND owner_id = auth.uid()
    AND status = 'active'
  FOR UPDATE;

  IF NOT FOUND THEN
    RAISE EXCEPTION 'Page unavailable'
      USING ERRCODE = '42501';
  END IF;

  SELECT content
  INTO v_content
  FROM public.creatorverse_posts
  WHERE id = p_post_id
    AND page_id = p_page_id
    AND author_id = auth.uid()
    AND status = 'draft'
  FOR UPDATE;

  IF NOT FOUND THEN
    RAISE EXCEPTION 'Draft unavailable'
      USING ERRCODE = '42501';
  END IF;

  -- Keep the existing text requirement.
  IF v_content IS NULL
     OR char_length(btrim(v_content)) > 5000
  THEN
    RAISE EXCEPTION 'Invalid post content'
      USING ERRCODE = '23514';
  END IF;

  -- Reject unexpected attachment records and
  -- anything not verified or explicitly rejected.
  IF EXISTS (
    SELECT 1
    FROM public.creatorverse_media AS m
    WHERE m.post_id = p_post_id
      AND (
        m.page_id IS DISTINCT FROM p_page_id
        OR m.purpose IS DISTINCT FROM 'post'
        OR m.verification IS NULL
        OR m.verification NOT IN ('verified', 'rejected')
      )
  ) THEN
    RAISE EXCEPTION 'Attachments are not ready'
      USING ERRCODE = '23514';
  END IF;

  SELECT count(*)
  INTO v_verified_count
  FROM public.creatorverse_media AS m
  WHERE m.post_id = p_post_id
    AND m.page_id = p_page_id
    AND m.verification = 'verified';


    IF btrim(v_content) = '' AND v_verified_count = 0 THEN
      RAISE EXCEPTION
        'Post requires text or verified media'
        USING ERRCODE = '23514';
    END IF;

    IF v_verified_count > 4 THEN

    RAISE EXCEPTION 'Attachment limit exceeded'
      USING ERRCODE = '23514';
  END IF;

  -- Every verified attachment must have valid metadata,
  -- an existing private Storage object, and evidence
  -- that its upload reservation was consumed.
  IF EXISTS (
    SELECT 1
    FROM public.creatorverse_media AS m
    WHERE m.post_id = p_post_id
      AND m.page_id = p_page_id
      AND m.verification = 'verified'
      AND (
        m.byte_size IS NULL
        OR m.byte_size <= 0
        OR m.display_order IS NULL
        OR m.display_order NOT BETWEEN 0 AND 3

        OR (
          m.media_type = 'image'
          AND (
            m.mime_type NOT IN (
              'image/jpeg', 'image/png', 'image/webp'
            )
            OR m.byte_size > 8388608
            OR m.duration_seconds IS NOT NULL
          )
        )

        OR (
          m.media_type = 'video'
          AND (
            m.mime_type NOT IN (
              'video/mp4', 'video/webm'
            )
            OR m.byte_size > 33554432
            OR m.duration_seconds IS NULL
            OR m.duration_seconds <= 0
            OR m.duration_seconds > 120
          )
        )

        OR m.media_type IS DISTINCT FROM 'image'
           AND m.media_type IS DISTINCT FROM 'video'

        OR NOT EXISTS (
          SELECT 1
          FROM storage.objects AS obj
          WHERE obj.bucket_id = 'creatorverse-media'
            AND obj.name = m.storage_path
        )

        OR NOT EXISTS (
          SELECT 1
          FROM creatorverse_private.upload_reservations AS r
          WHERE r.media_id = m.id
            AND r.page_id = p_page_id
            AND r.owner_id = auth.uid()
            AND r.consumed_at IS NOT NULL
        )
      )
  ) THEN
    RAISE EXCEPTION 'Verified media validation failed'
      USING ERRCODE = '23514';
  END IF;

  UPDATE public.creatorverse_posts
  SET
    status = 'published',
    updated_at = pg_catalog.now()
  WHERE id = p_post_id
    AND page_id = p_page_id
    AND status = 'draft';

  IF NOT FOUND THEN
    RAISE EXCEPTION 'Publication could not be completed'
      USING ERRCODE = '23514';
  END IF;

  RETURN p_post_id;
END;
$function$;

CREATE OR REPLACE FUNCTION creatorverse_private.storage_allowed(p_path text, p_operation text)
 RETURNS boolean
 LANGUAGE plpgsql
 STABLE SECURITY DEFINER
 SET search_path TO ''
AS $function$
DECLARE
  v_user uuid := auth.uid();
BEGIN
  IF v_user IS NULL
     OR p_path IS NULL
     OR p_operation IS NULL
  THEN
    RETURN false;
  END IF;

  IF p_operation = 'upload' THEN

    IF NOT creatorverse_private.write_enabled() THEN
      RETURN false;
    END IF;

    -- Existing post-media upload authorization.
    IF EXISTS (
      SELECT 1
      FROM public.creatorverse_media m
      JOIN public.creatorverse_posts p
        ON p.id = m.post_id
       AND p.page_id = m.page_id
      JOIN public.creatorverse_pages page
        ON page.id = m.page_id
      JOIN creatorverse_private.upload_reservations r
        ON r.media_id = m.id
       AND r.page_id = m.page_id
      WHERE m.storage_path = p_path
        AND m.purpose = 'post'
        AND m.verification = 'pending'
        AND p.status = 'draft'
        AND p.author_id = v_user
        AND page.owner_id = v_user
        AND page.status IN ('draft', 'active')
        AND creatorverse_private.owns(page.id)
        AND r.owner_id = v_user
        AND r.consumed_at IS NULL
        AND r.expires_at >
          pg_catalog.statement_timestamp()
    ) THEN
      RETURN true;
    END IF;

    -- New reserved profile and cover image authorization.
    RETURN creatorverse_private.page_image_storage_allowed(
      p_path, 'upload'
    );

  ELSIF p_operation = 'read' THEN

    -- Existing published post-media read authorization.
    IF EXISTS (
      SELECT 1
      FROM public.creatorverse_media m
      JOIN public.creatorverse_posts p
        ON p.id = m.post_id
       AND p.page_id = m.page_id
      JOIN public.creatorverse_pages page
        ON page.id = m.page_id
      WHERE m.storage_path = p_path
        AND m.purpose = 'post'
        AND m.verification = 'verified'
        AND p.status = 'published'
        AND page.status = 'active'
        AND creatorverse_private.read_post(p.id)
    ) THEN
      RETURN true;
    END IF;

    -- New verified and attached page-image read authorization.
    RETURN creatorverse_private.page_image_storage_allowed(
      p_path, 'read'
    );

  END IF;

  RETURN false;
END;
$function$;

CREATE OR REPLACE FUNCTION creatorverse_private.validate_links()
 RETURNS trigger
 LANGUAGE plpgsql
 SET search_path TO ''
AS $function$
DECLARE item jsonb;
BEGIN
  FOR item IN SELECT value FROM jsonb_array_elements(NEW.social_links) LOOP
    IF jsonb_typeof(item)<>'object' OR jsonb_typeof(item->'label') IS DISTINCT FROM 'string'
       OR jsonb_typeof(item->'url') IS DISTINCT FROM 'string'
       OR char_length(btrim(item->>'label')) NOT BETWEEN 1 AND 60
       OR char_length(item->>'url')>2048
       OR (item->>'url') !~ '^https?://[^[:space:]@/]+([/?#][^[:space:]]*)?$' THEN
      RAISE EXCEPTION 'Invalid public link';
    END IF;
  END LOOP;
  RETURN NEW;
END; $function$;

CREATE OR REPLACE FUNCTION creatorverse_private.validate_live_page_owner()
 RETURNS trigger
 LANGUAGE plpgsql
 SECURITY DEFINER
 SET search_path TO ''
AS $function$
BEGIN
  -- Existing personal broadcasts require no new permission.
  IF NEW.creatorverse_page_id IS NULL THEN
    RETURN NEW;
  END IF;

  -- Only the authenticated broadcaster can associate a page.
  IF auth.uid() IS DISTINCT FROM NEW.user_id THEN
    RAISE EXCEPTION
      'Unauthorized CreatorVerse livestream association'
      USING ERRCODE = '42501';
  END IF;

  -- The page must be active and belong to the broadcaster.
  IF NOT EXISTS (
    SELECT 1
    FROM public.creatorverse_pages AS p
    WHERE p.id = NEW.creatorverse_page_id
      AND p.owner_id = NEW.user_id
      AND p.status = 'active'
  ) THEN
    RAISE EXCEPTION
      'CreatorVerse page is not owned or active'
      USING ERRCODE = '42501';
  END IF;

  RETURN NEW;
END;
$function$;

CREATE OR REPLACE FUNCTION parapost_groups_private.allowed(gid uuid, level text)
 RETURNS boolean
 LANGUAGE sql
 STABLE SECURITY DEFINER
 SET search_path TO ''
AS $function$
 SELECT parapost_groups_private.enabled() AND EXISTS (
   SELECT 1 FROM public.groups g LEFT JOIN public.group_members m
     ON m.group_id=g.id AND m.user_id=auth.uid()
   WHERE g.id=gid AND NOT parapost_groups_private.blocked(auth.uid(),g.owner_id)
   AND CASE level
     WHEN 'directory' THEN true
     WHEN 'read' THEN coalesce(m.status NOT IN ('rejected','removed','banned'),true)
       AND (g.visibility='public' OR m.status='approved')
     WHEN 'member' THEN m.status='approved'
     WHEN 'admin' THEN m.status='approved' AND m.role IN ('owner','admin')
     WHEN 'moderate' THEN m.status='approved' AND m.role IN ('owner','admin','moderator')
     WHEN 'owner' THEN m.status='approved' AND m.role='owner' AND g.owner_id=auth.uid()
     ELSE false END
 );
$function$;

CREATE OR REPLACE FUNCTION parapost_groups_private.command(action text, gid uuid, payload jsonb)
 RETURNS jsonb
 LANGUAGE plpgsql
 SECURITY DEFINER
 SET search_path TO ''
AS $function$
DECLARE
  uid uuid := auth.uid();
  g public.groups;
  m public.group_members;
  p public.group_posts;
  target uuid;
  pid uuid;
  mid uuid;
  path text;
  mime text;
  role_name text;
  result jsonb;
  item record;
BEGIN
  IF NOT parapost_groups_private.enabled() THEN
    RAISE EXCEPTION 'Groups unavailable'
      USING ERRCODE = '42501';
  END IF;

  IF action = 'create' THEN
    IF (
      SELECT count(*)
      FROM public.groups
      WHERE owner_id = uid
    ) >= 100 THEN
      RAISE EXCEPTION 'Group creation limit reached';
    END IF;

    INSERT INTO public.groups(
      owner_id, name, description, visibility
    )
    VALUES (
      uid,
      btrim(payload->>'name'),
      coalesce(payload->>'description', ''),
      coalesce(payload->>'visibility', 'public')
    )
    RETURNING * INTO g;

    RETURN jsonb_build_object('id', g.id);
  END IF;

  SELECT * INTO g
  FROM public.groups
  WHERE id = gid
  FOR UPDATE;

  IF NOT FOUND
    OR NOT parapost_groups_private.allowed(
      gid, 'directory'
    ) THEN
    RAISE EXCEPTION 'Group unavailable'
      USING ERRCODE = '42501';
  END IF;

  SELECT * INTO m
  FROM public.group_members
  WHERE group_id = gid AND user_id = uid;

  IF action = 'join' THEN
    IF m.status IS NOT NULL THEN
      IF m.status NOT IN ('pending', 'approved') THEN
        RAISE EXCEPTION 'Membership unavailable';
      END IF;
    ELSE
      INSERT INTO public.group_members(
        group_id, user_id, role, status, approved_at
      )
      VALUES (
        gid, uid, 'member',
        CASE WHEN g.visibility = 'public'
          THEN 'approved' ELSE 'pending' END,
        CASE WHEN g.visibility = 'public'
          THEN now() END
      );
    END IF;

  ELSIF action = 'leave' THEN
    IF m.role = 'owner' THEN
      RAISE EXCEPTION
        'Owner must delete the Group; transfer is not supported';
    END IF;

    DELETE FROM public.group_members
    WHERE group_id = gid
      AND user_id = uid
      AND status IN ('pending', 'approved');

  ELSIF action IN (
    'approve', 'reject', 'remove',
    'ban', 'role', 'restore'
  ) THEN
    target := (payload->>'user_id')::uuid;

    IF NOT parapost_groups_private.allowed(gid, 'admin')
      OR target = uid
      OR target = g.owner_id THEN
      RAISE EXCEPTION 'Not authorized'
        USING ERRCODE = '42501';
    END IF;

    SELECT * INTO m
    FROM public.group_members
    WHERE group_id = gid AND user_id = target;

    IF NOT FOUND THEN
      RAISE EXCEPTION 'Membership unavailable';
    END IF;

    IF NOT parapost_groups_private.allowed(gid, 'owner')
      AND (
        m.role <> 'member'
        OR action IN ('role', 'restore')
      ) THEN
      RAISE EXCEPTION 'Owner permission required';
    END IF;

    IF action IN ('approve', 'restore')
      AND parapost_groups_private.blocked(uid, target) THEN
      RAISE EXCEPTION 'Membership unavailable';
    END IF;

    IF action IN ('approve', 'restore')
      AND parapost_groups_private.blocked(g.owner_id, target) THEN
      RAISE EXCEPTION 'Membership unavailable';
    END IF;

    IF action IN ('approve', 'reject') THEN
      IF m.status <> 'pending' THEN
        RAISE EXCEPTION 'Request is no longer pending';
      END IF;

      UPDATE public.group_members
      SET status = CASE
            WHEN action = 'approve'
            THEN 'approved' ELSE 'rejected' END,
          approved_at = CASE
            WHEN action = 'approve'
            THEN now() END
      WHERE group_id = gid AND user_id = target;

    ELSIF action = 'role' THEN
      role_name := payload->>'role';

      IF m.status <> 'approved'
        OR role_name IS NULL
        OR role_name NOT IN (
          'admin', 'moderator', 'member'
        ) THEN
        RAISE EXCEPTION 'Invalid role change';
      END IF;

      UPDATE public.group_members
      SET role = role_name
      WHERE group_id = gid AND user_id = target;

    ELSIF action = 'restore' THEN
      IF m.status NOT IN (
        'removed', 'rejected', 'banned'
      ) THEN
        RAISE EXCEPTION 'Invalid restoration';
      END IF;

      UPDATE public.group_members
      SET role = 'member',
          status = 'pending',
          approved_at = NULL
      WHERE group_id = gid AND user_id = target;

    ELSE
      UPDATE public.group_members
      SET role = 'member',
          status = CASE
            WHEN action = 'ban'
            THEN 'banned' ELSE 'removed' END,
          approved_at = NULL
      WHERE group_id = gid AND user_id = target;
    END IF;

  ELSIF action = 'edit' THEN
    IF NOT parapost_groups_private.allowed(gid, 'admin') THEN
      RAISE EXCEPTION 'Not authorized';
    END IF;

    IF payload->>'visibility' IS DISTINCT FROM g.visibility
      AND NOT parapost_groups_private.allowed(
        gid, 'owner'
      ) THEN
      RAISE EXCEPTION 'Owner permission required';
    END IF;

    UPDATE public.groups
    SET name = btrim(payload->>'name'),
        description = coalesce(payload->>'description', ''),
        visibility = payload->>'visibility',
        updated_at = now()
    WHERE id = gid;

  ELSIF action = 'delete' THEN
    IF NOT parapost_groups_private.allowed(gid, 'owner')
      OR payload->>'confirmation' IS DISTINCT FROM g.name
    THEN
      RAISE EXCEPTION
        'Owner and Group name confirmation required';
    END IF;

    DELETE FROM public.groups WHERE id = gid;

  ELSIF action = 'draft' THEN
    IF NOT parapost_groups_private.allowed(gid, 'member') THEN
      RAISE EXCEPTION 'Approved membership required';
    END IF;

    IF (
      SELECT count(*)
      FROM public.group_posts
      WHERE group_id = gid
        AND author_id = uid
        AND NOT published
    ) >= 10 THEN
      RAISE EXCEPTION 'Too many unfinished uploads';
    END IF;

    INSERT INTO public.group_posts(
      group_id, author_id, content
    )
    VALUES (
      gid, uid, coalesce(payload->>'content', '')
    )
    RETURNING id INTO pid;

    RETURN jsonb_build_object('id', pid);

  ELSIF action IN (
    'reserve_media', 'publish',
    'edit_post', 'delete_post'
  ) THEN
    pid := (payload->>'post_id')::uuid;

    SELECT * INTO p
    FROM public.group_posts
    WHERE id = pid AND group_id = gid;

    IF NOT FOUND OR p.is_deleted THEN
      RAISE EXCEPTION 'Post unavailable';
    END IF;

    IF NOT parapost_groups_private.allowed(gid, 'member')
      OR (
        p.author_id <> uid
        AND NOT (
          action = 'delete_post'
          AND parapost_groups_private.allowed(
            gid, 'moderate'
          )
        )
      ) THEN
      RAISE EXCEPTION 'Not authorized';
    END IF;

    IF action = 'delete_post' THEN
      DELETE FROM public.group_posts
      WHERE id = pid;

    ELSIF action = 'edit_post' THEN
      IF char_length(
        btrim(coalesce(payload->>'content', ''))
      ) = 0
        AND NOT EXISTS (
          SELECT 1 FROM public.group_post_media
          WHERE post_id = pid
        ) THEN
        RAISE EXCEPTION 'Post cannot be empty';
      END IF;

      UPDATE public.group_posts
      SET content = coalesce(payload->>'content', ''),
          updated_at = now()
      WHERE id = pid;

    ELSIF action = 'reserve_media' THEN
      IF p.published
        OR (
          SELECT count(*)
          FROM public.group_post_media
          WHERE post_id = pid
        ) >= 4 THEN
        RAISE EXCEPTION 'Media limit or published post';
      END IF;

      mime := payload->>'mime_type';

      IF mime IS NULL OR mime NOT IN (
        'image/jpeg', 'image/png', 'image/webp',
        'video/mp4', 'video/webm'
      ) THEN
        RAISE EXCEPTION 'Unsupported media';
      END IF;

      mid := gen_random_uuid();

      path := gid::text
        || '/posts/' || pid::text
        || '/' || uid::text
        || '/' || mid::text;

      INSERT INTO public.group_post_media(
        id, group_id, post_id, uploader_id,
        media_type, storage_path, mime_type,
        display_order, poster_path
      )
      VALUES (
        mid, gid, pid, uid,
        CASE WHEN mime LIKE 'image/%'
          THEN 'image' ELSE 'video' END,
        path, mime,
        (
          SELECT count(*)
          FROM public.group_post_media
          WHERE post_id = pid
        ),
        CASE WHEN mime LIKE 'video/%'
          THEN path || '.poster.jpg'
          ELSE NULL END
      );

      RETURN jsonb_build_object(
        'path', path,
        'poster_path',
        CASE WHEN mime LIKE 'video/%'
          THEN path || '.poster.jpg'
          ELSE NULL END
      );

    ELSE
      IF p.published THEN
        RETURN jsonb_build_object('id', pid);
      END IF;

      IF char_length(btrim(p.content)) = 0
        AND NOT EXISTS (
          SELECT 1 FROM public.group_post_media
          WHERE post_id = pid
        ) THEN
        RAISE EXCEPTION 'Post cannot be empty';
      END IF;

      FOR item IN
        SELECT * FROM public.group_post_media
        WHERE post_id = pid
      LOOP
        IF NOT EXISTS (
          SELECT 1 FROM storage.objects o
          WHERE o.bucket_id = 'group-media'
            AND o.name = item.storage_path
            AND o.metadata->>'mimetype' = item.mime_type
            AND (o.metadata->>'size')::bigint
              BETWEEN 1 AND
              CASE
                WHEN item.media_type = 'image'
                THEN 8388608
                ELSE 33554432
              END
        ) THEN
          RAISE EXCEPTION
            'Media upload incomplete or invalid';
        END IF;
      END LOOP;

      UPDATE public.group_posts
      SET published = true,
          updated_at = now()
      WHERE id = pid;
    END IF;

  ELSIF action = 'remove_cover' THEN
    IF NOT parapost_groups_private.allowed(gid, 'admin') THEN
      RAISE EXCEPTION 'Not authorized';
    END IF;

    UPDATE public.groups
    SET cover_storage_path = NULL,
        updated_at = now()
    WHERE id = gid;

  ELSIF action = 'reserve_cover' THEN
    IF NOT parapost_groups_private.allowed(gid, 'admin') THEN
      RAISE EXCEPTION 'Not authorized';
    END IF;

    IF (
      SELECT count(*)
      FROM parapost_groups_private.cover_uploads
      WHERE group_id = gid AND uploader_id = uid
    ) >= 10 THEN
      RAISE EXCEPTION 'Too many unfinished covers';
    END IF;

    path := gid::text
      || '/covers/'
      || gen_random_uuid()::text;

    INSERT INTO parapost_groups_private.cover_uploads(
      storage_path, group_id, uploader_id
    )
    VALUES (path, gid, uid);

    RETURN jsonb_build_object('path', path);

  ELSIF action = 'set_cover' THEN
    path := payload->>'path';

    IF NOT parapost_groups_private.allowed(gid, 'admin')
      OR NOT EXISTS (
        SELECT 1
        FROM parapost_groups_private.cover_uploads c
        JOIN storage.objects o
          ON o.name = c.storage_path
         AND o.bucket_id = 'group-media'
        WHERE c.storage_path = path
          AND c.group_id = gid
          AND c.uploader_id = uid
          AND o.metadata->>'mimetype'
            IN ('image/jpeg', 'image/png', 'image/webp')
          AND (o.metadata->>'size')::bigint
            BETWEEN 1 AND 8388608
      ) THEN
      RAISE EXCEPTION 'Invalid cover';
    END IF;

    UPDATE public.groups
    SET cover_storage_path = path,
        updated_at = now()
    WHERE id = gid;

    DELETE FROM parapost_groups_private.cover_uploads
    WHERE storage_path = path;

  ELSE
    RAISE EXCEPTION 'Unknown Group action';
  END IF;

  RETURN jsonb_build_object('id', gid);
END;
$function$;

CREATE OR REPLACE FUNCTION parapost_groups_private.guard_owner_delete()
 RETURNS trigger
 LANGUAGE plpgsql
 SECURITY DEFINER
 SET search_path TO ''
AS $function$
BEGIN
 -- Parent is already absent during a legitimate ON DELETE CASCADE.
 IF OLD.role='owner' AND EXISTS(SELECT 1 FROM public.groups WHERE id=OLD.group_id) THEN
   RAISE EXCEPTION 'Delete the Group instead of its owner membership';
 END IF;
 RETURN OLD;
END; $function$;

CREATE OR REPLACE FUNCTION parapost_groups_private.guard_owner_member()
 RETURNS trigger
 LANGUAGE plpgsql
 SET search_path TO ''
AS $function$
BEGIN
  IF TG_OP = 'INSERT' THEN
    IF NEW.role = 'owner' THEN
      IF NEW.status <> 'approved'
         OR NOT EXISTS (
           SELECT 1
           FROM public.groups g
           WHERE g.id = NEW.group_id
             AND g.owner_id = NEW.user_id
         ) THEN
        RAISE EXCEPTION
          'Invalid Group owner membership';
      END IF;
    END IF;

  ELSIF TG_OP = 'UPDATE' THEN
    IF OLD.role = 'owner'
       AND NEW IS DISTINCT FROM OLD THEN
      RAISE EXCEPTION
        'Owner membership cannot be modified';
    END IF;

    IF OLD.role <> 'owner'
       AND NEW.role = 'owner' THEN
      RAISE EXCEPTION
        'Owner promotion requires a separate transfer';
    END IF;
  END IF;

  RETURN NEW;
END;
$function$;

CREATE OR REPLACE FUNCTION parapost_groups_private.notify_owner_new_post_release()
 RETURNS trigger
 LANGUAGE plpgsql
 SECURITY DEFINER
 SET search_path TO ''
AS $function$
DECLARE
    recipient_id uuid;
BEGIN
    IF TG_OP = 'DELETE' THEN
        DELETE FROM public.notifications
        WHERE group_id = OLD.group_id
          AND community_post_id = OLD.id
          AND (type = 'group_post' OR community_event_key IS NOT NULL);
        RETURN OLD;
    END IF;

    IF NEW.published IS NOT TRUE
       OR NEW.is_deleted IS TRUE
    THEN
        DELETE FROM public.notifications
        WHERE group_id = NEW.group_id
          AND community_post_id = NEW.id
          AND (type = 'group_post' OR community_event_key IS NOT NULL);
        RETURN NEW;
    END IF;

    IF TG_OP = 'UPDATE' THEN
        IF OLD.published IS TRUE
           AND OLD.is_deleted IS NOT TRUE
        THEN
            RETURN NEW;
        END IF;
    END IF;

    SELECT owner_id INTO recipient_id
    FROM public.groups
    WHERE id = NEW.group_id;

    IF recipient_id IS NULL
       OR NEW.author_id IS NULL
       OR recipient_id = NEW.author_id
    THEN
        RETURN NEW;
    END IF;

    IF parapost_groups_private.blocked(
        recipient_id, NEW.author_id
    ) THEN
        RETURN NEW;
    END IF;

    BEGIN
        INSERT INTO public.notifications (
            user_id,
            actor_id,
            type,
            group_id,
            community_post_id,
            message,
            is_read
        )
        VALUES (
            recipient_id,
            NEW.author_id,
            'group_post',
            NEW.group_id,
            NEW.id,
            'A member published a new Group post.',
            false
        )
        ON CONFLICT DO NOTHING;

    EXCEPTION WHEN OTHERS THEN
        RAISE WARNING
            'Group post notification failed: %',
            SQLERRM;
    END;

    RETURN NEW;
EXCEPTION WHEN OTHERS THEN
    RAISE WARNING 'Group owner post notification failed: %', SQLERRM;
    IF TG_OP = 'DELETE' THEN RETURN OLD; END IF;
    RETURN NEW;
END;
$function$;

CREATE OR REPLACE FUNCTION parapost_groups_private.promotion_actor(gid uuid, actor uuid)
 RETURNS boolean
 LANGUAGE sql
 STABLE SECURITY DEFINER
 SET search_path TO ''
AS $function$
  SELECT actor IS NOT NULL AND EXISTS (
    SELECT 1
    FROM public.groups g
    LEFT JOIN public.group_members m
      ON m.group_id = g.id AND m.user_id = actor
    LEFT JOIN parapost_groups_private.promotion_settings s
      ON s.group_id = g.id
    WHERE g.id = gid
      AND NOT parapost_groups_private.blocked(actor, g.owner_id)
      AND coalesce(
        m.status NOT IN ('rejected','removed','banned'), true
      )
      AND (
        g.visibility = 'public'
        OR (
          m.status = 'approved'
          AND coalesce(s.enabled, false)
        )
      )
  );
$function$;

CREATE OR REPLACE FUNCTION parapost_groups_private.promotion_cover(gid uuid, cover_key text, source_id uuid DEFAULT NULL::uuid)
 RETURNS text
 LANGUAGE plpgsql
 STABLE SECURITY DEFINER
 SET search_path TO ''
AS $function$
DECLARE
  preview jsonb;
BEGIN
  preview := parapost_groups_private.promotion_preview(
    gid, source_id
  );

  IF preview IS NULL
     OR preview->>'cover_key' IS DISTINCT FROM cover_key
     OR cover_key IS NULL
  THEN
    RAISE EXCEPTION 'Cover unavailable'
      USING ERRCODE = '42501';
  END IF;

  RETURN (
    SELECT cover_storage_path
    FROM public.groups
    WHERE id = gid
  );
END;
$function$;

CREATE OR REPLACE FUNCTION parapost_groups_private.promotion_feed(sharer uuid DEFAULT NULL::uuid, page_limit integer DEFAULT 60, page_offset integer DEFAULT 0)
 RETURNS jsonb
 LANGUAGE sql
 STABLE SECURITY DEFINER
 SET search_path TO ''
AS $function$
  SELECT coalesce(
    jsonb_agg(x.item ORDER BY x.created_at DESC, x.id DESC),
    '[]'::jsonb
  )
  FROM (
    SELECT
      s.id,
      s.created_at,
      jsonb_build_object(
        'share_id', s.id,
        'group_id', s.group_id,
        'post_id', NULL,
        'sharer_id', s.user_id,
        'caption', s.caption,
        'shared_at', s.created_at,
        'group_name', preview->>'name',
        'original_author_id', NULL,
        'original_content', NULL,
        'original_created_at', NULL,
        'media', '[]'::jsonb,
        'promotion', preview
      ) AS item
    FROM public.group_promotional_shares s
    CROSS JOIN LATERAL (
      SELECT parapost_groups_private.promotion_preview(
        s.group_id, s.id
      ) AS preview
    ) p
    WHERE parapost_groups_private.enabled()
      AND (sharer IS NULL OR s.user_id = sharer)
      AND preview IS NOT NULL
      AND NOT EXISTS (
        SELECT 1
        FROM public.group_post_shares legacy
        JOIN public.groups g ON g.id = s.group_id
        WHERE g.visibility = 'public'
          AND legacy.post_id = s.post_id
          AND legacy.user_id = s.user_id
      )
    ORDER BY s.created_at DESC, s.id DESC
    LIMIT least(greatest(coalesce(page_limit,60),1),100)
    OFFSET least(greatest(coalesce(page_offset,0),0),10000)
  ) x;
$function$;

CREATE OR REPLACE FUNCTION parapost_groups_private.promotion_interact(p_share_id uuid, p_action text, p_content text DEFAULT NULL::text, p_parent_id uuid DEFAULT NULL::uuid)
 RETURNS jsonb
 LANGUAGE plpgsql
 SECURITY DEFINER
 SET search_path TO ''
AS $function$
DECLARE
  v_uid uuid := auth.uid();
  v_share public.group_promotional_shares%ROWTYPE;
  v_group public.groups%ROWTYPE;
  v_parent public.group_promotion_share_comments%ROWTYPE;
  v_id uuid;
  v_limit_kind text;
  v_limit_max integer;
  v_limit_now timestamptz;
  v_limit_count integer;
BEGIN
  IF v_uid IS NULL THEN
    RAISE EXCEPTION 'Sign in required'
      USING ERRCODE = '42501';
  END IF;

  SELECT * INTO v_share
  FROM public.group_promotional_shares
  WHERE id = p_share_id;

  IF NOT FOUND THEN
    RAISE EXCEPTION 'Share unavailable'
      USING ERRCODE = '42501';
  END IF;

  SELECT * INTO v_group
  FROM public.groups
  WHERE id = v_share.group_id
  FOR SHARE;

  PERFORM 1
  FROM public.group_promotional_shares
  WHERE id = p_share_id
  FOR SHARE;

  IF NOT FOUND
     OR parapost_groups_private.promotion_preview(
       v_share.group_id, p_share_id
     ) IS NULL THEN
    RAISE EXCEPTION 'Share unavailable'
      USING ERRCODE = '42501';
  END IF;

  IF p_action IN ('comment', 'reply') THEN
    v_limit_kind := 'comment';
    v_limit_max := 10;
  ELSIF p_action IN (
    'like', 'unlike', 'like_comment', 'unlike_comment'
  ) THEN
    v_limit_kind := 'reaction';
    v_limit_max := 60;
  END IF;

  IF v_limit_kind IS NOT NULL THEN
    v_limit_now := clock_timestamp();

    INSERT INTO parapost_groups_private.shared_post_interaction_rate_limits AS rl
      (user_id, action_kind, window_started_at, action_count)
    VALUES (v_uid, v_limit_kind, v_limit_now, 1)
    ON CONFLICT (user_id, action_kind)
    DO UPDATE SET
      window_started_at = CASE
        WHEN rl.window_started_at <=
             v_limit_now - interval '60 seconds'
        THEN v_limit_now
        ELSE rl.window_started_at
      END,
      action_count = CASE
        WHEN rl.window_started_at <=
             v_limit_now - interval '60 seconds'
        THEN 1
        ELSE rl.action_count + 1
      END
    WHERE rl.window_started_at <=
          v_limit_now - interval '60 seconds'
       OR rl.action_count < v_limit_max
    RETURNING action_count INTO v_limit_count;

    IF NOT FOUND THEN
      RAISE EXCEPTION 'Rate limit exceeded. Try again shortly.'
        USING ERRCODE = 'P0001';
    END IF;
  END IF;

  IF p_action = 'like' THEN
    INSERT INTO public.group_promotion_share_likes
      (share_id, user_id)
    VALUES (p_share_id, v_uid)
    ON CONFLICT (share_id, user_id) DO NOTHING;

    RETURN jsonb_build_object('liked', true);

  ELSIF p_action = 'unlike' THEN
    DELETE FROM public.group_promotion_share_likes
    WHERE share_id = p_share_id
      AND user_id = v_uid;

    RETURN jsonb_build_object('liked', false);

  ELSIF p_action = 'comment' THEN
    IF p_parent_id IS NOT NULL
       OR p_content IS NULL
       OR char_length(btrim(p_content))
          NOT BETWEEN 1 AND 1200 THEN
      RAISE EXCEPTION 'Invalid comment';
    END IF;

    INSERT INTO public.group_promotion_share_comments
      (share_id, author_id, content)
    VALUES (p_share_id, v_uid, btrim(p_content))
    RETURNING id INTO v_id;

    RETURN jsonb_build_object('id', v_id);

  ELSIF p_action IN (
    'reply', 'like_comment', 'unlike_comment',
    'edit_comment', 'delete_comment'
  ) THEN
    IF p_parent_id IS NULL THEN
      RAISE EXCEPTION 'Comment required';
    END IF;

    SELECT * INTO v_parent
    FROM public.group_promotion_share_comments
    WHERE id = p_parent_id
      AND share_id = p_share_id;

    IF NOT FOUND THEN
      RAISE EXCEPTION 'Comment unavailable'
        USING ERRCODE = '42501';
    END IF;

    IF parapost_groups_private.blocked(
      v_uid, v_parent.author_id
    )
       OR (
         v_parent.parent_id IS NOT NULL
         AND NOT EXISTS (
           SELECT 1
           FROM public.group_promotion_share_comments parent
           WHERE parent.id = v_parent.parent_id
             AND parent.share_id = p_share_id
             AND parent.parent_id IS NULL
             AND NOT parapost_groups_private.blocked(
               v_uid, parent.author_id
             )
         )
       ) THEN
      RAISE EXCEPTION 'Comment unavailable'
        USING ERRCODE = '42501';
    END IF;

    IF p_action = 'reply' THEN
      IF v_parent.parent_id IS NOT NULL
         OR parapost_groups_private.blocked(
           v_uid, v_parent.author_id
         )
         OR p_content IS NULL
         OR char_length(btrim(p_content))
            NOT BETWEEN 1 AND 1200 THEN
        RAISE EXCEPTION 'Reply unavailable'
          USING ERRCODE = '42501';
      END IF;

      INSERT INTO public.group_promotion_share_comments
        (share_id, author_id, parent_id, content)
      VALUES (
        p_share_id, v_uid, p_parent_id, btrim(p_content)
      )
      RETURNING id INTO v_id;

      RETURN jsonb_build_object('id', v_id);

    ELSIF p_action = 'like_comment' THEN
      IF parapost_groups_private.blocked(
        v_uid, v_parent.author_id
      )
      OR (
        v_parent.parent_id IS NOT NULL
        AND NOT EXISTS (
          SELECT 1
          FROM public.group_promotion_share_comments parent
          WHERE parent.id = v_parent.parent_id
            AND parent.share_id = p_share_id
            AND parent.parent_id IS NULL
            AND NOT parapost_groups_private.blocked(
              v_uid, parent.author_id
            )
        )
      ) THEN
        RAISE EXCEPTION 'Comment Like unavailable'
          USING ERRCODE = '42501';
      END IF;

      INSERT INTO public.group_promotion_share_comment_likes
        (comment_id, share_id, user_id)
      VALUES (p_parent_id, p_share_id, v_uid)
      ON CONFLICT (comment_id, user_id) DO NOTHING;

      RETURN jsonb_build_object('liked', true);

    ELSIF p_action = 'unlike_comment' THEN
      DELETE FROM public.group_promotion_share_comment_likes
      WHERE comment_id = p_parent_id
        AND share_id = p_share_id
        AND user_id = v_uid;

      RETURN jsonb_build_object('liked', false);

    ELSIF p_action = 'edit_comment' THEN
      IF v_parent.author_id <> v_uid
         OR p_content IS NULL
         OR char_length(btrim(p_content))
            NOT BETWEEN 1 AND 1200 THEN
        RAISE EXCEPTION 'Edit unavailable'
          USING ERRCODE = '42501';
      END IF;

      UPDATE public.group_promotion_share_comments
      SET content = btrim(p_content),
          updated_at = now()
      WHERE id = p_parent_id
        AND share_id = p_share_id
        AND author_id = v_uid;

      RETURN jsonb_build_object('updated', true);

    ELSIF p_action = 'delete_comment' THEN
      IF v_parent.author_id <> v_uid THEN
        RAISE EXCEPTION 'Delete unavailable'
          USING ERRCODE = '42501';
      END IF;

      DELETE FROM public.group_promotion_share_comments
      WHERE id = p_parent_id
        AND share_id = p_share_id
        AND author_id = v_uid;

      RETURN jsonb_build_object('deleted', true);
    END IF;
  END IF;

  RAISE EXCEPTION 'Unknown interaction action';
END;
$function$;

CREATE OR REPLACE FUNCTION parapost_groups_private.promotion_preview(gid uuid, source_id uuid DEFAULT NULL::uuid)
 RETURNS jsonb
 LANGUAGE plpgsql
 STABLE SECURITY DEFINER
 SET search_path TO ''
AS $function$
DECLARE
  g public.groups;
  s parapost_groups_private.promotion_settings;
  m public.group_members;
  source public.group_promotional_shares;
  post public.group_posts;
  expose boolean;
BEGIN
  IF NOT parapost_groups_private.enabled() THEN
    RETURN NULL;
  END IF;

  SELECT * INTO g
  FROM public.groups
  WHERE id = gid;

  IF NOT FOUND
     OR parapost_groups_private.blocked(auth.uid(), g.owner_id)
  THEN
    RETURN NULL;
  END IF;

  SELECT * INTO m
  FROM public.group_members
  WHERE group_id = gid AND user_id = auth.uid();

  IF m.status IN ('rejected','removed','banned') THEN
    RETURN NULL;
  END IF;

  SELECT * INTO s
  FROM parapost_groups_private.promotion_settings
  WHERE group_id = gid;

  IF g.visibility = 'private'
     AND NOT coalesce(s.enabled, false)
  THEN
    RETURN NULL;
  END IF;

  IF source_id IS NOT NULL THEN
    SELECT * INTO source
    FROM public.group_promotional_shares
    WHERE id = source_id AND group_id = gid;

    IF NOT FOUND
       OR parapost_groups_private.blocked(
         auth.uid(), source.user_id
       )
       OR NOT parapost_groups_private.promotion_actor(
         gid, source.user_id
       )
    THEN
      RETURN NULL;
    END IF;

    IF source.post_id IS NOT NULL THEN
      SELECT * INTO post
      FROM public.group_posts
      WHERE id = source.post_id
        AND group_id = gid
        AND published
        AND NOT is_deleted;

      IF NOT FOUND
         OR parapost_groups_private.blocked(
           auth.uid(), post.author_id
         )
         OR parapost_groups_private.blocked(
           source.user_id, post.author_id
         )
         OR NOT EXISTS (
           SELECT 1 FROM public.group_members
           WHERE group_id = gid
             AND user_id = source.user_id
             AND status = 'approved'
         )
      THEN
        RETURN NULL;
      END IF;
    END IF;
  END IF;

  expose := g.visibility = 'public';

  RETURN jsonb_build_object(
    'group_id', gid,
    'visibility', g.visibility,
    'name',
      CASE WHEN expose OR s.show_name
        THEN g.name ELSE 'Private Group' END,
    'description',
      CASE WHEN expose OR s.show_description
        THEN g.description ELSE NULL END,
    'member_count',
      CASE WHEN expose THEN (
        SELECT count(*)
        FROM public.group_members
        WHERE group_id = gid AND status = 'approved'
      ) ELSE NULL END,
    'cover_key',
      CASE
        WHEN (expose OR s.show_cover)
          AND g.cover_storage_path IS NOT NULL
        THEN md5(g.cover_storage_path)
        ELSE NULL
      END,
    'can_share',
      parapost_groups_private.promotion_actor(
        gid, auth.uid()
      ) AND (
        source.post_id IS NULL
        OR (
          m.status = 'approved'
          AND NOT parapost_groups_private.blocked(
            auth.uid(), post.author_id
          )
        )
      ),
    'approved', coalesce(m.status = 'approved', false),
    'post_id',
      CASE WHEN m.status = 'approved'
        THEN source.post_id ELSE NULL END,
    'kind',
      CASE WHEN source.post_id IS NULL
        THEN 'group' ELSE 'invitation' END
  );
END;
$function$;

CREATE OR REPLACE FUNCTION parapost_groups_private.promotion_settings_command(gid uuid, changes jsonb DEFAULT NULL::jsonb)
 RETURNS jsonb
 LANGUAGE plpgsql
 SECURITY DEFINER
 SET search_path TO ''
AS $function$
DECLARE
  s parapost_groups_private.promotion_settings;
BEGIN
  PERFORM 1
  FROM public.groups
  WHERE id = gid
  FOR UPDATE;

  IF NOT parapost_groups_private.allowed(gid, 'owner') THEN
    RAISE EXCEPTION 'Not authorized'
      USING ERRCODE = '42501';
  END IF;

  IF changes IS NOT NULL THEN
    IF jsonb_typeof(changes) <> 'object'
       OR EXISTS (
         SELECT 1
         FROM jsonb_each(changes) e
         WHERE e.key NOT IN (
           'enabled','show_name','show_cover','show_description'
         )
         OR jsonb_typeof(e.value) <> 'boolean'
       )
       OR NOT (
         changes ?& ARRAY[
           'enabled','show_name','show_cover','show_description'
         ]
       )
    THEN
      RAISE EXCEPTION 'Invalid settings';
    END IF;

    INSERT INTO parapost_groups_private.promotion_settings
      (group_id, enabled, show_name, show_cover, show_description)
    VALUES (
      gid,
      (changes->>'enabled')::boolean,
      (changes->>'show_name')::boolean,
      (changes->>'show_cover')::boolean,
      (changes->>'show_description')::boolean
    )
    ON CONFLICT (group_id)
    DO UPDATE SET
      enabled = EXCLUDED.enabled,
      show_name = EXCLUDED.show_name,
      show_cover = EXCLUDED.show_cover,
      show_description = EXCLUDED.show_description;
  END IF;

  SELECT * INTO s
  FROM parapost_groups_private.promotion_settings
  WHERE group_id = gid;

  RETURN jsonb_build_object(
    'enabled', coalesce(s.enabled, false),
    'show_name', coalesce(s.show_name, false),
    'show_cover', coalesce(s.show_cover, false),
    'show_description', coalesce(s.show_description, false)
  );
END;
$function$;

CREATE OR REPLACE FUNCTION parapost_groups_private.promotion_share(gid uuid, pid uuid DEFAULT NULL::uuid, message text DEFAULT NULL::text, source_id uuid DEFAULT NULL::uuid)
 RETURNS jsonb
 LANGUAGE plpgsql
 SECURITY DEFINER
 SET search_path TO ''
AS $function$
DECLARE
  sid uuid;
  source public.group_promotional_shares;
  author uuid;
BEGIN
  PERFORM 1 FROM public.groups
  WHERE id = gid FOR UPDATE;

  IF NOT parapost_groups_private.enabled()
     OR NOT parapost_groups_private.promotion_actor(
       gid, auth.uid()
     ) THEN
    RAISE EXCEPTION 'Sharing unavailable'
      USING ERRCODE = '42501';
  END IF;

  -- Only approved members may create Group shares.
  IF NOT parapost_groups_private.allowed(gid, 'member') THEN
    RAISE EXCEPTION 'Sharing unavailable'
      USING ERRCODE = '42501';
  END IF;

  IF char_length(message) > 1000 THEN
    RAISE EXCEPTION 'Message too long';
  END IF;

  IF source_id IS NOT NULL THEN
    IF parapost_groups_private.promotion_preview(
      gid, source_id
    ) IS NULL THEN
      RAISE EXCEPTION 'Share unavailable'
        USING ERRCODE = '42501';
    END IF;

    SELECT * INTO source
    FROM public.group_promotional_shares
    WHERE id = source_id AND group_id = gid;

    IF NOT FOUND THEN
      RAISE EXCEPTION 'Share unavailable'
        USING ERRCODE = '42501';
    END IF;

    pid := source.post_id;
  END IF;

  IF pid IS NOT NULL THEN
    SELECT author_id INTO author
    FROM public.group_posts
    WHERE id = pid
      AND group_id = gid
      AND published
      AND NOT is_deleted;

    IF NOT FOUND
       OR NOT parapost_groups_private.allowed(
         gid, 'member'
       )
       OR parapost_groups_private.blocked(
         auth.uid(), author
       ) THEN
      RAISE EXCEPTION 'Post unavailable'
        USING ERRCODE = '42501';
    END IF;

    -- Preserve existing public-post sharing.
    IF (
      SELECT visibility FROM public.groups WHERE id = gid
    ) = 'public' THEN
      RETURN public.groups_share_post(gid, pid, message);
    END IF;
  END IF;

  PERFORM parapost_groups_private.record_share_cooldown(
    CASE WHEN pid IS NULL THEN 'group' ELSE 'post' END,
    COALESCE(pid, gid)
  );

  INSERT INTO public.group_promotional_shares
    (group_id, post_id, user_id, caption)
  VALUES (
    gid, pid, auth.uid(), NULLIF(btrim(message), '')
  )
  RETURNING id INTO sid;

  RETURN jsonb_build_object('id', sid);
END;
$function$;

CREATE OR REPLACE FUNCTION parapost_groups_private.promotion_stats(p_share_id uuid)
 RETURNS jsonb
 LANGUAGE plpgsql
 STABLE SECURITY DEFINER
 SET search_path TO ''
AS $function$
DECLARE
  v_uid uuid := auth.uid();
  v_group_id uuid;
  v_sharer_id uuid;
  v_comments jsonb;
BEGIN
  IF v_uid IS NULL THEN
    RAISE EXCEPTION 'Sign in required'
      USING ERRCODE = '42501';
  END IF;

  SELECT group_id, user_id INTO v_group_id, v_sharer_id
  FROM public.group_promotional_shares
  WHERE id = p_share_id;

  IF NOT FOUND
     OR parapost_groups_private.promotion_preview(
       v_group_id, p_share_id
     ) IS NULL THEN
    RAISE EXCEPTION 'Share unavailable'
      USING ERRCODE = '42501';
  END IF;

  SELECT COALESCE(
    jsonb_agg(
      jsonb_build_object(
        'id', c.id,
        'author_id', c.author_id,
        'parent_id', c.parent_id,
        'content', c.content,
        'created_at', c.created_at,
        'updated_at', c.updated_at,
        'like_count', c.like_count,
        'liked', c.liked
      )
      ORDER BY c.created_at, c.id
    ),
    '[]'::jsonb
  )
  INTO v_comments
  FROM (
    SELECT
      cm.*,
      (
        SELECT count(*)::integer
        FROM public.group_promotion_share_comment_likes cl
        WHERE cl.comment_id = cm.id
          AND NOT parapost_groups_private.blocked(
            v_uid, cl.user_id
          )
      ) AS like_count,
      EXISTS (
        SELECT 1
        FROM public.group_promotion_share_comment_likes cl
        WHERE cl.comment_id = cm.id
          AND cl.user_id = v_uid
      ) AS liked
    FROM public.group_promotion_share_comments cm
    WHERE cm.share_id = p_share_id
      AND NOT parapost_groups_private.blocked(
        v_uid, cm.author_id
      )
      AND (
        cm.parent_id IS NULL
        OR EXISTS (
          SELECT 1
          FROM public.group_promotion_share_comments parent
          WHERE parent.id = cm.parent_id
            AND parent.share_id = cm.share_id
            AND parent.parent_id IS NULL
            AND NOT parapost_groups_private.blocked(
              v_uid, parent.author_id
            )
        )
      )
    ORDER BY cm.created_at, cm.id
    LIMIT 150
  ) c;

  RETURN jsonb_build_object(
    'liked', EXISTS (
      SELECT 1
      FROM public.group_promotion_share_likes l
      WHERE l.share_id = p_share_id
        AND l.user_id = v_uid
    ),
    'like_count', (
      SELECT count(*)::integer
      FROM public.group_promotion_share_likes l
      WHERE l.share_id = p_share_id
        AND NOT parapost_groups_private.blocked(
          v_uid, l.user_id
        )
    ),
    'comment_count', (
      SELECT count(*)::integer
      FROM public.group_promotion_share_comments c
      WHERE c.share_id = p_share_id
        AND NOT parapost_groups_private.blocked(
          v_uid, c.author_id
        )
        AND (
          c.parent_id IS NULL
          OR EXISTS (
            SELECT 1
            FROM public.group_promotion_share_comments parent
            WHERE parent.id = c.parent_id
              AND parent.share_id = c.share_id
              AND parent.parent_id IS NULL
              AND NOT parapost_groups_private.blocked(
                v_uid, parent.author_id
              )
          )
        )
    ),
    'comments_limited', (
      SELECT count(*) > 150
      FROM public.group_promotion_share_comments c
      WHERE c.share_id = p_share_id
        AND NOT parapost_groups_private.blocked(
          v_uid, c.author_id
        )
        AND (
          c.parent_id IS NULL
          OR EXISTS (
            SELECT 1
            FROM public.group_promotion_share_comments parent
            WHERE parent.id = c.parent_id
              AND parent.share_id = c.share_id
              AND parent.parent_id IS NULL
              AND NOT parapost_groups_private.blocked(
                v_uid, parent.author_id
              )
          )
        )
    ),
    'comments', v_comments
  );
END;
$function$;

CREATE OR REPLACE FUNCTION parapost_groups_private.queue_avatar_cleanup()
 RETURNS trigger
 LANGUAGE plpgsql
 SECURITY DEFINER
 SET search_path TO ''
AS $function$
BEGIN
  IF OLD.avatar_storage_path IS NULL THEN
    RETURN OLD;
  END IF;

  IF TG_OP = 'DELETE' THEN
    INSERT INTO parapost_groups_private.media_cleanup(
      storage_path
    )
    VALUES (OLD.avatar_storage_path)
    ON CONFLICT DO NOTHING;
  ELSIF OLD.avatar_storage_path
        IS DISTINCT FROM NEW.avatar_storage_path THEN
    INSERT INTO parapost_groups_private.media_cleanup(
      storage_path
    )
    VALUES (OLD.avatar_storage_path)
    ON CONFLICT DO NOTHING;
  END IF;

  RETURN OLD;
END;
$function$;

 SECURITY DEFINER
 SET search_path TO ''

BEGIN
  IF NOT EXISTS (
    SELECT 1 FROM public.groups

  ) THEN
    INSERT INTO parapost_groups_private.media_cleanup(
      storage_path
    )
    VALUES (OLD.storage_path)
    ON CONFLICT DO NOTHING;
  END IF;

  RETURN OLD;
END;
$function$;

CREATE OR REPLACE FUNCTION parapost_groups_private.queue_media_cleanup()
 RETURNS trigger
 LANGUAGE plpgsql
 SECURITY DEFINER
 SET search_path TO ''
AS $function$

 IF TG_TABLE_NAME='group_post_media' THEN
   INSERT INTO parapost_groups_private.media_cleanup(storage_path) VALUES (OLD.storage_path) ON CONFLICT DO NOTHING;
 ELSE

   END IF;
 END IF;
 RETURN OLD;
END; $function$;

CREATE OR REPLACE FUNCTION parapost_groups_private.record_share_cooldown(p_target_type text, p_target_id uuid)
 RETURNS void
 LANGUAGE plpgsql
 SET search_path TO ''
AS $function$
DECLARE
  v_previous timestamptz;
  v_gap interval;
  v_remaining_seconds integer;
BEGIN
  IF auth.uid() IS NULL
     OR p_target_id IS NULL
     OR p_target_type IS NULL
     OR p_target_type NOT IN ('group', 'post') THEN
    RAISE EXCEPTION 'Sharing unavailable'
      USING ERRCODE = '42501';
  END IF;

  v_gap := CASE
    WHEN p_target_type = 'group'
      THEN interval '72 hours'
    ELSE interval '7 days'
  END;

  SELECT last_shared_at INTO v_previous
  FROM parapost_groups_private.group_share_cooldowns
  WHERE user_id = auth.uid()
    AND target_type = p_target_type
    AND target_id = p_target_id
  FOR UPDATE;

  IF v_previous IS NOT NULL
     AND now() < v_previous + v_gap THEN

    v_remaining_seconds := CEIL(
      EXTRACT(EPOCH FROM (v_previous + v_gap - now()))
    )::integer;

    RAISE EXCEPTION 'GROUP_SHARE_COOLDOWN'
      USING
        ERRCODE = 'P0001',
        DETAIL = jsonb_build_object(
          'kind', p_target_type,
          'retry_after_seconds', v_remaining_seconds
        )::text;
  END IF;

  INSERT INTO parapost_groups_private.group_share_cooldowns
    (user_id, target_type, target_id, last_shared_at)
  VALUES (auth.uid(), p_target_type, p_target_id, now())
  ON CONFLICT (user_id, target_type, target_id)
  DO UPDATE SET last_shared_at = EXCLUDED.last_shared_at;
END;
$function$;

CREATE OR REPLACE FUNCTION parapost_groups_private.request_membership(p_group_id uuid)
 RETURNS text
 LANGUAGE plpgsql
 SECURITY DEFINER
 SET search_path TO ''
AS $function$
DECLARE
  v_user_id uuid := auth.uid();
  v_visibility text;
  v_owner_id uuid;
  v_status text;
BEGIN
  IF v_user_id IS NULL THEN
    RAISE EXCEPTION 'Authentication required'
      USING ERRCODE = '28000';
  END IF;

  SELECT g.visibility, g.owner_id
    INTO v_visibility, v_owner_id
  FROM public.groups g
  WHERE g.id = p_group_id
  FOR SHARE;

  IF NOT FOUND THEN
    RAISE EXCEPTION 'Group unavailable';
  END IF;

  IF EXISTS (
    SELECT 1
    FROM public.blocked_users b
    WHERE
      (b.blocker_id = v_user_id
       AND b.blocked_id = v_owner_id)
      OR
      (b.blocked_id = v_user_id
       AND b.blocker_id = v_owner_id)
  ) THEN
    RAISE EXCEPTION 'Membership unavailable';
  END IF;

  INSERT INTO public.group_members (
    group_id,
    user_id,
    role,
    status,
    approved_at
  )
  VALUES (
    p_group_id,
    v_user_id,
    'member',
    CASE
      WHEN v_visibility = 'public'
      THEN 'approved'
      ELSE 'pending'
    END,
    CASE
      WHEN v_visibility = 'public'
      THEN now()
      ELSE NULL
    END
  )
  ON CONFLICT (group_id, user_id)
  DO NOTHING;

  SELECT gm.status
    INTO v_status
  FROM public.group_members gm
  WHERE gm.group_id = p_group_id
    AND gm.user_id = v_user_id;

  IF v_status NOT IN ('approved', 'pending')
     OR v_status IS NULL THEN
    RAISE EXCEPTION 'Membership unavailable';
  END IF;

  RETURN v_status;
END;
$function$;

CREATE OR REPLACE FUNCTION parapost_groups_private.reset_group_share_cooldown_after_delete()
 RETURNS trigger
 LANGUAGE plpgsql
 SECURITY DEFINER
 SET search_path TO ''
AS $function$
DECLARE
  v_latest timestamptz;
BEGIN
  -- Only Group promotions reset their cooldown.
  -- Individual Group posts retain the seven-day rule.
  IF OLD.post_id IS NOT NULL THEN
    RETURN OLD;
  END IF;

  -- Check whether the member has another share
  -- of this same Group still active.
  SELECT MAX(created_at)
  INTO v_latest
  FROM public.group_promotional_shares
  WHERE user_id = OLD.user_id
    AND group_id = OLD.group_id
    AND post_id IS NULL;

  IF v_latest IS NULL THEN
    DELETE FROM parapost_groups_private.group_share_cooldowns
    WHERE user_id = OLD.user_id
      AND target_type = 'group'
      AND target_id = OLD.group_id;
  ELSE
    INSERT INTO parapost_groups_private.group_share_cooldowns
      (user_id, target_type, target_id, last_shared_at)
    VALUES
      (OLD.user_id, 'group', OLD.group_id, v_latest)
    ON CONFLICT (user_id, target_type, target_id)
    DO UPDATE SET
      last_shared_at = EXCLUDED.last_shared_at;
  END IF;

  RETURN OLD;
END;
$function$;

CREATE OR REPLACE FUNCTION parapost_groups_private.review_membership(p_group_id uuid, p_target_user_id uuid, p_decision text)
 RETURNS text
 LANGUAGE plpgsql
 SECURITY DEFINER
 SET search_path TO ''
AS $function$
DECLARE
  v_reviewer uuid := auth.uid();
  v_owner_id uuid;
  v_reviewer_role text;
  v_final_status text;
BEGIN
  IF v_reviewer IS NULL THEN
    RAISE EXCEPTION 'Authentication required';
  END IF;

  IF p_group_id IS NULL
     OR p_target_user_id IS NULL
     OR p_decision IS NULL
     OR p_decision NOT IN ('approved', 'rejected') THEN
    RAISE EXCEPTION 'Invalid membership review';
  END IF;

  SELECT g.owner_id
    INTO v_owner_id
  FROM public.groups g
  WHERE g.id = p_group_id
  FOR SHARE;

  IF NOT FOUND THEN
    RAISE EXCEPTION 'Group unavailable';
  END IF;

  SELECT gm.role
    INTO v_reviewer_role
  FROM public.group_members gm
  WHERE gm.group_id = p_group_id
    AND gm.user_id = v_reviewer
    AND gm.status = 'approved'
    AND gm.role IN ('owner', 'admin')
  FOR SHARE;

  IF NOT FOUND THEN
    RAISE EXCEPTION 'Not authorized to review membership';
  END IF;

  IF p_decision = 'approved' AND EXISTS (
    SELECT 1
    FROM public.blocked_users b
    WHERE
      (
        b.blocker_id = p_target_user_id
        AND b.blocked_id IN (v_owner_id, v_reviewer)
      )
      OR
      (
        b.blocked_id = p_target_user_id
        AND b.blocker_id IN (v_owner_id, v_reviewer)
      )
  ) THEN
    RAISE EXCEPTION 'Membership unavailable';
  END IF;

  UPDATE public.group_members gm
  SET
    status = p_decision,
    approved_at = CASE
      WHEN p_decision = 'approved'
      THEN now()
      ELSE NULL
    END
  WHERE gm.group_id = p_group_id
    AND gm.user_id = p_target_user_id
    AND gm.role = 'member'
    AND gm.status = 'pending'
  RETURNING gm.status INTO v_final_status;

  IF NOT FOUND THEN
    RAISE EXCEPTION 'Pending membership unavailable';
  END IF;

  RETURN v_final_status;
END;
$function$;

CREATE OR REPLACE FUNCTION parapost_groups_private.storage_allowed(path text, operation text)
 RETURNS boolean
 LANGUAGE sql
 STABLE SECURITY DEFINER
 SET search_path TO ''
AS $function$

SELECT parapost_groups_private.enabled() AND (
  CASE
    WHEN operation = 'read' THEN
      EXISTS (
        SELECT 1
        FROM public.groups g
        WHERE g.cover_storage_path = path
          AND parapost_groups_private.allowed(
            g.id, 'directory'
          )
      )
      OR EXISTS (
        SELECT 1
        FROM public.groups g
        WHERE g.avatar_storage_path = path
          AND parapost_groups_private.allowed(
            g.id, 'directory'
          )
      )
      OR EXISTS (
        SELECT 1
        FROM public.group_post_media m
        JOIN public.group_posts p
          ON p.id = m.post_id
         AND p.group_id = m.group_id
        WHERE (
          m.storage_path = path
          OR (
            m.poster_path = path
            AND m.media_type = 'video'
            AND m.poster_path = m.storage_path || '.poster.jpg'
          )
        )
          AND NOT p.is_deleted
          AND parapost_groups_private.allowed(
            p.group_id, 'read'
          )

          -- Blocked-user privacy protection
          AND parapost_groups_private.author_visible_to_viewer(
            p.author_id
          )

          AND (
            p.published
            OR (
              p.author_id = auth.uid()
              AND parapost_groups_private.allowed(
                p.group_id, 'member'
              )
            )
          )
      )

    WHEN operation = 'insert' THEN
      EXISTS (
        SELECT 1
        FROM parapost_groups_private.cover_uploads c
        WHERE c.storage_path = path
          AND c.uploader_id = auth.uid()
          AND c.created_at > now() - interval '1 hour'
          AND parapost_groups_private.allowed(
            c.group_id, 'admin'
          )
      )
      OR EXISTS (
        SELECT 1
        FROM parapost_groups_private.avatar_uploads a
        JOIN public.groups g ON g.id = a.group_id
        WHERE a.storage_path = path
          AND a.uploader_id = auth.uid()
          AND g.owner_id = auth.uid()
          AND a.created_at > now() - interval '1 hour'
      )
      OR EXISTS (
        SELECT 1
        FROM public.group_post_media m
        JOIN public.group_posts p
          ON p.id = m.post_id
         AND p.group_id = m.group_id
        WHERE (
          m.storage_path = path
          OR (
            m.poster_path = path
            AND m.media_type = 'video'
            AND m.poster_path = m.storage_path || '.poster.jpg'
          )
        )
          AND m.uploader_id = auth.uid()
          AND p.author_id = auth.uid()
          AND NOT p.published
          AND NOT p.is_deleted
          AND p.created_at > now() - interval '1 hour'
          AND parapost_groups_private.allowed(
            p.group_id, 'member'
          )
      )
    ELSE false
  END
);

$function$;

CREATE OR REPLACE FUNCTION public.creatorverse_cancel_page_media(p_page_id uuid, p_media_id uuid, p_owner_id uuid)
 RETURNS boolean
 LANGUAGE plpgsql
 SET search_path TO ''
AS $function$
DECLARE
  v_media record;
BEGIN
  IF current_user <> 'service_role'
     OR p_page_id IS NULL
     OR p_media_id IS NULL
     OR p_owner_id IS NULL
  THEN
    RAISE EXCEPTION 'Not authorized'
      USING ERRCODE = '42501';
  END IF;

  -- Keep lock order consistent with page ownership.
  PERFORM 1
  FROM public.creatorverse_pages
  WHERE id = p_page_id
    AND owner_id = p_owner_id
    AND status IN ('draft', 'active')
  FOR UPDATE;

  IF NOT FOUND THEN
    RETURN false;
  END IF;

  SELECT id, purpose, verification
  INTO v_media
  FROM public.creatorverse_media
  WHERE id = p_media_id
    AND page_id = p_page_id
    AND purpose IN ('logo', 'cover')
    AND post_id IS NULL
    AND media_type = 'image'
  FOR UPDATE;

  IF NOT FOUND OR v_media.verification <> 'pending' THEN
    RETURN false;
  END IF;

  -- Prevent any further upload through the reservation.
  UPDATE creatorverse_private.upload_reservations
  SET expires_at = pg_catalog.statement_timestamp()
  WHERE page_id = p_page_id
    AND media_id = p_media_id
    AND owner_id = p_owner_id
    AND consumed_at IS NULL;

  UPDATE public.creatorverse_media
  SET verification = 'rejected'
  WHERE id = p_media_id
    AND page_id = p_page_id
    AND verification = 'pending';

  IF NOT FOUND THEN
    RETURN false;
  END IF;

  -- Never remove storage.objects here.
  -- Cleanup must account for late uploads and retries.
  RETURN true;
END;
$function$;

CREATE OR REPLACE FUNCTION public.creatorverse_claim_media_cleanup_batch(p_limit integer DEFAULT 5)
 RETURNS TABLE(job_id uuid, claim_token uuid, page_id uuid, post_id uuid, media_id uuid, storage_path text)
 LANGUAGE plpgsql
 SET search_path TO ''
AS $function$
DECLARE
  candidate record;
  media_record record;
  job_record record;
  token uuid;
  claims integer := 0;
BEGIN
  IF current_user <> 'service_role' THEN
    RAISE EXCEPTION 'Not authorized'
      USING ERRCODE = '42501';
  END IF;

  IF p_limit IS NULL OR p_limit NOT BETWEEN 1 AND 5
  THEN
    RAISE EXCEPTION 'Invalid cleanup limit';
  END IF;

  FOR candidate IN
    SELECT
      q.job_id AS candidate_job,
      q.media_id AS candidate_media,
      q.page_id AS candidate_page,
      q.post_id AS candidate_post
    FROM creatorverse_private.media_cleanup_jobs q
    WHERE (
      q.status = 'queued'
      AND q.eligible_after <=
        pg_catalog.statement_timestamp()
    ) OR (
      q.status = 'claimed'
      AND q.lease_expires_at <
        pg_catalog.statement_timestamp()
    )
    ORDER BY q.eligible_after, q.job_id
    LIMIT p_limit * 10
  LOOP
    PERFORM 1
    FROM public.creatorverse_pages p
    WHERE p.id = candidate.candidate_page
    FOR UPDATE SKIP LOCKED;

    IF NOT FOUND THEN CONTINUE; END IF;

    PERFORM 1
    FROM public.creatorverse_posts p
    WHERE p.id = candidate.candidate_post
      AND p.page_id = candidate.candidate_page
      AND p.status IN ('draft', 'published')
    FOR UPDATE SKIP LOCKED;

    IF NOT FOUND THEN CONTINUE; END IF;

    SELECT m.*
    INTO media_record
    FROM public.creatorverse_media m
    WHERE m.id = candidate.candidate_media
      AND m.page_id = candidate.candidate_page
      AND m.post_id = candidate.candidate_post
      AND m.purpose = 'post'
      AND m.verification = 'rejected'
    FOR UPDATE SKIP LOCKED;

    IF NOT FOUND THEN CONTINUE; END IF;

    SELECT q.*
    INTO job_record
    FROM creatorverse_private.media_cleanup_jobs q
    WHERE q.job_id = candidate.candidate_job
      AND q.media_id = candidate.candidate_media
    FOR UPDATE SKIP LOCKED;

    IF NOT FOUND THEN CONTINUE; END IF;

    IF media_record.storage_path IS DISTINCT FROM
       job_record.storage_path
       OR NOT (
         (
           job_record.status = 'queued'
           AND job_record.eligible_after <=
             pg_catalog.statement_timestamp()
         )
         OR (
           job_record.status = 'claimed'
           AND job_record.lease_expires_at <
             pg_catalog.statement_timestamp()
         )
       )
    THEN
      CONTINUE;
    END IF;

    token := pg_catalog.gen_random_uuid();

    UPDATE creatorverse_private.media_cleanup_jobs q
    SET
      status = 'claimed',
      claim_token = token,
      lease_expires_at =
        pg_catalog.statement_timestamp()
          + interval '10 minutes',
      attempts = q.attempts + 1,
      last_error = NULL
    WHERE q.job_id = candidate.candidate_job;

    job_id := candidate.candidate_job;
    claim_token := token;
    page_id := candidate.candidate_page;
    post_id := candidate.candidate_post;
    media_id := candidate.candidate_media;
    storage_path := job_record.storage_path;

    RETURN NEXT;

    claims := claims + 1;
    EXIT WHEN claims >= p_limit;
  END LOOP;
END;
$function$;

CREATE OR REPLACE FUNCTION public.creatorverse_complete_media_cleanup(p_job_id uuid, p_claim_token uuid, p_success boolean, p_error text DEFAULT NULL::text)
 RETURNS boolean
 LANGUAGE plpgsql
 SET search_path TO ''
AS $function$
DECLARE
  acknowledged boolean;
BEGIN
  IF current_user <> 'service_role' THEN
    RETURN false;
  END IF;

  IF p_success IS NULL THEN
    RETURN false;
  END IF;

  UPDATE creatorverse_private.media_cleanup_jobs q
  SET
    successful_removals = CASE
      WHEN p_success
        THEN q.successful_removals + 1
      ELSE q.successful_removals
    END,

    status = CASE
      WHEN p_success AND q.successful_removals >= 1
        THEN 'completed'
      ELSE 'queued'
    END,

    claim_token = NULL,
    lease_expires_at = NULL,

    completed_at = CASE
      WHEN p_success AND q.successful_removals >= 1
        THEN pg_catalog.statement_timestamp()
      ELSE NULL
    END,

    eligible_after = CASE
      WHEN p_success AND q.successful_removals = 0
        THEN pg_catalog.statement_timestamp()
          + interval '24 hours'
      WHEN NOT p_success
        THEN pg_catalog.statement_timestamp()
          + interval '10 minutes'
      ELSE q.eligible_after
    END,

    last_error = CASE
      WHEN p_success THEN NULL
      ELSE pg_catalog.left(
        coalesce(p_error, 'cleanup_failed'),
        160
      )
    END
  WHERE q.job_id = p_job_id
    AND q.claim_token = p_claim_token
    AND q.status = 'claimed'
    AND q.lease_expires_at >
      pg_catalog.statement_timestamp()
  RETURNING true INTO acknowledged;

  RETURN coalesce(acknowledged, false);
END;
$function$;

CREATE OR REPLACE FUNCTION public.creatorverse_create_post_draft(p_page_id uuid, p_content text)
 RETURNS uuid
 LANGUAGE plpgsql
 SECURITY DEFINER
 SET search_path TO ''
AS $function$
DECLARE
  v_user uuid;
  v_count integer;
BEGIN
  v_user := auth.uid();

  IF v_user IS NULL
     OR NOT creatorverse_private.write_enabled()
     OR NOT creatorverse_private.owns(p_page_id)
  THEN
    RAISE EXCEPTION 'Posting unavailable'
      USING ERRCODE = '42501';
  END IF;

  -- Serialize posting requests for this account.
  PERFORM pg_catalog.pg_advisory_xact_lock(
    pg_catalog.hashtextextended(
      'creatorverse_post:' || v_user::text,
      0
    )
  );

  SELECT count(*)
  INTO v_count
  FROM public.creatorverse_posts
  WHERE author_id = v_user
    AND created_at >= pg_catalog.now() - interval '1 hour';

  IF v_count >= 10 THEN
    RAISE EXCEPTION 'Posting limit reached. Try again later.'
      USING ERRCODE = 'P0001';
  END IF;

  RETURN creatorverse_private.create_text_draft(
    p_page_id,
    p_content
  );
END;
$function$;

CREATE OR REPLACE FUNCTION public.creatorverse_delete_post(p_page_id uuid, p_post_id uuid)
 RETURNS uuid
 LANGUAGE plpgsql
 SECURITY DEFINER
 SET search_path TO ''
AS $function$
DECLARE

  v_post uuid;

     OR NOT creatorverse_private.read_page(p_page_id)
  THEN
    RAISE EXCEPTION 'Post management unavailable'
      USING ERRCODE = '42501';

CREATE OR REPLACE FUNCTION public.creatorverse_delete_share(p_share_id uuid)
 RETURNS boolean
 LANGUAGE plpgsql
 SECURITY DEFINER
 SET search_path TO ''
AS $function$
DECLARE
  v_user uuid := auth.uid();
BEGIN
  IF v_user IS NULL THEN
    RAISE EXCEPTION 'Authentication required'
      USING ERRCODE = '28000';
  END IF;

  IF NOT creatorverse_private.eligible() THEN
    RAISE EXCEPTION 'Account not eligible'
      USING ERRCODE = '42501';
  END IF;

  IF p_share_id IS NULL THEN
    RETURN false;

  -- Cannot delete the original CreatorVerse post.

  WHERE id = p_share_id
    AND sharer_id = v_user;

  RETURN FOUND;
END;
$function$;

CREATE OR REPLACE FUNCTION public.creatorverse_edit_post(p_page_id uuid, p_post_id uuid, p_content text)
 RETURNS uuid
 LANGUAGE plpgsql
 SECURITY DEFINER
 SET search_path TO ''
AS $function$
DECLARE
  v_user uuid := auth.uid();
  v_post uuid;
  v_content text := btrim(p_content);
BEGIN
  IF v_user IS NULL
     OR NOT creatorverse_private.write_enabled()
     OR NOT creatorverse_private.owns(p_page_id)
     OR NOT creatorverse_private.read_page(p_page_id)
  THEN
    RAISE EXCEPTION 'Post management unavailable'
      USING ERRCODE = '42501';
  END IF;

  PERFORM 1
  FROM public.creatorverse_pages
  WHERE id = p_page_id
  FOR UPDATE;

  SELECT id INTO v_post
  FROM public.creatorverse_posts
  WHERE id = p_post_id
    AND page_id = p_page_id
    AND author_id = v_user
    AND status = 'published'
  FOR UPDATE;

  IF v_post IS NULL THEN
    RAISE EXCEPTION 'Post unavailable'
      USING ERRCODE = '42501';
  END IF;

  IF v_content IS NULL
     OR char_length(v_content) > 5000
     OR translate(v_content, E'\n\r\t', '') ~ '[[:cntrl:]]'
  THEN
    RAISE EXCEPTION 'Invalid post content'
      USING ERRCODE = '23514';
  END IF;

  IF v_content = '' AND NOT EXISTS (
    SELECT 1
    FROM public.creatorverse_media
    WHERE page_id = p_page_id
      AND post_id = p_post_id
      AND purpose = 'post'
      AND verification = 'verified'
  ) THEN
    RAISE EXCEPTION 'Post cannot be empty'
      USING ERRCODE = '23514';
  END IF;

  UPDATE public.creatorverse_posts
  SET content = v_content,
      updated_at = now()
  WHERE id = v_post
  RETURNING id INTO v_post;

  RETURN v_post;
END;
$function$;

CREATE OR REPLACE FUNCTION public.creatorverse_expire_pending_media_batch(p_limit integer DEFAULT 20)
 RETURNS integer
 LANGUAGE plpgsql
 SET search_path TO ''
AS $function$
DECLARE
  candidate record;
  expired_count integer := 0;
BEGIN
  IF current_user <> 'service_role' THEN
    RAISE EXCEPTION 'Not authorized'
      USING ERRCODE = '42501';
  END IF;

  IF p_limit IS NULL OR p_limit NOT BETWEEN 1 AND 20
  THEN
    RAISE EXCEPTION 'Invalid expiration limit';
  END IF;

  FOR candidate IN
    SELECT
      m.id AS media_id,
      m.page_id,
      m.post_id
    FROM creatorverse_private.upload_reservations r
    JOIN public.creatorverse_media m
      ON m.id = r.media_id
    JOIN public.creatorverse_posts p
      ON p.id = m.post_id
    WHERE r.expires_at <=
      pg_catalog.statement_timestamp()
      AND r.consumed_at IS NULL
      AND m.verification = 'pending'
      AND m.purpose = 'post'
      AND p.status = 'draft'
    ORDER BY r.expires_at, m.id
    LIMIT p_limit * 8
  LOOP
    PERFORM 1
    FROM public.creatorverse_pages
    WHERE id = candidate.page_id
    FOR UPDATE SKIP LOCKED;

    IF NOT FOUND THEN CONTINUE; END IF;

    PERFORM 1
    FROM public.creatorverse_posts
    WHERE id = candidate.post_id
      AND page_id = candidate.page_id
      AND status = 'draft'
    FOR UPDATE SKIP LOCKED;

    IF NOT FOUND THEN CONTINUE; END IF;

    PERFORM 1
    FROM public.creatorverse_media
    WHERE id = candidate.media_id
      AND verification = 'pending'
    FOR UPDATE SKIP LOCKED;

    IF NOT FOUND THEN CONTINUE; END IF;

    PERFORM 1
    FROM creatorverse_private.upload_reservations
    WHERE media_id = candidate.media_id
      AND expires_at <=
        pg_catalog.statement_timestamp()
      AND consumed_at IS NULL
    FOR UPDATE SKIP LOCKED;

    IF NOT FOUND THEN CONTINUE; END IF;

    UPDATE public.creatorverse_media
    SET verification = 'rejected'
    WHERE id = candidate.media_id
      AND verification = 'pending';

    IF FOUND THEN
      expired_count := expired_count + 1;
      EXIT WHEN expired_count >= p_limit;
    END IF;
  END LOOP;

  RETURN expired_count;
END;
$function$;

CREATE OR REPLACE FUNCTION public.creatorverse_finalize_page_media(p_media_id uuid, p_verified_bytes bigint, p_detected_mime_type text)
 RETURNS boolean
 LANGUAGE plpgsql
 SET search_path TO ''
AS $function$
DECLARE
  v_media record;
  v_owner uuid;
  v_page_status text;
  v_reservation uuid;
  v_extension text;
BEGIN
  IF current_user <> 'service_role' THEN
    RAISE EXCEPTION 'Not authorized'
      USING ERRCODE = '42501';
  END IF;

  -- Read the page identifier without locking media.
  -- It is immutable because of the existing foreign key.
  SELECT page_id
  INTO v_media
  FROM public.creatorverse_media
  WHERE id = p_media_id
    AND purpose IN ('logo', 'cover')
    AND post_id IS NULL
    AND media_type = 'image';

  IF NOT FOUND THEN
    RAISE EXCEPTION 'Pending page image not found';
  END IF;

  -- Acquire locks consistently: page, media, reservation.
  SELECT owner_id, status
  INTO v_owner, v_page_status
  FROM public.creatorverse_pages
  WHERE id = v_media.page_id
  FOR UPDATE;

  IF NOT FOUND
     OR v_page_status NOT IN ('draft', 'active')
  THEN
    RAISE EXCEPTION 'Page unavailable';
  END IF;

  SELECT page_id, purpose, mime_type,
         storage_path, verification
  INTO v_media
  FROM public.creatorverse_media
  WHERE id = p_media_id
    AND page_id = v_media.page_id
    AND purpose IN ('logo', 'cover')
    AND post_id IS NULL
    AND media_type = 'image'
  FOR UPDATE;

  IF NOT FOUND OR v_media.verification <> 'pending' THEN
    RAISE EXCEPTION 'Pending page image not found';
  END IF;

  SELECT id
  INTO v_reservation
  FROM creatorverse_private.upload_reservations
  WHERE media_id = p_media_id
    AND page_id = v_media.page_id
    AND owner_id = v_owner
    AND consumed_at IS NULL
    AND expires_at > pg_catalog.statement_timestamp()
  FOR UPDATE;

  IF NOT FOUND THEN
    RAISE EXCEPTION 'Reservation expired or consumed';
  END IF;

  v_extension := CASE v_media.mime_type
    WHEN 'image/jpeg' THEN 'jpg'
    WHEN 'image/png' THEN 'png'
    WHEN 'image/webp' THEN 'webp'
    ELSE NULL
  END;

  IF v_extension IS NULL
     OR p_detected_mime_type IS DISTINCT FROM v_media.mime_type
     OR p_verified_bytes IS NULL
     OR p_verified_bytes NOT BETWEEN 1 AND 8388608
  THEN
    RAISE EXCEPTION 'Invalid image verification';
  END IF;

  IF v_media.storage_path IS DISTINCT FROM (
    v_media.page_id::text || '/' ||
    v_media.purpose || '/' ||
    p_media_id::text || '.' || v_extension
  ) THEN
    RAISE EXCEPTION 'Invalid image storage path';
  END IF;

  IF NOT EXISTS (
    SELECT 1
    FROM storage.objects
    WHERE bucket_id = 'creatorverse-media'
      AND name = v_media.storage_path
  ) THEN
    RAISE EXCEPTION 'Uploaded image missing';
  END IF;

  UPDATE creatorverse_private.upload_reservations
  SET consumed_at = pg_catalog.statement_timestamp()
  WHERE id = v_reservation
    AND consumed_at IS NULL;

  IF NOT FOUND THEN
    RAISE EXCEPTION 'Reservation consumption failed';
  END IF;

  UPDATE public.creatorverse_media
  SET verification = 'verified',
      byte_size = p_verified_bytes
  WHERE id = p_media_id
    AND verification = 'pending';

  IF NOT FOUND THEN
    RAISE EXCEPTION 'Image finalization failed';
  END IF;

  IF v_media.purpose = 'logo' THEN
    UPDATE public.creatorverse_pages
    SET logo_media_id = p_media_id
    WHERE id = v_media.page_id
      AND owner_id = v_owner
      AND status IN ('draft', 'active');
  ELSE
    UPDATE public.creatorverse_pages
    SET cover_media_id = p_media_id
    WHERE id = v_media.page_id
      AND owner_id = v_owner
      AND status IN ('draft', 'active');
  END IF;

  IF NOT FOUND THEN
    RAISE EXCEPTION 'Image attachment failed';
  END IF;

  RETURN true;
END;
$function$;

CREATE OR REPLACE FUNCTION public.creatorverse_finalize_post_media(p_media_id uuid, p_verified_bytes bigint, p_detected_mime_type text, p_verified_duration_seconds numeric)
 RETURNS boolean
 LANGUAGE plpgsql
 SET search_path TO ''
AS $function$
DECLARE
  v_page_id uuid;
  v_post_id uuid;
  v_owner_id uuid;
  v_page_status text;
  v_author_id uuid;
  v_post_status text;
  v_media record;
  v_reservation_id uuid;
  v_extension text;
BEGIN
  IF current_user <> 'service_role' THEN
    RAISE EXCEPTION 'Not authorized'
      USING ERRCODE = '42501';
  END IF;

  SELECT page_id, post_id
  INTO v_page_id, v_post_id
  FROM public.creatorverse_media
  WHERE id = p_media_id
    AND purpose = 'post';

  IF NOT FOUND OR v_post_id IS NULL THEN
    RAISE EXCEPTION 'Media not found';
  END IF;

  -- Lock page, post, media, reservation.
  SELECT owner_id, status
  INTO v_owner_id, v_page_status
  FROM public.creatorverse_pages
  WHERE id = v_page_id
  FOR UPDATE;

  IF NOT FOUND
     OR v_page_status NOT IN ('draft', 'active')
  THEN
    RAISE EXCEPTION 'Page unavailable';
  END IF;

  SELECT author_id, status
  INTO v_author_id, v_post_status
  FROM public.creatorverse_posts
  WHERE id = v_post_id
    AND page_id = v_page_id
  FOR UPDATE;

  IF NOT FOUND
     OR v_post_status <> 'draft'
     OR v_author_id IS DISTINCT FROM v_owner_id
  THEN
    RAISE EXCEPTION 'Owned draft unavailable';
  END IF;

  SELECT
    storage_path, media_type, mime_type, verification
  INTO v_media
  FROM public.creatorverse_media
  WHERE id = p_media_id
    AND page_id = v_page_id
    AND post_id = v_post_id
    AND purpose = 'post'
  FOR UPDATE;

  IF NOT FOUND OR v_media.verification <> 'pending' THEN
    RAISE EXCEPTION 'Media is not pending';
  END IF;

  SELECT id
  INTO v_reservation_id
  FROM creatorverse_private.upload_reservations
  WHERE media_id = p_media_id
    AND page_id = v_page_id
    AND owner_id = v_owner_id
    AND consumed_at IS NULL
    AND expires_at > pg_catalog.statement_timestamp()
  ORDER BY expires_at DESC
  LIMIT 1
  FOR UPDATE;

  IF NOT FOUND THEN
    RAISE EXCEPTION 'Reservation expired or consumed';
  END IF;

  v_extension := CASE v_media.mime_type
    WHEN 'image/jpeg' THEN 'jpg'
    WHEN 'image/png' THEN 'png'
    WHEN 'image/webp' THEN 'webp'
    WHEN 'video/mp4' THEN 'mp4'
    WHEN 'video/webm' THEN 'webm'
    ELSE NULL
  END;

  IF v_extension IS NULL
     OR v_media.storage_path IS DISTINCT FROM (
       v_page_id::text || '/' ||
       v_post_id::text || '/' ||
       p_media_id::text || '.' || v_extension
     )
  THEN
    RAISE EXCEPTION 'Invalid storage path';
  END IF;

  IF NOT EXISTS (
    SELECT 1
    FROM storage.objects
    WHERE bucket_id = 'creatorverse-media'
      AND name = v_media.storage_path
  ) THEN
    RAISE EXCEPTION 'Uploaded object missing';
  END IF;

  IF p_detected_mime_type IS DISTINCT FROM v_media.mime_type
  THEN
    RAISE EXCEPTION 'Verified MIME mismatch';
  END IF;

  IF v_media.media_type = 'image' THEN
    IF v_media.mime_type NOT IN (
        'image/jpeg', 'image/png', 'image/webp'
      )
      OR p_verified_bytes IS NULL
      OR p_verified_bytes NOT BETWEEN 1 AND 8388608
      OR p_verified_duration_seconds IS NOT NULL
    THEN
      RAISE EXCEPTION 'Invalid image verification';
    END IF;

  ELSIF v_media.media_type = 'video' THEN
    IF v_media.mime_type NOT IN ('video/mp4', 'video/webm')
      OR p_verified_bytes IS NULL
      OR p_verified_bytes NOT BETWEEN 1 AND 33554432
      OR p_verified_duration_seconds IS NULL
      OR p_verified_duration_seconds <= 0
      OR p_verified_duration_seconds > 120
    THEN
      RAISE EXCEPTION 'Invalid video verification';
    END IF;

  ELSE
    RAISE EXCEPTION 'Unsupported media type';
  END IF;

  UPDATE creatorverse_private.upload_reservations
  SET consumed_at = pg_catalog.statement_timestamp()
  WHERE id = v_reservation_id
    AND consumed_at IS NULL;

  IF NOT FOUND THEN
    RAISE EXCEPTION 'Reservation consumption failed';
  END IF;

  UPDATE public.creatorverse_media
  SET
    byte_size = p_verified_bytes,
    duration_seconds = p_verified_duration_seconds,
    verification = 'verified'
  WHERE id = p_media_id
    AND verification = 'pending';

  IF NOT FOUND THEN
    RAISE EXCEPTION 'Media finalization failed';
  END IF;

  RETURN true;
END;
$function$;

CREATE OR REPLACE FUNCTION public.creatorverse_post_engagement(p_page_id uuid, p_post_id uuid)
 RETURNS jsonb
 LANGUAGE plpgsql
 SECURITY DEFINER
 SET search_path TO ''
AS $function$
DECLARE
  v_data jsonb;
BEGIN
  IF NOT creatorverse_private.read_post(p_post_id)
    OR NOT EXISTS (
      SELECT 1
      FROM public.creatorverse_posts AS post
      JOIN public.creatorverse_pages AS page
        ON page.id = post.page_id
      WHERE post.id = p_post_id
        AND post.page_id = p_page_id
        AND post.status = 'published'
        AND page.status = 'active'
    ) THEN
    RETURN NULL;
  END IF;

  SELECT pg_catalog.jsonb_build_object(
    'comments_enabled', (
      SELECT coalesce(post.comments_enabled, false)
      FROM public.creatorverse_posts AS post
      WHERE post.id = p_post_id
        AND post.page_id = p_page_id
        AND post.status = 'published'
    ),
    'like_count', (
      SELECT pg_catalog.count(*)
      FROM public.creatorverse_post_likes AS l
      WHERE l.post_id = p_post_id
        AND NOT creatorverse_private.blocked(l.user_id)
    ),
    'comment_count', (
      SELECT pg_catalog.count(*)
      FROM public.creatorverse_comments AS c
      WHERE c.post_id = p_post_id
        AND NOT c.is_deleted
        AND creatorverse_private.read_comment(c.id)
    ),
    'share_count', (
      SELECT pg_catalog.count(*)
      FROM public.creatorverse_shares AS s
      WHERE s.post_id = p_post_id
        AND NOT creatorverse_private.blocked(s.sharer_id)
    ),
    'liked_by_me', (
      SELECT EXISTS (
        SELECT 1
        FROM public.creatorverse_post_likes AS l
        WHERE l.post_id = p_post_id
          AND l.user_id = auth.uid()
      )
    ),
    'comments', (
      SELECT coalesce(
        pg_catalog.jsonb_agg(
          pg_catalog.jsonb_build_object(
            'id', c.id,
            'post_id', c.post_id,
            'author_id', c.author_id,
            'parent_id', c.parent_id,
            'content', c.content,
            'created_at', c.created_at,
            'updated_at', c.updated_at,
            'like_count', (
              SELECT pg_catalog.count(*)
              FROM public.creatorverse_comment_likes AS cl
              WHERE cl.comment_id = c.id
                AND NOT creatorverse_private.blocked(cl.user_id)
            ),
            'liked_by_me', (
              SELECT EXISTS (
                SELECT 1
                FROM public.creatorverse_comment_likes AS cl
                WHERE cl.comment_id = c.id
                  AND cl.user_id = auth.uid()
              )
            )
          )
          ORDER BY c.created_at
        ),
        '[]'::jsonb
      )
      FROM (
        SELECT
          id,
          post_id,
          author_id,
          parent_id,
          content,
          created_at,
          updated_at
        FROM public.creatorverse_comments
        WHERE post_id = p_post_id
          AND NOT is_deleted
          AND creatorverse_private.read_comment(id)
        ORDER BY created_at
        LIMIT 100
      ) AS c
    )
  )
  INTO v_data;

  RETURN v_data;
END;
$function$;

CREATE OR REPLACE FUNCTION public.creatorverse_publish_post(p_page_id uuid, p_post_id uuid)
 RETURNS uuid
 LANGUAGE plpgsql
 SECURITY DEFINER
 SET search_path TO ''
AS $function$
DECLARE
  v_user uuid;
  v_count integer;
BEGIN
  v_user := auth.uid();

  IF v_user IS NULL
     OR NOT creatorverse_private.write_enabled()
     OR NOT creatorverse_private.owns(p_page_id)
  THEN
    RAISE EXCEPTION 'Publishing unavailable'
      USING ERRCODE = '42501';
  END IF;

  -- Use the same account-level lock.
  PERFORM pg_catalog.pg_advisory_xact_lock(
    pg_catalog.hashtextextended(
      'creatorverse_post:' || v_user::text,
      0
    )
  );

  SELECT count(*)
  INTO v_count
  FROM public.creatorverse_posts
  WHERE author_id = v_user
    AND status = 'published'
    AND updated_at >= pg_catalog.now() - interval '1 hour';

  IF v_count >= 10 THEN
    RAISE EXCEPTION 'Publishing limit reached. Try again later.'
      USING ERRCODE = 'P0001';
  END IF;

  RETURN creatorverse_private.publish_text_draft(
    p_page_id,
    p_post_id
  );
END;
$function$;

CREATE OR REPLACE FUNCTION public.creatorverse_reserve_page_media(p_page_id uuid, p_purpose text, p_mime_type text, p_byte_size bigint)
 RETURNS jsonb
 LANGUAGE plpgsql
 SECURITY DEFINER
 SET search_path TO ''
AS $function$
DECLARE
  v_user uuid := auth.uid();
  v_id uuid;
  v_extension text;
  v_path text;
  v_now timestamptz := pg_catalog.statement_timestamp();
BEGIN
  IF v_user IS NULL
     OR NOT creatorverse_private.write_enabled()
     OR NOT creatorverse_private.owns(p_page_id)
  THEN
    RAISE EXCEPTION 'Page image reservation unavailable'
      USING ERRCODE = '42501';
  END IF;

  -- Lock page to serialize owner reservations.
  PERFORM 1
  FROM public.creatorverse_pages
  WHERE id = p_page_id
    AND owner_id = v_user
    AND status IN ('draft', 'active')
  FOR UPDATE;

  IF NOT FOUND THEN
    RAISE EXCEPTION 'Page unavailable'
      USING ERRCODE = '42501';
  END IF;

  IF p_purpose NOT IN ('logo', 'cover')
     OR p_purpose IS NULL
  THEN
    RAISE EXCEPTION 'Invalid page image purpose'
      USING ERRCODE = '22023';
  END IF;

  v_extension := CASE p_mime_type
    WHEN 'image/jpeg' THEN 'jpg'
    WHEN 'image/png' THEN 'png'
    WHEN 'image/webp' THEN 'webp'
    ELSE NULL
  END;

  IF v_extension IS NULL
     OR p_byte_size IS NULL
     OR p_byte_size NOT BETWEEN 1 AND 8388608
  THEN
    RAISE EXCEPTION 'Invalid image format or size'
      USING ERRCODE = '22023';
  END IF;

  -- Limit pending unexpired reservations for each image purpose.
  IF (
    SELECT count(*)
    FROM public.creatorverse_media m
    JOIN creatorverse_private.upload_reservations r
      ON r.media_id = m.id
     AND r.page_id = m.page_id
    WHERE m.page_id = p_page_id
      AND m.purpose = p_purpose
      AND m.verification = 'pending'
      AND r.owner_id = v_user
      AND r.consumed_at IS NULL
      AND r.expires_at > v_now
  ) >= 3 THEN
    RAISE EXCEPTION 'Too many pending page image uploads'
      USING ERRCODE = '42901';
  END IF;

  -- Bound reservation attempts within one hour.
  IF (
    SELECT count(*)
    FROM public.creatorverse_media
    WHERE page_id = p_page_id
      AND purpose IN ('logo', 'cover')
      AND created_at > v_now - interval '1 hour'
  ) >= 12 THEN
    RAISE EXCEPTION 'Page image reservation limit reached'
      USING ERRCODE = '42901';
  END IF;

  v_id := pg_catalog.gen_random_uuid();

  v_path :=
    p_page_id::text || '/' ||
    p_purpose || '/' ||
    v_id::text || '.' || v_extension;

  INSERT INTO public.creatorverse_media (
    id,
    page_id,
    post_id,
    purpose,
    media_type,
    mime_type,
    storage_path,
    verification,
    display_order
  )
  VALUES (
    v_id,
    p_page_id,
    NULL,
    p_purpose,
    'image',
    p_mime_type,
    v_path,
    'pending',
    0
  );

  INSERT INTO creatorverse_private.upload_reservations (
    page_id,
    media_id,
    owner_id,
    expires_at
  )
  VALUES (
    p_page_id,
    v_id,
    v_user,
    v_now + interval '30 minutes'
  );

  RETURN pg_catalog.jsonb_build_object(
    'media_id', v_id,
    'storage_path', v_path,
    'verification', 'pending'
  );
END;
$function$;

CREATE OR REPLACE FUNCTION public.creatorverse_reserve_post_media(p_page_id uuid, p_post_id uuid, p_media_type text, p_mime_type text, p_display_order integer)
 RETURNS jsonb
 LANGUAGE plpgsql
 SECURITY DEFINER
 SET search_path TO ''
AS $function$
DECLARE
  v_user uuid := auth.uid();
  v_id uuid;
  v_extension text;
  v_path text;
BEGIN
  IF v_user IS NULL
     OR NOT creatorverse_private.write_enabled()
     OR NOT creatorverse_private.owns(p_page_id)
  THEN
    RAISE EXCEPTION 'Media reservation unavailable'
      USING ERRCODE = '42501';
  END IF;

  PERFORM 1
  FROM public.creatorverse_pages
  WHERE id = p_page_id
    AND owner_id = v_user
    AND status IN ('draft', 'active')
  FOR UPDATE;

  IF NOT FOUND THEN
    RAISE EXCEPTION 'Page unavailable';
  END IF;

  PERFORM 1
  FROM public.creatorverse_posts
  WHERE id = p_post_id
    AND page_id = p_page_id
    AND author_id = v_user
    AND status = 'draft'
  FOR UPDATE;

  IF NOT FOUND THEN
    RAISE EXCEPTION 'Owned draft unavailable';
  END IF;

  IF p_display_order IS NULL
     OR p_display_order NOT BETWEEN 0 AND 3
  THEN
    RAISE EXCEPTION 'Invalid media position';
  END IF;

  v_extension := CASE
    WHEN p_media_type = 'image'
      AND p_mime_type = 'image/jpeg' THEN 'jpg'
    WHEN p_media_type = 'image'
      AND p_mime_type = 'image/png' THEN 'png'
    WHEN p_media_type = 'image'
      AND p_mime_type = 'image/webp' THEN 'webp'
    WHEN p_media_type = 'video'
      AND p_mime_type = 'video/mp4' THEN 'mp4'
    WHEN p_media_type = 'video'
      AND p_mime_type = 'video/webm' THEN 'webm'
    ELSE NULL
  END;

  IF v_extension IS NULL THEN
    RAISE EXCEPTION 'Unsupported media type';
  END IF;

  IF EXISTS (
    SELECT 1
    FROM public.creatorverse_media
    WHERE post_id = p_post_id
      AND display_order = p_display_order
      AND verification IN ('pending', 'verified')
  ) THEN
    RAISE EXCEPTION 'Attachment position occupied';
  END IF;

  IF (
    SELECT count(*)
    FROM public.creatorverse_media
    WHERE post_id = p_post_id
      AND verification IN ('pending', 'verified')
  ) >= 4 THEN
    RAISE EXCEPTION 'Maximum four attachments';
  END IF;

  v_id := pg_catalog.gen_random_uuid();

  v_path :=
    p_page_id::text || '/' ||
    p_post_id::text || '/' ||
    v_id::text || '.' || v_extension;

  INSERT INTO public.creatorverse_media (
    id, page_id, post_id, purpose,
    media_type, mime_type, storage_path,
    verification, display_order
  )
  VALUES (
    v_id, p_page_id, p_post_id, 'post',
    p_media_type, p_mime_type, v_path,
    'pending', p_display_order
  );

  INSERT INTO creatorverse_private.upload_reservations (
    id, page_id, media_id, owner_id, expires_at
  )
  VALUES (
    pg_catalog.gen_random_uuid(),
    p_page_id,
    v_id,
    v_user,
    pg_catalog.statement_timestamp()
      + interval '30 minutes'
  );

  RETURN pg_catalog.jsonb_build_object(
    'media_id', v_id,
    'storage_path', v_path,
    'verification', 'pending'
  );
END;
$function$;

CREATE OR REPLACE FUNCTION public.creatorverse_shared_feed(p_sharer_id uuid DEFAULT NULL::uuid, p_limit integer DEFAULT 60, p_offset integer DEFAULT 0)
 RETURNS TABLE(share_id uuid, post_id uuid, page_id uuid, sharer_id uuid, shared_at timestamp with time zone, caption text, page_name text, original_content text, original_author_id uuid, original_created_at timestamp with time zone)
 LANGUAGE sql
 STABLE SECURITY DEFINER
 SET search_path TO ''
AS $function$
  SELECT
    s.id,
    s.post_id,
    post.page_id,
    s.sharer_id,
    s.created_at,
    s.caption,
    page.name,
    post.content,
    page.owner_id,
    post.created_at
  FROM public.creatorverse_shares AS s
  JOIN public.creatorverse_posts AS post
    ON post.id = s.post_id
  JOIN public.creatorverse_pages AS page
    ON page.id = post.page_id
  WHERE auth.uid() IS NOT NULL
    AND creatorverse_private.enabled()
    AND post.status = 'published'
    AND page.status = 'active'
    AND creatorverse_private.read_post(post.id)
    AND NOT creatorverse_private.blocked(s.sharer_id)
    AND (
      p_sharer_id IS NULL
      OR s.sharer_id = p_sharer_id
    )
  ORDER BY s.created_at DESC, s.id DESC
  LIMIT LEAST(
    GREATEST(COALESCE(p_limit, 60), 1),
    60
  )
  OFFSET LEAST(
    GREATEST(COALESCE(p_offset, 0), 0),
    10000
  );
$function$;

CREATE OR REPLACE FUNCTION public.creatorverse_social_interact(p_page_id uuid, p_post_id uuid, p_action text, p_target_id uuid, p_body text, p_request_id uuid)
 RETURNS uuid
 LANGUAGE plpgsql
 SECURITY DEFINER
 SET search_path TO ''
AS $function$
DECLARE
  v_user uuid := auth.uid();
  v_action text := pg_catalog.lower(
    pg_catalog.btrim(coalesce(p_action, ''))
  );
  v_text text := pg_catalog.btrim(coalesce(p_body, ''));
  v_id uuid;
  v_existing_post uuid;
  v_existing_parent uuid;
  v_existing_content text;
  v_owner uuid;
  v_comments_enabled boolean;
BEGIN

  IF v_user IS NULL THEN
    RAISE EXCEPTION 'Authentication required'
      USING ERRCODE = '28000';
  END IF;

  IF NOT creatorverse_private.write_enabled() THEN
    RAISE EXCEPTION 'CreatorVerse writes are disabled'
      USING ERRCODE = '42501';
  END IF;

  SELECT page.owner_id, post.comments_enabled
    INTO v_owner, v_comments_enabled
  FROM public.creatorverse_posts AS post
  JOIN public.creatorverse_pages AS page
    ON page.id = post.page_id
  WHERE post.id = p_post_id
    AND post.page_id = p_page_id
    AND post.status = 'published'
    AND page.status = 'active';

  IF NOT FOUND
    OR NOT creatorverse_private.read_post(p_post_id)
    OR creatorverse_private.blocked(v_owner) THEN
    RAISE EXCEPTION
      'Published post not found or inaccessible'
      USING ERRCODE = '42501';
  END IF;

  IF v_action NOT IN ('delete_comment', 'edit_comment') AND NOT (
    creatorverse_private.owns(p_page_id)
    OR EXISTS (
      SELECT 1
      FROM public.creatorverse_follows AS f
      WHERE f.page_id = p_page_id
        AND f.user_id = v_user
    )
  ) THEN
    RAISE EXCEPTION 'Follow this page to interact'
      USING ERRCODE = '42501';
  END IF;

  -- LIKES

  IF v_action IN ('like', 'unlike') THEN

    IF p_target_id IS NOT NULL OR v_text <> '' THEN
      RAISE EXCEPTION 'Unexpected like parameters'
        USING ERRCODE = '22023';
    END IF;

    IF v_action = 'like' THEN
      INSERT INTO public.creatorverse_post_likes
        (post_id, user_id, created_at)
      VALUES (p_post_id, v_user, pg_catalog.now())
      ON CONFLICT DO NOTHING;
    ELSE
      DELETE FROM public.creatorverse_post_likes
      WHERE post_id = p_post_id
        AND user_id = v_user;
    END IF;

    RETURN p_post_id;

  -- COMMENTS AND REPLIES

  ELSIF v_action = 'comment' THEN

    IF p_request_id IS NULL THEN
      RAISE EXCEPTION 'Comment request ID required'
        USING ERRCODE = '22023';
    END IF;

    -- Existing comment retry: return the original ID.

    SELECT id, post_id, parent_id, content
      INTO v_id, v_existing_post,
           v_existing_parent, v_existing_content
    FROM public.creatorverse_comments
    WHERE author_id = v_user
      AND request_id = p_request_id;

    IF FOUND THEN

      IF v_existing_post IS DISTINCT FROM p_post_id
        OR v_existing_parent IS DISTINCT FROM p_target_id
        OR (
          v_existing_content IS DISTINCT FROM v_text
          AND NOT EXISTS (
            SELECT 1
            FROM public.creatorverse_comments
            WHERE id = v_id AND is_deleted
          )
        ) THEN
        RAISE EXCEPTION 'Comment request ID collision'
          USING ERRCODE = '22023';
      END IF;

      RETURN v_id;
    END IF;

    IF NOT coalesce(v_comments_enabled, false)
      OR pg_catalog.length(v_text) NOT BETWEEN 1 AND 1200
      OR NOT creatorverse_private.comment_allowed(
        p_post_id, p_target_id
      ) THEN
      RAISE EXCEPTION
        'Comments disabled or invalid reply'
        USING ERRCODE = '22023';
    END IF;

    IF (
      SELECT pg_catalog.count(*)
      FROM public.creatorverse_comments
      WHERE author_id = v_user
        AND created_at >
          pg_catalog.now() - interval '1 minute'
    ) >= 10 THEN
      RAISE EXCEPTION 'Comment rate limit reached'
        USING ERRCODE = '42501';
    END IF;

    v_id := pg_catalog.gen_random_uuid();

    INSERT INTO public.creatorverse_comments (
      id, post_id, author_id, parent_id,
      content, is_deleted, created_at,
      updated_at, request_id
    )
    VALUES (
      v_id, p_post_id, v_user, p_target_id,
      v_text, false, pg_catalog.now(),
      pg_catalog.now(), p_request_id
    )
    ON CONFLICT (author_id, request_id) DO NOTHING;

    -- Also handle simultaneous retries safely.

    SELECT id, post_id, parent_id, content
      INTO v_id, v_existing_post,
           v_existing_parent, v_existing_content
    FROM public.creatorverse_comments
    WHERE author_id = v_user
      AND request_id = p_request_id;

    IF NOT FOUND
      OR v_existing_post IS DISTINCT FROM p_post_id
      OR v_existing_parent IS DISTINCT FROM p_target_id THEN
      RAISE EXCEPTION 'Unable to verify comment request'
        USING ERRCODE = '42501';
    END IF;

    RETURN v_id;


  -- LIKE / UNLIKE COMMENT OR REPLY

  ELSIF v_action IN ('like_comment', 'unlike_comment') THEN

    IF p_target_id IS NULL OR v_text <> '' THEN
      RAISE EXCEPTION 'Invalid comment Like parameters'
        USING ERRCODE = '22023';
    END IF;

    IF NOT EXISTS (
      SELECT 1
      FROM public.creatorverse_comments AS c
      WHERE c.id = p_target_id
        AND c.post_id = p_post_id
        AND NOT c.is_deleted
        AND creatorverse_private.read_comment(c.id)
    ) THEN
      RAISE EXCEPTION 'Comment unavailable or inaccessible'
        USING ERRCODE = '42501';
    END IF;

    IF v_action = 'like_comment' THEN

      INSERT INTO public.creatorverse_comment_likes
        (comment_id, user_id, created_at)
      VALUES (
        p_target_id,
        v_user,
        pg_catalog.now()
      )
      ON CONFLICT DO NOTHING;

    ELSE

      DELETE FROM public.creatorverse_comment_likes
      WHERE comment_id = p_target_id
        AND user_id = v_user;

    END IF;

    RETURN p_target_id;


  -- EDIT OWN COMMENT OR REPLY

  ELSIF v_action = 'edit_comment' THEN

    IF p_target_id IS NULL
      OR pg_catalog.length(v_text) NOT BETWEEN 1 AND 1200
    THEN
      RAISE EXCEPTION 'Invalid comment edit'
        USING ERRCODE = '22023';
    END IF;

    UPDATE public.creatorverse_comments
    SET
      content = v_text,
      updated_at = pg_catalog.now()
    WHERE id = p_target_id
      AND post_id = p_post_id
      AND author_id = v_user
      AND NOT is_deleted;

    IF NOT FOUND THEN
      RAISE EXCEPTION 'Comment not found or unauthorized'
        USING ERRCODE = '42501';
    END IF;

    RETURN p_target_id;


  -- DELETE OWN COMMENT

  ELSIF v_action = 'delete_comment' THEN

    IF p_target_id IS NULL OR v_text <> '' THEN
      RAISE EXCEPTION 'Invalid comment deletion'
        USING ERRCODE = '22023';
    END IF;

    UPDATE public.creatorverse_comments
    SET
      is_deleted = true,
      content = '',
      updated_at = pg_catalog.now()
    WHERE id = p_target_id
      AND post_id = p_post_id
      AND author_id = v_user
      AND NOT is_deleted;

    IF NOT FOUND THEN
      RAISE EXCEPTION
        'Comment not found or unauthorized'
        USING ERRCODE = '42501';
    END IF;

    RETURN p_target_id;

  -- SHARES: EXISTING PROTECTION PRESERVED

  ELSIF v_action = 'share' THEN

    IF p_target_id IS NOT NULL
      OR pg_catalog.length(v_text) > 1000
      OR p_request_id IS NULL THEN
      RAISE EXCEPTION 'Invalid share parameters'
        USING ERRCODE = '22023';
    END IF;

    SELECT id, post_id
      INTO v_id, v_existing_post
    FROM public.creatorverse_shares
    WHERE sharer_id = v_user
      AND request_id = p_request_id;

    IF FOUND THEN
      IF v_existing_post IS DISTINCT FROM p_post_id THEN
        RAISE EXCEPTION
          'Request ID already used for another post'
          USING ERRCODE = '22023';
      END IF;

      RETURN v_id;
    END IF;

    IF (
      SELECT pg_catalog.count(*)
      FROM public.creatorverse_shares
      WHERE sharer_id = v_user
        AND created_at >
          pg_catalog.now() - interval '1 hour'
    ) >= 15 THEN
      RAISE EXCEPTION 'Share rate limit reached'
        USING ERRCODE = '42501';
    END IF;

    v_id := pg_catalog.gen_random_uuid();

    INSERT INTO public.creatorverse_shares (
      id, post_id, sharer_id,
      caption, request_id, created_at
    )
    VALUES (
      v_id, p_post_id, v_user,
      v_text, p_request_id, pg_catalog.now()
    )
    ON CONFLICT DO NOTHING;

    SELECT id, post_id
      INTO v_id, v_existing_post
    FROM public.creatorverse_shares
    WHERE sharer_id = v_user
      AND request_id = p_request_id;

    IF v_id IS NULL
      OR v_existing_post IS DISTINCT FROM p_post_id THEN
      RAISE EXCEPTION
        'Unable to verify share or request ID collision'
        USING ERRCODE = '42501';
    END IF;

    RETURN v_id;

  END IF;

  RAISE EXCEPTION 'Unknown interaction action'
    USING ERRCODE = '22023';

END;
$function$;

CREATE OR REPLACE FUNCTION public.creatorverse_update_page_details(p_page_id uuid, p_name text, p_category text, p_short_bio text, p_description text)
 RETURNS uuid
 LANGUAGE plpgsql
 SECURITY DEFINER
 SET search_path TO ''
AS $function$
DECLARE
  v_updated_id uuid;
BEGIN
  IF NOT COALESCE(
    public.creatorverse_details_can_edit(p_page_id),
    false
  ) THEN
    RAISE EXCEPTION 'Page editing is not permitted'
      USING ERRCODE = '42501';
  END IF;

  IF p_name IS NULL
     OR p_category IS NULL
     OR p_short_bio IS NULL
     OR p_description IS NULL THEN
    RAISE EXCEPTION 'Missing required page information'
      USING ERRCODE = '22023';
  END IF;

  UPDATE public.creatorverse_pages
  SET
    name = pg_catalog.btrim(p_name),
    category = p_category,
    short_bio = p_short_bio,
    description = p_description
  WHERE id = p_page_id
    AND owner_id = auth.uid()
    AND status IN ('draft', 'active')
  RETURNING id INTO v_updated_id;

  IF v_updated_id IS NULL THEN
    RAISE EXCEPTION 'Page editing is not permitted'
      USING ERRCODE = '42501';
  END IF;

  RETURN v_updated_id;
END;
$function$;

CREATE OR REPLACE FUNCTION public.creatorverse_update_page_profile(p_page_id uuid, p_name text, p_category text, p_short_bio text, p_description text, p_website text, p_public_email text, p_location text, p_social_links jsonb)
 RETURNS uuid
 LANGUAGE plpgsql
 SECURITY DEFINER
 SET search_path TO ''
AS $function$
DECLARE
  v_id uuid;
  v_link jsonb;
BEGIN
  -- Dedicated switch, valid user and ownership check.
  IF NOT COALESCE(
    public.creatorverse_details_can_edit(p_page_id),
    false
  ) THEN
    RAISE EXCEPTION 'Page editing is not permitted'
      USING ERRCODE = '42501';
  END IF;

  -- Required fields.
  IF p_name IS NULL
     OR p_category IS NULL
     OR p_short_bio IS NULL
     OR p_description IS NULL
     OR p_location IS NULL
     OR p_social_links IS NULL THEN
    RAISE EXCEPTION 'Missing page information'
      USING ERRCODE = '22023';
  END IF;

  -- Only an array containing up to eight links.
  IF jsonb_typeof(p_social_links) IS DISTINCT FROM 'array' THEN
    RAISE EXCEPTION 'Invalid social links'
      USING ERRCODE = '22023';
  END IF;

  IF jsonb_array_length(p_social_links) > 8 THEN
    RAISE EXCEPTION 'Too many social links'
      USING ERRCODE = '22023';
  END IF;

  -- Validate each link, including direct RPC requests.
  FOR v_link IN
    SELECT value
    FROM jsonb_array_elements(p_social_links)
  LOOP
    IF jsonb_typeof(v_link) <> 'object'
       OR jsonb_typeof(v_link->'label') <> 'string'
       OR jsonb_typeof(v_link->'url') <> 'string'
       OR length(btrim(v_link->>'label')) NOT BETWEEN 1 AND 60
       OR length(v_link->>'url') > 2048
       OR (v_link->>'url') !~
          '^https?://[^[:space:]@/]+([/?#][^[:space:]]*)?$'
    THEN
      RAISE EXCEPTION 'Invalid social link'
        USING ERRCODE = '22023';
    END IF;
  END LOOP;

  -- Existing table constraints validate names,
  -- categories, descriptions, email and website.

  UPDATE public.creatorverse_pages
  SET
    name = btrim(p_name),
    category = p_category,
    short_bio = p_short_bio,
    description = p_description,
    website = p_website,
    public_email = p_public_email,
    location = p_location,
    social_links = p_social_links
  WHERE id = p_page_id
    AND owner_id = auth.uid()
    AND status IN ('draft', 'active')
  RETURNING id INTO v_id;

  IF v_id IS NULL THEN
    RAISE EXCEPTION 'Page editing is not permitted'
      USING ERRCODE = '42501';
  END IF;

  RETURN v_id;
END;
$function$;

CREATE OR REPLACE FUNCTION public.groups_avatar_command(p_action text, p_group_id uuid, p_path text DEFAULT NULL::text)
 RETURNS jsonb
 LANGUAGE plpgsql
 SECURITY DEFINER
 SET search_path TO ''
AS $function$
DECLARE
  uid uuid := auth.uid();
  new_path text;
BEGIN
  IF uid IS NULL
     OR NOT parapost_groups_private.enabled()
     OR NOT EXISTS (
       SELECT 1
       FROM public.groups
       WHERE id = p_group_id
         AND owner_id = uid
     )
  THEN
    RAISE EXCEPTION 'Not authorized';
  END IF;

  IF p_action = 'reserve' THEN
    -- Limit outstanding recent upload reservations.
    IF (
      SELECT count(*)
      FROM parapost_groups_private.avatar_uploads
      WHERE group_id = p_group_id
        AND uploader_id = uid
        AND created_at > now() - interval '1 hour'
    ) >= 5 THEN
      RAISE EXCEPTION 'Too many unfinished avatar uploads';
    END IF;

    new_path :=
      p_group_id::text || '/avatars/' ||
      gen_random_uuid()::text;

    INSERT INTO parapost_groups_private.avatar_uploads (
      storage_path, group_id, uploader_id
    )
    VALUES (new_path, p_group_id, uid);

    RETURN jsonb_build_object('path', new_path);

  ELSIF p_action = 'save' THEN
    IF p_path IS NULL OR NOT EXISTS (
      SELECT 1
      FROM parapost_groups_private.avatar_uploads a
      JOIN storage.objects o
        ON o.bucket_id = 'group-media'
       AND o.name = a.storage_path
      WHERE a.storage_path = p_path
        AND a.group_id = p_group_id
        AND a.uploader_id = uid
        AND a.created_at > now() - interval '1 hour'
        AND o.metadata->>'mimetype'
          IN ('image/jpeg','image/png','image/webp')
        AND (o.metadata->>'size')::bigint
          BETWEEN 1 AND 5242880
    ) THEN
      RAISE EXCEPTION 'Invalid or incomplete avatar upload';
    END IF;

    UPDATE public.groups
    SET avatar_storage_path = p_path,
        updated_at = now()
    WHERE id = p_group_id
      AND owner_id = uid;

    DELETE FROM parapost_groups_private.avatar_uploads
    WHERE storage_path = p_path;

    RETURN jsonb_build_object('path', p_path);

  ELSIF p_action = 'remove' THEN
    UPDATE public.groups
    SET avatar_storage_path = NULL,
        updated_at = now()
    WHERE id = p_group_id
      AND owner_id = uid;

    RETURN jsonb_build_object('path', NULL);

  ELSE
    RAISE EXCEPTION 'Invalid avatar action';
  END IF;
END;
$function$;

CREATE OR REPLACE FUNCTION public.groups_cleanup_batch()
 RETURNS SETOF text
 LANGUAGE plpgsql
 SECURITY DEFINER
 SET search_path TO ''
AS $function$
DECLARE target record;
BEGIN
 FOR target IN SELECT g.id FROM public.groups g
 WHERE EXISTS(SELECT 1 FROM public.group_posts p WHERE p.group_id=g.id AND NOT p.published AND p.created_at<now()-interval '24 hours')
    OR EXISTS(SELECT 1 FROM parapost_groups_private.cover_uploads c WHERE c.group_id=g.id AND c.created_at<now()-interval '24 hours')
 ORDER BY g.id LIMIT 100 FOR UPDATE LOOP
   DELETE FROM public.group_posts WHERE group_id=target.id AND NOT published AND created_at<now()-interval '24 hours';
   DELETE FROM parapost_groups_private.cover_uploads WHERE group_id=target.id AND created_at<now()-interval '24 hours';
 END LOOP;
 RETURN QUERY SELECT q.storage_path FROM parapost_groups_private.media_cleanup q
 WHERE q.queued_at<now()-interval '24 hours'
   AND NOT EXISTS(SELECT 1 FROM public.groups g WHERE g.cover_storage_path=q.storage_path)
   AND NOT EXISTS(SELECT 1 FROM public.group_post_media m WHERE m.storage_path=q.storage_path)
   AND NOT EXISTS(SELECT 1 FROM parapost_groups_private.cover_uploads c WHERE c.storage_path=q.storage_path)
 ORDER BY q.queued_at LIMIT 100;
END; $function$;

CREATE OR REPLACE FUNCTION public.groups_comment_like_stats(p_group_id uuid, p_post_id uuid)
 RETURNS TABLE(comment_id uuid, like_count bigint, liked boolean)
 LANGUAGE sql
 STABLE SECURITY DEFINER
 SET search_path TO ''
AS $function$
  WITH visible_comments AS (
    SELECT c.id, c.created_at
    FROM public.group_post_comments c
    JOIN public.group_posts p ON p.id = c.post_id
    WHERE c.post_id = p_post_id
      AND p.group_id = p_group_id
      AND p.published
      AND NOT p.is_deleted
      AND auth.uid() IS NOT NULL
      AND parapost_groups_private.allowed(p_group_id, 'read')
    ORDER BY c.created_at, c.id
    LIMIT 100
  )
  SELECT
    c.id,
    (
      SELECT count(*)
      FROM public.group_comment_likes l
      WHERE l.comment_id = c.id
    ),
    EXISTS (
      SELECT 1
      FROM public.group_comment_likes l
      WHERE l.comment_id = c.id
        AND l.user_id = auth.uid()
    )
  FROM visible_comments c
  ORDER BY c.created_at, c.id;
$function$;

CREATE OR REPLACE FUNCTION public.groups_directory(p_search text DEFAULT ''::text, p_filter text DEFAULT 'all'::text, p_group_id uuid DEFAULT NULL::uuid, p_offset integer DEFAULT 0)
 RETURNS jsonb
 LANGUAGE sql
 STABLE SECURITY DEFINER
 SET search_path TO ''
AS $function$
 SELECT coalesce(jsonb_agg(row_to_json(x)),'[]'::jsonb) FROM (
   SELECT g.*,coalesce(pr.display_name,pr.username,'Member') AS owner_name,
     m.status AS membership_status,m.role AS membership_role,
     (SELECT count(*) FROM public.group_members c WHERE c.group_id=g.id AND c.status='approved' AND NOT parapost_groups_private.blocked(c.user_id,g.owner_id)) AS member_count
   FROM public.groups g LEFT JOIN public.group_members m ON m.group_id=g.id AND m.user_id=auth.uid()
   LEFT JOIN public.profiles pr ON pr.id=g.owner_id
   WHERE parapost_groups_private.allowed(g.id,'directory') AND (p_group_id IS NULL OR g.id=p_group_id)
     AND (p_filter='all' OR p_filter=g.visibility OR (p_filter='mine' AND m.status='approved'))
     AND (g.name || ' ' || g.description) ILIKE '%' || left(coalesce(p_search,''),100) || '%'
   ORDER BY g.created_at DESC,g.id LIMIT 24 OFFSET greatest(0,least(coalesce(p_offset,0),100000))
 ) x;
$function$;

CREATE OR REPLACE FUNCTION public.groups_interact(p_group_id uuid, p_post_id uuid, p_action text, p_content text DEFAULT NULL::text, p_parent_id uuid DEFAULT NULL::uuid)
 RETURNS jsonb
 LANGUAGE plpgsql
 SECURITY DEFINER
 SET search_path TO ''
AS $function$
DECLARE
  v_uid uuid := auth.uid();
  v_group public.groups;
  v_post public.group_posts;
  v_parent public.group_post_comments;
  v_id uuid;
BEGIN
  IF v_uid IS NULL THEN
    RAISE EXCEPTION 'Sign in required' USING ERRCODE='42501';
  END IF;

  SELECT * INTO v_group
  FROM public.groups
  WHERE id=p_group_id
  FOR UPDATE;

  IF NOT FOUND OR NOT parapost_groups_private.allowed(p_group_id,'member') THEN
    RAISE EXCEPTION 'Approved membership required'
      USING ERRCODE='42501';
  END IF;

  SELECT * INTO v_post
  FROM public.group_posts
  WHERE id=p_post_id
    AND group_id=p_group_id
    AND published
    AND NOT is_deleted;

  IF NOT FOUND THEN
    RAISE EXCEPTION 'Post unavailable' USING ERRCODE='42501';
  END IF;

  IF p_action IN ('like','comment','reply','like_comment')
     AND parapost_groups_private.blocked(v_uid,v_post.author_id) THEN
    RAISE EXCEPTION 'Interaction unavailable'
      USING ERRCODE='42501';
  END IF;

  IF p_action='like' THEN

    INSERT INTO public.group_post_likes(post_id,user_id)
    VALUES(p_post_id,v_uid)
    ON CONFLICT (post_id,user_id) DO NOTHING;

    RETURN jsonb_build_object('liked',true);

  ELSIF p_action='unlike' THEN

    DELETE FROM public.group_post_likes
    WHERE post_id=p_post_id AND user_id=v_uid;

    RETURN jsonb_build_object('liked',false);

  ELSIF p_action IN ('like_comment','unlike_comment') THEN

    IF p_parent_id IS NULL THEN
      RAISE EXCEPTION 'Comment required';
    END IF;

    SELECT * INTO v_parent
    FROM public.group_post_comments
    WHERE id=p_parent_id AND post_id=p_post_id;

    IF NOT FOUND THEN
      RAISE EXCEPTION 'Comment unavailable'
        USING ERRCODE='42501';
    END IF;

    IF p_action='like_comment' THEN

      IF parapost_groups_private.blocked(v_uid,v_parent.author_id) THEN
        RAISE EXCEPTION 'Comment Like unavailable'
          USING ERRCODE='42501';
      END IF;

      INSERT INTO public.group_comment_likes
        (comment_id,post_id,user_id)
      VALUES (p_parent_id,p_post_id,v_uid)
      ON CONFLICT (comment_id,user_id) DO NOTHING;

      RETURN jsonb_build_object('liked',true);

    ELSE

      DELETE FROM public.group_comment_likes
      WHERE comment_id=p_parent_id AND user_id=v_uid;

      RETURN jsonb_build_object('liked',false);

    END IF;

  ELSIF p_action IN ('comment','reply') THEN

    IF p_content IS NULL OR
       char_length(btrim(p_content)) NOT BETWEEN 1 AND 1200 THEN
      RAISE EXCEPTION 'Invalid comment';
    END IF;

    IF p_action='comment' AND p_parent_id IS NOT NULL THEN
      RAISE EXCEPTION 'Invalid comment parent';
    END IF;

    IF p_action='reply' THEN

      IF p_parent_id IS NULL THEN
        RAISE EXCEPTION 'Reply parent required';
      END IF;

      SELECT * INTO v_parent
      FROM public.group_post_comments
      WHERE id=p_parent_id AND post_id=p_post_id;

      IF NOT FOUND OR v_parent.parent_id IS NOT NULL THEN
        RAISE EXCEPTION 'Invalid reply parent';
      END IF;

      IF parapost_groups_private.blocked(v_uid,v_parent.author_id) THEN
        RAISE EXCEPTION 'Reply unavailable'
          USING ERRCODE='42501';
      END IF;

    END IF;

    INSERT INTO public.group_post_comments
      (post_id,author_id,parent_id,content)
    VALUES (p_post_id,v_uid,p_parent_id,btrim(p_content))
    RETURNING id INTO v_id;

    RETURN jsonb_build_object('id',v_id);

  ELSE
    RAISE EXCEPTION 'Unknown interaction action';
  END IF;
END;
$function$;

CREATE OR REPLACE FUNCTION public.groups_manage_comment(p_group_id uuid, p_post_id uuid, p_comment_id uuid, p_action text, p_content text DEFAULT NULL::text)
 RETURNS jsonb
 LANGUAGE plpgsql
 SECURITY DEFINER
 SET search_path TO ''
AS $function$
DECLARE
  v_uid uuid := auth.uid();
  v_owner uuid;
  v_author uuid;
  v_content text;
BEGIN
  IF v_uid IS NULL THEN
    RAISE EXCEPTION 'Sign in required'
      USING ERRCODE = '42501';
  END IF;

  IF p_action NOT IN ('edit_comment', 'delete_comment') THEN
    RAISE EXCEPTION 'Invalid action';
  END IF;

  SELECT g.owner_id INTO v_owner
  FROM public.groups g
  WHERE g.id = p_group_id;

  IF NOT FOUND THEN
    RAISE EXCEPTION 'Group unavailable'
      USING ERRCODE = '42501';
  END IF;

  IF NOT parapost_groups_private.allowed(
    p_group_id, 'member'
  ) THEN
    RAISE EXCEPTION 'Approved membership required'
      USING ERRCODE = '42501';
  END IF;

  PERFORM 1
  FROM public.group_posts p
  WHERE p.id = p_post_id
    AND p.group_id = p_group_id
    AND p.published
    AND NOT p.is_deleted;

  IF NOT FOUND THEN
    RAISE EXCEPTION 'Post unavailable'
      USING ERRCODE = '42501';
  END IF;

  SELECT c.author_id INTO v_author
  FROM public.group_post_comments c
  WHERE c.id = p_comment_id
    AND c.post_id = p_post_id
  FOR UPDATE;

  IF NOT FOUND THEN
    RAISE EXCEPTION 'Comment unavailable'
      USING ERRCODE = '42501';
  END IF;

  IF p_action = 'edit_comment' THEN

    IF v_uid IS DISTINCT FROM v_author THEN
      RAISE EXCEPTION 'Only the author can edit'
        USING ERRCODE = '42501';
    END IF;

    v_content := btrim(p_content);

    IF v_content IS NULL OR
       char_length(v_content) NOT BETWEEN 1 AND 1200 THEN
      RAISE EXCEPTION 'Invalid comment';
    END IF;

    UPDATE public.group_post_comments
    SET content = v_content,
        updated_at = now()
    WHERE id = p_comment_id
      AND post_id = p_post_id;

    RETURN jsonb_build_object('edited', true);

  ELSE

    IF v_uid IS DISTINCT FROM v_author
       AND v_uid IS DISTINCT FROM v_owner THEN
      RAISE EXCEPTION 'Not permitted to delete comment'
        USING ERRCODE = '42501';
    END IF;

    DELETE FROM public.group_comment_likes
    WHERE comment_id IN (
      SELECT c.id
      FROM public.group_post_comments c
      WHERE c.post_id = p_post_id
        AND (
          c.id = p_comment_id
          OR c.parent_id = p_comment_id
        )
    );

    DELETE FROM public.group_post_comments
    WHERE id = p_comment_id
      AND post_id = p_post_id;

    RETURN jsonb_build_object('deleted', true);

  END IF;
END;
$function$;

CREATE OR REPLACE FUNCTION public.groups_remove_share(p_share_id uuid, p_kind text)
 RETURNS jsonb
 LANGUAGE plpgsql
 SECURITY DEFINER
 SET search_path TO ''
AS $function$
DECLARE
  v_uid uuid := auth.uid();
  v_owner uuid;
  v_target uuid;
  v_target_type text;
  v_shared_at timestamptz;
BEGIN
  IF v_uid IS NULL THEN
    RAISE EXCEPTION 'Sign in required'
      USING ERRCODE = '42501';
  END IF;

  IF p_share_id IS NULL
     OR p_kind NOT IN ('post', 'promotion')
     OR p_kind IS NULL
  THEN
    RAISE EXCEPTION 'Invalid share'
      USING ERRCODE = '42501';
  END IF;

  IF p_kind = 'post' THEN
    SELECT user_id, post_id, created_at
    INTO v_owner, v_target, v_shared_at
    FROM public.group_post_shares
    WHERE id = p_share_id
    FOR UPDATE;

    v_target_type := 'post';

  ELSE
    SELECT
      user_id,
      COALESCE(post_id, group_id),
      created_at,
      CASE WHEN post_id IS NULL THEN 'group' ELSE 'post' END
    INTO v_owner, v_target, v_shared_at, v_target_type
    FROM public.group_promotional_shares
    WHERE id = p_share_id
    FOR UPDATE;
  END IF;

  IF NOT FOUND OR v_owner IS DISTINCT FROM v_uid THEN
    RAISE EXCEPTION 'Share unavailable'
      USING ERRCODE = '42501';
  END IF;

  -- Retain the original share date for cooldown enforcement.
  INSERT INTO parapost_groups_private.group_share_cooldowns (
    user_id, target_type, target_id, last_shared_at
  )
  VALUES (
    v_uid, v_target_type, v_target, v_shared_at
  )
  ON CONFLICT (user_id, target_type, target_id)
  DO UPDATE SET
    last_shared_at = GREATEST(
      group_share_cooldowns.last_shared_at,
      EXCLUDED.last_shared_at
    );

  IF p_kind = 'post' THEN
    DELETE FROM public.group_post_shares
    WHERE id = p_share_id AND user_id = v_uid;
  ELSE
    DELETE FROM public.group_promotional_shares
    WHERE id = p_share_id AND user_id = v_uid;
  END IF;

  RETURN jsonb_build_object(
    'deleted', true,
    'share_id', p_share_id
  );
END;
$function$;

CREATE OR REPLACE FUNCTION public.groups_share_post(p_group_id uuid, p_post_id uuid, p_caption text DEFAULT NULL::text)
 RETURNS jsonb
 LANGUAGE plpgsql
 SECURITY DEFINER
 SET search_path TO ''
AS $function$
DECLARE
  v_uid uuid := auth.uid();
  v_group public.groups%ROWTYPE;
  v_post public.group_posts%ROWTYPE;
  v_share_id uuid;
BEGIN
  IF v_uid IS NULL THEN
    RAISE EXCEPTION 'Sign in required'
      USING ERRCODE = '42501';
  END IF;

  -- Serialize sharing attempts across devices.
  SELECT * INTO v_group
  FROM public.groups
  WHERE id = p_group_id
  FOR UPDATE;

  IF NOT FOUND
     OR v_group.visibility <> 'public'
     OR NOT parapost_groups_private.allowed(
       p_group_id, 'member'
     ) THEN
    RAISE EXCEPTION 'Public Group membership required'
      USING ERRCODE = '42501';
  END IF;

  SELECT * INTO v_post
  FROM public.group_posts
  WHERE id = p_post_id
    AND group_id = p_group_id
    AND published
    AND NOT is_deleted;

  IF NOT FOUND THEN
    RAISE EXCEPTION 'Group post unavailable'
      USING ERRCODE = '42501';
  END IF;

  IF parapost_groups_private.blocked(
    v_uid, v_post.author_id
  ) THEN
    RAISE EXCEPTION 'Sharing unavailable'
      USING ERRCODE = '42501';
  END IF;

  IF p_caption IS NOT NULL
     AND char_length(p_caption) > 1000 THEN
    RAISE EXCEPTION 'Share caption too long';
  END IF;

  PERFORM parapost_groups_private.record_share_cooldown(
    'post', p_post_id
  );

  INSERT INTO public.group_post_shares
    (group_id, post_id, user_id, caption)
  VALUES (
    p_group_id, p_post_id, v_uid,
    NULLIF(btrim(p_caption), '')
  )
  RETURNING id INTO v_share_id;

  RETURN jsonb_build_object('id', v_share_id);
END;
$function$;

CREATE OR REPLACE FUNCTION public.groups_shared_feed(p_sharer_id uuid DEFAULT NULL::uuid, p_limit integer DEFAULT 60, p_offset integer DEFAULT 0)
 RETURNS TABLE(share_id uuid, group_id uuid, post_id uuid, sharer_id uuid, caption text, shared_at timestamp with time zone, group_name text, original_author_id uuid, original_content text, original_created_at timestamp with time zone, media jsonb)
 LANGUAGE sql
 STABLE SECURITY DEFINER
 SET search_path TO ''
AS $function$
  SELECT
    s.id,
    s.group_id,
    s.post_id,
    s.user_id,
    s.caption,
    s.created_at,
    g.name,
    p.author_id,
    p.content,
    p.created_at,
    COALESCE(media_items.items, '[]'::jsonb)
  FROM public.group_post_shares s
  JOIN public.groups g
    ON g.id = s.group_id
  JOIN public.group_posts p
    ON p.id = s.post_id
   AND p.group_id = s.group_id
  LEFT JOIN LATERAL (
    SELECT jsonb_agg(
      jsonb_build_object(
        'media_type', m.media_type,
        'storage_path', m.storage_path,
        'display_order', m.display_order
      )
      ORDER BY m.display_order, m.id
    ) AS items
    FROM public.group_post_media m
    WHERE m.post_id = p.id
      AND m.group_id = g.id
  ) media_items ON true
  WHERE auth.uid() IS NOT NULL
    AND g.visibility = 'public'
    AND p.published
    AND NOT p.is_deleted
    AND parapost_groups_private.allowed(
      g.id, 'read'
    )
    AND NOT parapost_groups_private.blocked(
      auth.uid(), s.user_id
    )
    AND NOT parapost_groups_private.blocked(
      auth.uid(), p.author_id
    )
    AND (
      p_sharer_id IS NULL
      OR s.user_id = p_sharer_id
    )
  ORDER BY s.created_at DESC, s.id DESC
  LIMIT LEAST(GREATEST(COALESCE(p_limit,60),1),100)
  OFFSET LEAST(GREATEST(COALESCE(p_offset,0),0),10000);
$function$;

CREATE OR REPLACE FUNCTION public.groups_shared_post_interact(p_share_id uuid, p_action text, p_content text DEFAULT NULL::text, p_parent_id uuid DEFAULT NULL::uuid)
 RETURNS jsonb
 LANGUAGE plpgsql
 SECURITY DEFINER
 SET search_path TO ''
AS $function$
DECLARE
  v_uid uuid := auth.uid();
  v_share public.group_post_shares%ROWTYPE;
  v_group public.groups%ROWTYPE;
  v_post public.group_posts%ROWTYPE;
  v_parent public.group_post_share_comments%ROWTYPE;
  v_id uuid;
  v_limit_kind text;
  v_limit_max integer;
  v_limit_now timestamptz;
  v_limit_count integer;
BEGIN
  IF v_uid IS NULL THEN
    RAISE EXCEPTION 'Sign in required'
      USING ERRCODE = '42501';
  END IF;

  SELECT * INTO v_share
  FROM public.group_post_shares
  WHERE id = p_share_id;

  IF NOT FOUND THEN
    RAISE EXCEPTION 'Share unavailable'
      USING ERRCODE = '42501';
  END IF;

  SELECT * INTO v_group
  FROM public.groups
  WHERE id = v_share.group_id
  FOR SHARE;

  IF NOT FOUND
     OR v_group.visibility <> 'public'
     OR NOT parapost_groups_private.allowed(
       v_share.group_id, 'read'
     )
  THEN
    RAISE EXCEPTION 'Group unavailable'
      USING ERRCODE = '42501';
  END IF;

  SELECT * INTO v_post
  FROM public.group_posts
  WHERE id = v_share.post_id
    AND group_id = v_share.group_id
    AND NOT is_deleted;

  IF NOT FOUND THEN
    RAISE EXCEPTION 'Post unavailable'
      USING ERRCODE = '42501';
  END IF;

  IF parapost_groups_private.blocked(v_uid, v_share.user_id)
     OR parapost_groups_private.blocked(
       v_uid, v_post.author_id
     )
  THEN
    RAISE EXCEPTION 'Interaction unavailable'
      USING ERRCODE = '42501';
  END IF;

  -- Atomic per-user limits across all shared Group posts.
  IF p_action IN ('comment', 'reply') THEN
    v_limit_kind := 'comment';
    v_limit_max := 10;
  ELSIF p_action IN (
    'like', 'unlike', 'like_comment', 'unlike_comment'
  ) THEN
    v_limit_kind := 'reaction';
    v_limit_max := 60;
  END IF;

  IF v_limit_kind IS NOT NULL THEN
    v_limit_now := clock_timestamp();

    INSERT INTO parapost_groups_private.shared_post_interaction_rate_limits AS rl
      (user_id, action_kind, window_started_at, action_count)
    VALUES (v_uid, v_limit_kind, v_limit_now, 1)
    ON CONFLICT (user_id, action_kind)
    DO UPDATE SET
      window_started_at = CASE
        WHEN rl.window_started_at <=
             v_limit_now - interval '60 seconds'
        THEN v_limit_now
        ELSE rl.window_started_at
      END,
      action_count = CASE
        WHEN rl.window_started_at <=
             v_limit_now - interval '60 seconds'
        THEN 1
        ELSE rl.action_count + 1
      END
    WHERE rl.window_started_at <=
          v_limit_now - interval '60 seconds'
       OR rl.action_count < v_limit_max
    RETURNING action_count INTO v_limit_count;

    IF NOT FOUND THEN
      RAISE EXCEPTION 'Rate limit exceeded. Try again shortly.'
        USING ERRCODE = 'P0001';
    END IF;
  END IF;

  IF p_action = 'like' THEN
    INSERT INTO public.group_post_share_likes
      (share_id, user_id)
    VALUES (p_share_id, v_uid)
    ON CONFLICT (share_id, user_id) DO NOTHING;

    RETURN jsonb_build_object('liked', true);

  ELSIF p_action = 'unlike' THEN
    DELETE FROM public.group_post_share_likes
    WHERE share_id = p_share_id
      AND user_id = v_uid;

    RETURN jsonb_build_object('liked', false);

  ELSIF p_action = 'comment' THEN
    IF p_parent_id IS NOT NULL
       OR p_content IS NULL
       OR char_length(btrim(p_content)) NOT BETWEEN 1 AND 1200
    THEN
      RAISE EXCEPTION 'Invalid comment';
    END IF;

    INSERT INTO public.group_post_share_comments
      (share_id, author_id, content)
    VALUES (p_share_id, v_uid, btrim(p_content))
    RETURNING id INTO v_id;

    RETURN jsonb_build_object('id', v_id);

  ELSIF p_action IN (
    'reply', 'like_comment', 'unlike_comment',
    'edit_comment', 'delete_comment'
  ) THEN
    IF p_parent_id IS NULL THEN
      RAISE EXCEPTION 'Comment required';
    END IF;

    SELECT * INTO v_parent
    FROM public.group_post_share_comments
    WHERE id = p_parent_id
      AND share_id = p_share_id;

    IF NOT FOUND THEN
      RAISE EXCEPTION 'Comment unavailable'
        USING ERRCODE = '42501';
    END IF;

    IF p_action = 'reply' THEN
      IF v_parent.parent_id IS NOT NULL
         OR parapost_groups_private.blocked(
           v_uid, v_parent.author_id
         )
         OR p_content IS NULL
         OR char_length(btrim(p_content)) NOT BETWEEN 1 AND 1200
      THEN
        RAISE EXCEPTION 'Reply unavailable'
          USING ERRCODE = '42501';
      END IF;

      INSERT INTO public.group_post_share_comments
        (share_id, author_id, parent_id, content)
      VALUES (
        p_share_id, v_uid, p_parent_id, btrim(p_content)
      )
      RETURNING id INTO v_id;

      RETURN jsonb_build_object('id', v_id);

    ELSIF p_action = 'like_comment' THEN
      IF parapost_groups_private.blocked(
        v_uid, v_parent.author_id
      )
      OR (
        v_parent.parent_id IS NOT NULL
        AND NOT EXISTS (
          SELECT 1
          FROM public.group_post_share_comments parent
          WHERE parent.id = v_parent.parent_id
            AND parent.share_id = p_share_id
            AND parent.parent_id IS NULL
            AND NOT parapost_groups_private.blocked(
              v_uid, parent.author_id
            )
        )
      ) THEN
        RAISE EXCEPTION 'Comment Like unavailable'
          USING ERRCODE = '42501';
      END IF;

      INSERT INTO public.group_post_share_comment_likes
        (comment_id, share_id, user_id)
      VALUES (p_parent_id, p_share_id, v_uid)
      ON CONFLICT (comment_id, user_id) DO NOTHING;

      RETURN jsonb_build_object('liked', true);

    ELSIF p_action = 'unlike_comment' THEN
      DELETE FROM public.group_post_share_comment_likes
      WHERE comment_id = p_parent_id
        AND share_id = p_share_id
        AND user_id = v_uid;

      RETURN jsonb_build_object('liked', false);

    ELSIF p_action = 'edit_comment' THEN
      IF v_parent.author_id <> v_uid
         OR p_content IS NULL
         OR char_length(btrim(p_content)) NOT BETWEEN 1 AND 1200
      THEN
        RAISE EXCEPTION 'Edit unavailable'
          USING ERRCODE = '42501';
      END IF;

      UPDATE public.group_post_share_comments
      SET content = btrim(p_content),
          updated_at = now()
      WHERE id = p_parent_id
        AND share_id = p_share_id
        AND author_id = v_uid;

      RETURN jsonb_build_object('updated', true);

    ELSIF p_action = 'delete_comment' THEN
      IF v_parent.author_id <> v_uid THEN
        RAISE EXCEPTION 'Delete unavailable'
          USING ERRCODE = '42501';
      END IF;

      DELETE FROM public.group_post_share_comments
      WHERE id = p_parent_id
        AND share_id = p_share_id
        AND author_id = v_uid;

      RETURN jsonb_build_object('deleted', true);
    END IF;
  END IF;

  RAISE EXCEPTION 'Unknown interaction action';
END;
$function$;

CREATE OR REPLACE FUNCTION public.groups_shared_post_stats(p_share_id uuid)
 RETURNS jsonb
 LANGUAGE plpgsql
 STABLE SECURITY DEFINER
 SET search_path TO ''
AS $function$
DECLARE
  v_uid uuid := auth.uid();
  v_group_id uuid;
  v_sharer_id uuid;
  v_author_id uuid;
  v_comments jsonb;
BEGIN
  IF v_uid IS NULL THEN
    RAISE EXCEPTION 'Sign in required'
      USING ERRCODE = '42501';
  END IF;

  SELECT s.group_id, s.user_id, p.author_id
  INTO v_group_id, v_sharer_id, v_author_id
  FROM public.group_post_shares s
  JOIN public.groups g ON g.id = s.group_id
  JOIN public.group_posts p
    ON p.id = s.post_id AND p.group_id = s.group_id
  WHERE s.id = p_share_id
    AND g.visibility = 'public'
    AND NOT p.is_deleted;

  IF NOT FOUND THEN
    RAISE EXCEPTION 'Share unavailable'
      USING ERRCODE = '42501';
  END IF;

  IF NOT parapost_groups_private.allowed(v_group_id, 'read')
     OR parapost_groups_private.blocked(v_uid, v_sharer_id)
     OR parapost_groups_private.blocked(v_uid, v_author_id)
  THEN
    RAISE EXCEPTION 'Share unavailable'
      USING ERRCODE = '42501';
  END IF;

  SELECT COALESCE(
    jsonb_agg(
      jsonb_build_object(
        'id', c.id,
        'author_id', c.author_id,
        'parent_id', c.parent_id,
        'content', c.content,
        'created_at', c.created_at,
        'updated_at', c.updated_at,
        'like_count', c.like_count,
        'liked', c.liked
      )
      ORDER BY c.created_at, c.id
    ),
    '[]'::jsonb
  )
  INTO v_comments
  FROM (
    SELECT
      cm.*,
      (
        SELECT count(*)::integer
        FROM public.group_post_share_comment_likes cl
        WHERE cl.comment_id = cm.id
          AND NOT parapost_groups_private.blocked(
            v_uid, cl.user_id
          )
      ) AS like_count,
      EXISTS (
        SELECT 1
        FROM public.group_post_share_comment_likes cl
        WHERE cl.comment_id = cm.id
          AND cl.user_id = v_uid
      ) AS liked
    FROM public.group_post_share_comments cm
    WHERE cm.share_id = p_share_id
      AND NOT parapost_groups_private.blocked(
        v_uid, cm.author_id
      )
      AND (
        cm.parent_id IS NULL
        OR EXISTS (
          SELECT 1
          FROM public.group_post_share_comments parent
          WHERE parent.id = cm.parent_id
            AND parent.share_id = cm.share_id
            AND parent.parent_id IS NULL
            AND NOT parapost_groups_private.blocked(
              v_uid, parent.author_id
            )
        )
      )
    ORDER BY cm.created_at, cm.id
    LIMIT 150
  ) c;

  RETURN jsonb_build_object(
    'liked', EXISTS (
      SELECT 1
      FROM public.group_post_share_likes l
      WHERE l.share_id = p_share_id
        AND l.user_id = v_uid
    ),
    'like_count', (
      SELECT count(*)::integer
      FROM public.group_post_share_likes l
      WHERE l.share_id = p_share_id
        AND NOT parapost_groups_private.blocked(
          v_uid, l.user_id
        )
    ),
    'comment_count', (
      SELECT count(*)::integer
      FROM public.group_post_share_comments c
      WHERE c.share_id = p_share_id
        AND NOT parapost_groups_private.blocked(
          v_uid, c.author_id
        )
        AND (
          c.parent_id IS NULL
          OR EXISTS (
            SELECT 1
            FROM public.group_post_share_comments parent
            WHERE parent.id = c.parent_id
              AND parent.share_id = c.share_id
              AND parent.parent_id IS NULL
              AND NOT parapost_groups_private.blocked(
                v_uid, parent.author_id
              )
          )
        )
    ),
    'comments_limited', (
      SELECT count(*) > 150
      FROM public.group_post_share_comments c
      WHERE c.share_id = p_share_id
        AND NOT parapost_groups_private.blocked(
          v_uid, c.author_id
        )
        AND (
          c.parent_id IS NULL
          OR EXISTS (
            SELECT 1
            FROM public.group_post_share_comments parent
            WHERE parent.id = c.parent_id
              AND parent.share_id = c.share_id
              AND parent.parent_id IS NULL
              AND NOT parapost_groups_private.blocked(
                v_uid, parent.author_id
              )
          )
        )
    ),
    'comments', v_comments
  );
END;
$function$;
